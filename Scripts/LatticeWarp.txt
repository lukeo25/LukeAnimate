  function bounds(points){
    var xs=points.map(function(p){return p.x;}),ys=points.map(function(p){return p.y;});
    return{x1:Math.min.apply(null,xs),y1:Math.min.apply(null,ys),x2:Math.max.apply(null,xs),y2:Math.max.apply(null,ys)};
  }
  function center(points){var b=bounds(points);return{x:(b.x1+b.x2)/2,y:(b.y1+b.y2)/2};}
  function pointInPolygon(p,poly){
    var inside=false;
    for(var i=0,j=poly.length-1;i<poly.length;j=i++){
      var a=poly[i],b=poly[j];
      if(((a.y>p.y)!==(b.y>p.y))&&(p.x<(b.x-a.x)*(p.y-a.y)/(b.y-a.y||1e-9)+a.x))inside=!inside;
    }
    return inside;
  }
  function restBoundary(points){
    var b=bounds(points),c=center(points),hw=Math.max(1,(b.x2-b.x1)/2),hh=Math.max(1,(b.y2-b.y1)/2);
    return points.map(function(p){
      var dx=p.x-c.x,dy=p.y-c.y,scale=1/Math.max(Math.abs(dx)/hw,Math.abs(dy)/hh,1e-6);
      return{x:c.x+dx*scale,y:c.y+dy*scale};
    });
  }
  function cageData(cage,time){
    var baseTarget=(cage.points||[]).map(function(p){return{x:num(p.x,0),y:num(p.y,0)};});
    var baseRest=Array.isArray(cage.restPoints)&&cage.restPoints.length===baseTarget.length?cage.restPoints:restBoundary(baseTarget);
    var effective=API.getEffectiveObject?API.getEffectiveObject(cage,time===undefined?API.time:time):cage;
    var target=baseTarget.map(function(p){return localToStage(cage,effective,p);});
    var rest=baseRest.map(function(p){return localToStage(cage,effective,p);});
    return{target:target,rest:rest,box:bounds(rest)};
  }
  function warpPoint(cage,p,time){
    var data=cageData(cage,time),rest=data.rest,target=data.target,b=data.box;
    var margin=Math.max(24,Math.max(b.x2-b.x1,b.y2-b.y1)*.2);
    if(p.x<b.x1-margin||p.x>b.x2+margin||p.y<b.y1-margin||p.y>b.y2+margin)return{x:p.x,y:p.y};
    var sum=0,dx=0,dy=0;
    for(var i=0;i<rest.length;i++){
      var rx=p.x-rest[i].x,ry=p.y-rest[i].y,d2=rx*rx+ry*ry;
      var w=1/Math.pow(Math.max(16,d2),1.15);
      sum+=w;dx+=(target[i].x-rest[i].x)*w;dy+=(target[i].y-rest[i].y)*w;
    }
    if(!sum)return{x:p.x,y:p.y};
    var edgeX=Math.min((p.x-(b.x1-margin))/margin,(b.x2+margin-p.x)/margin,1);
    var edgeY=Math.min((p.y-(b.y1-margin))/margin,(b.y2+margin-p.y)/margin,1);
    var influence=clamp(Math.min(edgeX,edgeY),0,1);
    if(!pointInPolygon(p,rest))influence*=.45;
    return{x:p.x+dx/sum*influence,y:p.y+dy/sum*influence};
  }
  function transformFor(o,e){
    var clip=o.type==="clipinstance";
    var uniform=num(clip?e.scale:e.tscale,1),ax=num(clip?e.scaleX:e.tscaleX,1),ay=num(clip?e.scaleY:e.tscaleY,1);
    return{x:num(clip?e.x:e.tx,0),y:num(clip?e.y:e.ty,0),rotation:num(clip?e.rotation:e.trotation,0)*Math.PI/180,sx:uniform*ax,sy:uniform*ay};
  }
  function objectBounds(o){
    if(o.type==="image"||o.type==="video")return{x1:num(o.x,0),y1:num(o.y,0),x2:num(o.x,0)+Math.max(1,num(o.width,1)),y2:num(o.y,0)+Math.max(1,num(o.height,1))};
    var pts=o.points||[];
    if(!pts.length&&o.x1!==undefined)pts=[{x:o.x1,y:o.y1},{x:o.x2,y:o.y2}];
    if(!pts.length)return{x1:0,y1:0,x2:1,y2:1};
    return bounds(pts);
  }
  function localToStage(o,e,p){
    var b=objectBounds(o),cx=(b.x1+b.x2)/2,cy=(b.y1+b.y2)/2,t=transformFor(o,e),dx=(p.x-cx)*t.sx,dy=(p.y-cy)*t.sy,c=Math.cos(t.rotation),s=Math.sin(t.rotation);
    return{x:cx+t.x+dx*c-dy*s,y:cy+t.y+dx*s+dy*c};
  }
  function stageToLocal(o,e,p){
    var b=objectBounds(o),cx=(b.x1+b.x2)/2,cy=(b.y1+b.y2)/2,t=transformFor(o,e),dx=p.x-(cx+t.x),dy=p.y-(cy+t.y),c=Math.cos(-t.rotation),s=Math.sin(-t.rotation);
    return{x:cx+(dx*c-dy*s)/(Math.abs(t.sx)<1e-6?1e-6:t.sx),y:cy+(dx*s+dy*c)/(Math.abs(t.sy)<1e-6?1e-6:t.sy)};
  }
  function warpedLocal(o,e,cage,p,time){return stageToLocal(o,e,warpPoint(cage,localToStage(o,e,p),time));}
  function activeCage(){return cageById(state.activeCageId);}
  function nearestEdgeIndex(points,p){
    var best=0,bestDistance=Infinity;
    for(var i=0;i<points.length;i++){
      var a=points[i],b=points[(i+1)%points.length],dx=b.x-a.x,dy=b.y-a.y,length2=dx*dx+dy*dy;
      var t=length2?clamp(((p.x-a.x)*dx+(p.y-a.y)*dy)/length2,0,1):0;
      var x=a.x+dx*t,y=a.y+dy*t,d=Math.hypot(p.x-x,p.y-y);
      if(d<bestDistance){bestDistance=d;best=i;}
    }
    return best;
  }
  function addPointToCage(cage,stagePoint){
    if(!cage||!Array.isArray(cage.points)||cage.points.length<3)return;
    var effective=API.getEffectiveObject?API.getEffectiveObject(cage,API.time):cage;
    var local=stageToLocal(cage,effective,stagePoint),edge=nearestEdgeIndex(cage.points,local),next=(edge+1)%cage.points.length;
    var rest=Array.isArray(cage.restPoints)&&cage.restPoints.length===cage.points.length?cage.restPoints.slice():restBoundary(cage.points);
    var a=rest[edge],b=rest[next],restPoint={x:(a.x+b.x)/2,y:(a.y+b.y)/2};
    API.pushUndo();cage.points.splice(edge+1,0,local);rest.splice(edge+1,0,restPoint);cage.restPoints=rest;cage.editorBounds=bounds(cage.points);state.addingPoint=false;refresh();
  }

    cage.points=nextPoints;cage.restPoints=nextRest;cage.editorBounds=bounds(nextPoints);refresh();
  }
  function smoothPoints(){

  }
  function distributedPoints(kind,count,b){
    count=Math.max(4,count||8);var cx=(b.x1+b.x2)/2,cy=(b.y1+b.y2)/2,w=Math.max(40,b.x2-b.x1),h=Math.max(40,b.y2-b.y1),out=[],i;
    if(kind==="limb"||kind==="bone"){
      var horizontal=w>=h,half=Math.max(2,Math.floor(count/2)),length=horizontal?w:h,thickness=Math.max(18,(horizontal?h:w)*(kind==="bone"?.55:.72));
      for(i=0;i<half;i++){var u=i/(half-1),along=-length/2+u*length,taper=kind==="bone"?(.45+.55*Math.sin(Math.PI*u)):1;out.push(horizontal?{x:cx+along,y:cy-thickness*taper/2}:{x:cx-thickness*taper/2,y:cy+along});}
      for(i=half-1;i>=0;i--){var v=i/(half-1),along2=-length/2+v*length,taper2=kind==="bone"?(.45+.55*Math.sin(Math.PI*v)):1;out.push(horizontal?{x:cx+along2,y:cy+thickness*taper2/2}:{x:cx+thickness*taper2/2,y:cy+along2});}
      return out;
    }
    var per=Math.max(1,Math.ceil(count/4));
    for(i=0;i<per;i++)out.push({x:b.x1+(b.x2-b.x1)*i/per,y:b.y1});
    for(i=0;i<per;i++)out.push({x:b.x2,y:b.y1+(b.y2-b.y1)*i/per});
    for(i=0;i<per;i++)out.push({x:b.x2-(b.x2-b.x1)*i/per,y:b.y2});
    for(i=0;i<per;i++)out.push({x:b.x1,y:b.y2-(b.y2-b.y1)*i/per});return out;
  }

  function evaluator(o,time,e){
    if(!o||!o[LINK]||o.type===TYPE||!Array.isArray(e.points))return e;
    var cage=cageById(o[LINK]);if(!cage)return e;
    var out=Object.assign({},e);
    out.points=e.points.map(function(p){return warpedLocal(o,e,cage,p,time);});
    if(Array.isArray(e.bezierHandles)&&e.bezierHandles.length===e.points.length){
      out.bezierHandles=e.bezierHandles.map(function(h,i){
        var p=e.points[i],wi=out.points[i];
        function handle(x,y){var w=warpedLocal(o,e,cage,{x:p.x+num(x,0),y:p.y+num(y,0)},time);return{x:w.x-wi.x,y:w.y-wi.y};}
        var a=handle(h.inX,h.inY),b=handle(h.outX,h.outY);return{inX:a.x,inY:a.y,outX:b.x,outY:b.y};
      });
    }
    out._latticeWarped=true;return out;
  }

  function triangle(ctx,img,s0,s1,s2,d0,d1,d2){
    var den=s0.x*(s1.y-s2.y)+s1.x*(s2.y-s0.y)+s2.x*(s0.y-s1.y);if(Math.abs(den)<1e-7)return;
    var a=(d0.x*(s1.y-s2.y)+d1.x*(s2.y-s0.y)+d2.x*(s0.y-s1.y))/den;
    var b=(d0.y*(s1.y-s2.y)+d1.y*(s2.y-s0.y)+d2.y*(s0.y-s1.y))/den;
    var c=(d0.x*(s2.x-s1.x)+d1.x*(s0.x-s2.x)+d2.x*(s1.x-s0.x))/den;
    var d=(d0.y*(s2.x-s1.x)+d1.y*(s0.x-s2.x)+d2.y*(s1.x-s0.x))/den;
    var tx=(d0.x*(s1.x*s2.y-s2.x*s1.y)+d1.x*(s2.x*s0.y-s0.x*s2.y)+d2.x*(s0.x*s1.y-s1.x*s0.y))/den;
    var ty=(d0.y*(s1.x*s2.y-s2.x*s1.y)+d1.y*(s2.x*s0.y-s0.x*s2.y)+d2.y*(s0.x*s1.y-s1.x*s0.y))/den;
    ctx.save();ctx.beginPath();ctx.moveTo(d0.x,d0.y);ctx.lineTo(d1.x,d1.y);ctx.lineTo(d2.x,d2.y);ctx.closePath();ctx.clip();ctx.transform(a,b,c,d,tx,ty);ctx.drawImage(img,0,0);ctx.restore();
  }
  function renderer(ctx,o,e){
    if(o.type===TYPE)return true;
    if(!o[LINK]||o.type!=="image")return false;
    var cage=cageById(o[LINK]),img=imageFor(String(e.src||o.src||""));if(!cage||!img||!img.complete||!img.naturalWidth)return false;
    var x=num(o.x,0),y=num(o.y,0),w=Math.max(1,num(o.width,img.naturalWidth)),h=Math.max(1,num(o.height,img.naturalHeight)),cols=12,rows=12;
    function vertex(ix,iy){var lp={x:x+w*ix/cols,y:y+h*iy/rows};return warpedLocal(o,e,cage,lp,API.time);}
    for(var iy=0;iy<rows;iy++)for(var ix=0;ix<cols;ix++){
      var u0=img.naturalWidth*ix/cols,u1=img.naturalWidth*(ix+1)/cols,v0=img.naturalHeight*iy/rows,v1=img.naturalHeight*(iy+1)/rows;
      var p00=vertex(ix,iy),p10=vertex(ix+1,iy),p01=vertex(ix,iy+1),p11=vertex(ix+1,iy+1);
      triangle(ctx,img,{x:u0,y:v0},{x:u1,y:v0},{x:u1,y:v1},p00,p10,p11);
      triangle(ctx,img,{x:u0,y:v0},{x:u1,y:v1},{x:u0,y:v1},p00,p11,p01);
    }
    return true;
  }
  function overlay(ctx){
    cages().forEach(function(cage){var pts=cageData(cage,API.time).target;if(pts.length<2)return;ctx.save();ctx.lineWidth=2;ctx.strokeStyle=String(cage.id)===state.activeCageId?"#a855f7":"rgba(117,71,168,.72)";ctx.fillStyle="rgba(168,85,247,.08)";ctx.setLineDash([7,5]);ctx.beginPath();ctx.moveTo(pts[0].x,pts[0].y);for(var i=1;i<pts.length;i++)ctx.lineTo(pts[i].x,pts[i].y);ctx.closePath();ctx.fill();ctx.stroke();ctx.setLineDash([]);pts.forEach(function(p){ctx.beginPath();ctx.arc(p.x,p.y,5,0,Math.PI*2);ctx.fillStyle="#fff";ctx.fill();ctx.strokeStyle="#7547a8";ctx.stroke();});ctx.restore();});
    if(state.drawing&&state.points.length){ctx.save();ctx.strokeStyle="#a855f7";ctx.lineWidth=2;ctx.setLineDash([6,4]);ctx.beginPath();ctx.moveTo(state.points[0].x,state.points[0].y);for(var i=1;i<state.points.length;i++)ctx.lineTo(state.points[i].x,state.points[i].y);ctx.stroke();state.points.forEach(function(p){ctx.beginPath();ctx.arc(p.x,p.y,4,0,Math.PI*2);ctx.fillStyle="#fff";ctx.fill();ctx.stroke();});ctx.restore();}
  }

  }
