/*
  Luke Animate Life Type Tool v1.2
  Displays a live value through selected text or a text object inside a Clip.
*/
(function(){
  "use strict";

  const PANEL_ID="lukeanimate-life-panel";
  const STYLE_ID="lukeanimate-life-style";
  const TOOL_ID="life";
  const TOOL_GROUP_ID=String(group&&group.id?group.id:"type");
  const TOOL_ITEM_ID=String(item&&(item.id||item.label)?(item.id||item.label):"Life");
  const TOOL_ICON_URL=String(item&&item.iconUrl?item.iconUrl:"https://lukeo25.github.io/LukeAnimate/SVGs/Life.svg");
  const DEFAULT_CODE=`const { config } = context;
const value = context.readValue(config.varName, config.defaultValue);
return {
  value,
  text: config.prefix + String(value) + config.suffix,
  fontFamily: config.fontFamily
};`;

  if(!LukeAnimate||Number(LukeAnimate.version||0)<121||typeof LukeAnimate.setTypeTool!=="function"){
    throw new Error("Life requires Luke Animate v121 or later.");
  }

  function setToolSelected(active){
    if(typeof LukeAnimate.setConfiguredToolActive==="function")LukeAnimate.setConfiguredToolActive(TOOL_GROUP_ID,TOOL_ITEM_ID,!!active);
  }

  const existing=document.getElementById(PANEL_ID);
  if(existing){
    existing.style.display="flex";existing.classList.add("open");setToolSelected(true);
    LukeAnimate.bringFloatingPanelToFront(existing);LukeAnimate.clampFloatingPanelToViewport(existing);return;
  }

  if(!document.getElementById(STYLE_ID)){
    const style=document.createElement("style");
    style.id=STYLE_ID;
    style.textContent=`
      #${PANEL_ID}{position:fixed;z-index:10000;left:28px;top:18px;width:374px;max-width:calc(100vw - 16px);height:min(650px,calc(100vh - 36px));display:flex;flex-direction:column;background:#f1f3f6;border:1px solid #bfc6cf;border-radius:14px;box-shadow:0 20px 60px rgba(0,0,0,.3);color:#27313d;font:13px Inter,-apple-system,BlinkMacSystemFont,Segoe UI,sans-serif;overflow:hidden}
      #${PANEL_ID} *{box-sizing:border-box}
      #${PANEL_ID} .lt-head{display:flex;align-items:center;gap:9px;padding:8px 10px;background:#fff;border-bottom:1px solid #d3d8df;cursor:move;user-select:none}
      #${PANEL_ID} .lt-head img{width:27px;height:27px;object-fit:contain;filter:grayscale(1) brightness(.72)}
      #${PANEL_ID} .lt-title{flex:1}.lt-title strong{display:block;font-size:14px;color:#29434a}.lt-title span{font-size:9px;letter-spacing:.11em;text-transform:uppercase;color:#7b8491}
      #${PANEL_ID} .lt-close{width:32px;height:32px;border:1px solid #bbc3cd;border-radius:8px;background:#fff;font-size:20px;color:#566270;cursor:pointer}
      #${PANEL_ID} .lt-body{flex:1;min-height:0;padding:8px 7px 46px;overflow:auto}
      #${PANEL_ID} .lt-selected{padding:8px 9px;margin-bottom:7px;border:1px solid #d2d7de;border-radius:10px;background:#fff;white-space:nowrap;overflow:hidden;text-overflow:ellipsis}
      #${PANEL_ID} .lt-selected strong{color:#0e8f7c}
      #${PANEL_ID} .lt-grid{display:grid;grid-template-columns:1fr 1fr;gap:7px;padding:8px;border:1px solid #d2d7de;border-radius:10px;background:#fff}
      #${PANEL_ID} label.lt-field{display:flex;flex-direction:column;gap:4px;font-size:11px;font-weight:700;color:#596674}
      #${PANEL_ID} label.lt-field.wide{grid-column:1/-1}
      #${PANEL_ID} input[type=text],#${PANEL_ID} select{width:100%;padding:7px;border:1px solid #b9c3ce;border-radius:6px;background:#fff;color:#263442;font:13px Arial,sans-serif}
      #${PANEL_ID} .lt-check{display:flex;align-items:center;gap:7px;margin:9px 0}
      #${PANEL_ID} .lt-check input{accent-color:#0e9c86}
      #${PANEL_ID} .lt-muted{color:#74808d;font-size:11px;line-height:1.35}
      #${PANEL_ID} details{margin-top:7px;border:1px solid #d2d7de;border-radius:10px;background:#fff;overflow:hidden}
      #${PANEL_ID} summary{padding:8px 9px;cursor:pointer;font-size:11px;letter-spacing:.08em;text-transform:uppercase;font-weight:700;color:#66717e;background:#fff}
      #${PANEL_ID} .lt-code-wrap{padding:9px}
      #${PANEL_ID} textarea{width:100%;min-height:165px;resize:vertical;padding:8px;border:1px solid #aeb8c4;border-radius:6px;background:#17212b;color:#e7f5f2;font:12px/1.45 Consolas,monospace;tab-size:2}
      #${PANEL_ID} .lt-code-tools{display:flex;justify-content:flex-end;margin-top:6px}
      #${PANEL_ID} .lt-actions{display:grid;grid-template-columns:1fr 1fr;gap:8px;margin-top:12px}
      #${PANEL_ID} .lt-btn{padding:7px;border:1px solid #b8c0ca;border-radius:8px;background:#eceff2;color:#4b5766;font-size:10px;cursor:pointer}
      #${PANEL_ID} .lt-btn:hover{border-color:#0e9c86} #${PANEL_ID} .lt-btn.primary{border-color:#0e9c86;background:#0e9c86;color:#fff;font-weight:700}
      #${PANEL_ID} .lt-btn:disabled{opacity:.45;cursor:not-allowed}
      #${PANEL_ID} .lt-status{position:absolute;left:7px;right:7px;bottom:7px;min-height:26px;max-height:42px;overflow:auto;border:1px solid #cbd2da;border-radius:8px;background:rgba(255,255,255,.97);padding:8px 10px;font-size:10px;line-height:1.35;color:#596573}
      #${PANEL_ID} .lt-status.good{border-color:#0e9c86;color:#117d6d} #${PANEL_ID} .lt-status.bad{border-color:#c66a6a;color:#a33d3d} #${PANEL_ID} .lt-status.warn{border-color:#d4a45c;color:#8a5a12}
      @media(max-width:560px){#${PANEL_ID}{left:8px;top:8px;width:calc(100vw - 16px);height:calc(100vh - 16px)}}
    `;
    document.head.appendChild(style);
  }

  const panel=document.createElement("section");
  panel.id=PANEL_ID;
  panel.className="open";
  panel.innerHTML=`
    <div class="lt-head" data-floating-panel-drag><img src="${TOOL_ICON_URL.replace(/"/g,"&quot;")}" alt=""><div class="lt-title"><strong>Life</strong><span>Text value display</span></div><button class="lt-close" type="button" title="Close">×</button></div>
    <div class="lt-body">
      <div class="lt-selected"><strong>Selected:</strong> <span data-lt-selected>No text or Clip selected</span></div>
      <p class="lt-muted">Select a text object directly, or select a Clip and choose one of its text objects. No object naming is required.</p>
      <div class="lt-grid">
        <label class="lt-field"><span>Variable name</span><input data-lt-variable type="text" value="LT_Life"></label>
        <label class="lt-field"><span>Default value</span><input data-lt-default type="text" value="0"></label>
        <label class="lt-field wide"><span>Text object</span><select data-lt-text-object><option value="">Select a text object or Clip</option></select></label>
        <label class="lt-field"><span>Prefix</span><input data-lt-prefix type="text" value=""></label>
        <label class="lt-field"><span>Suffix</span><input data-lt-suffix type="text" value=""></label>
        <label class="lt-field wide"><span>Font</span><input data-lt-font type="text" value="Poppins" list="lukeanimate-life-fonts"><datalist id="lukeanimate-life-fonts"><option>Montserrat</option><option>Poppins</option><option>Roboto</option><option>Open Sans</option><option>Inter</option><option>Orbitron</option><option>Bangers</option><option>Press Start 2P</option></datalist></label>
      </div>
      <label class="lt-check"><input data-lt-console type="checkbox"> Log changed Life values to the console</label>
      <details>
        <summary>View / edit code</summary>
        <div class="lt-code-wrap">
          <p class="lt-muted">Return an object containing <code>value</code>, <code>text</code> and/or <code>fontFamily</code>. The <code>context</code> also provides config, object, textObject, time, readValue and writeValue.</p>
          <textarea data-lt-code spellcheck="false"></textarea>
          <div class="lt-code-tools"><button class="lt-btn" data-lt-reset-code type="button">Reset code</button></div>
        </div>
      </details>
      <div class="lt-actions"><button class="lt-btn" data-lt-remove type="button">Remove</button><button class="lt-btn primary" data-lt-apply type="button">Apply</button></div>
      <div class="lt-status" data-lt-status></div>
    </div>`;
  document.body.appendChild(panel);
  LukeAnimate.registerFloatingPanel(panel);LukeAnimate.bringFloatingPanelToFront(panel);LukeAnimate.clampFloatingPanelToViewport(panel);setToolSelected(true);

  const get=selector=>panel.querySelector(selector);
  const status=get("[data-lt-status]");
  const applyButton=get("[data-lt-apply]");
  const removeButton=get("[data-lt-remove]");
  let selectedObject=null,selectedKey="";

  function setStatus(message,kind){status.textContent=String(message||"");status.className="lt-status"+(kind?" "+kind:"");}
  function selectedTarget(){
    const selected=LukeAnimate.getSelectedDrawObjects()||[];
    if(selected.length!==1||!selected[0])return null;
    if(selected[0].type==="text")return selected[0];
    if(selected[0].type!=="clipinstance")return null;
    const clips=LukeAnimate.getClips?LukeAnimate.getClips():{};
    return clips&&clips[selected[0].clipId]?selected[0]:null;
  }
  function objectName(object){
    if(!object)return "No text or Clip selected";
    if(object.type==="text")return "Text — "+(String(object.text||"").replace(/\s+/g," ").trim().slice(0,48)||"Untitled text");
    const clips=LukeAnimate.getClips?LukeAnimate.getClips():{};
    return String(object.name||(clips[object.clipId]&&clips[object.clipId].name)||"Clip");
  }
  function textObjectsForTarget(object){
    const clips=LukeAnimate.getClips?LukeAnimate.getClips():{},clip=object&&clips[object.clipId],found=[],seen=new Set(),seenIds=new Set();
    function visit(value){
      if(!value||typeof value!=="object"||seen.has(value))return;seen.add(value);
      if(value.type==="text"){
        const id=String(value.id||"");
        if(id&&!seenIds.has(id)){
          seenIds.add(id);
          const preview=String(value.text||"").replace(/\s+/g," ").trim().slice(0,48)||"Untitled text";
          found.push({id,name:String(value.name||""),fontFamily:String(value.fontFamily||""),object:value,label:"Text "+(found.length+1)+" — "+preview});
        }
      }
      if(Array.isArray(value.children))value.children.forEach(visit);
    }
    if(object&&object.type==="text")visit(object);
    else if(clip){
      (clip.drawObjects||[]).forEach(visit);
      (clip.sceneFrames||[]).forEach(frame=>(frame&&frame.objects||[]).forEach(visit));
      (clip.drawingLayers||[]).forEach(layer=>(layer&&layer.frames||[]).forEach(frame=>visit(frame&&frame.object)));
    }
    return found;
  }
  function fillTextObjects(object){
    const choices=textObjectsForTarget(object),select=get("[data-lt-text-object]");
    select.innerHTML=choices.length?choices.map(choice=>`<option value="${choice.id.replace(/"/g,"&quot;")}">${choice.label.replace(/&/g,"&amp;").replace(/</g,"&lt;")}</option>`).join(""):'<option value="">Select a text object or a Clip containing text</option>';
    select.disabled=!choices.length;
    return choices;
  }
  function loadSettings(settings,choices){
    const config=settings||{};
    get("[data-lt-variable]").value=String(config.varName||"LT_Life");
    get("[data-lt-default]").value=String(config.defaultValue===undefined?"0":config.defaultValue);
    const select=get("[data-lt-text-object]"),available=Array.isArray(choices)?choices:[];
    let selectedId=String(config.textObjectId||"");
    if(!available.some(choice=>choice.id===selectedId)&&config.textName){const legacy=available.find(choice=>choice.name===String(config.textName));selectedId=legacy?legacy.id:"";}
    if(!available.some(choice=>choice.id===selectedId))selectedId=available.length?available[0].id:"";
    select.value=selectedId;
    get("[data-lt-prefix]").value=String(config.prefix||"");
    get("[data-lt-suffix]").value=String(config.suffix||"");
    const selectedChoice=available.find(choice=>choice.id===selectedId);
    get("[data-lt-font]").value=String(config.fontFamily||(selectedChoice&&selectedChoice.fontFamily)||"Poppins");
    get("[data-lt-console]").checked=!!config.showConsole;
    get("[data-lt-code]").value=String(config.code||DEFAULT_CODE);
  }
  function refreshSelection(force){
    const next=selectedTarget(),nextKey=next?String(next.id||next.clipId):"";
    if(!force&&nextKey===selectedKey){
      if(next){const error=LukeAnimate.getTypeToolError(next,TOOL_ID);if(error)setStatus("Code error: "+error,"bad");}
      return;
    }
    selectedObject=next;selectedKey=nextKey;get("[data-lt-selected]").textContent=objectName(next);
    const choices=fillTextObjects(next);applyButton.disabled=!next||!choices.length;
    const settings=next?LukeAnimate.getTypeTool(next,TOOL_ID):null;removeButton.disabled=!settings;loadSettings(settings,choices);
    setStatus(next?(settings?"Life is attached to this "+(next.type==="text"?"text object.":"Clip."):"Ready to attach Life."):"Select one text object or Clip on the stage.",next?"good":"");
  }
  function currentSettings(){return{enabled:true,varName:get("[data-lt-variable]").value,textObjectId:get("[data-lt-text-object]").value,textName:"",defaultValue:get("[data-lt-default]").value,prefix:get("[data-lt-prefix]").value,suffix:get("[data-lt-suffix]").value,fontFamily:get("[data-lt-font]").value,showConsole:get("[data-lt-console]").checked,code:get("[data-lt-code]").value};}

  get("[data-lt-reset-code]").addEventListener("click",()=>{get("[data-lt-code]").value=DEFAULT_CODE;setStatus("Life code reset. Click Apply to save it.","good");});
  get("[data-lt-text-object]").addEventListener("change",()=>{
    const choice=textObjectsForTarget(selectedObject).find(item=>item.id===get("[data-lt-text-object]").value);
    if(choice&&choice.fontFamily)get("[data-lt-font]").value=choice.fontFamily;
  });
  applyButton.addEventListener("click",()=>{
    refreshSelection(false);if(!selectedObject){setStatus("Select one text object or Clip first.","bad");return;}
    const settings=currentSettings(),choices=textObjectsForTarget(selectedObject),found=choices.some(choice=>choice.id===settings.textObjectId);
    LukeAnimate.setTypeTool(selectedObject,TOOL_ID,settings);removeButton.disabled=false;
    setStatus(found?"Life applied. Press Play or scrub the timeline to see it.":"Choose a text object inside this Clip first.",found?"good":"warn");
  });
  removeButton.addEventListener("click",()=>{
    refreshSelection(false);if(!selectedObject||!LukeAnimate.removeTypeTool(selectedObject,TOOL_ID)){setStatus("The selected object does not have Life attached.","bad");return;}
    removeButton.disabled=true;loadSettings(null,textObjectsForTarget(selectedObject));setStatus("Life removed.","good");
  });
  const pollTimer=setInterval(()=>refreshSelection(false),250);
  get(".lt-close").addEventListener("click",()=>{clearInterval(pollTimer);panel.remove();setToolSelected(false);});
  refreshSelection(true);
})();
