      panel.style.left = (event.clientX - drag.dx) + "px";
      panel.style.top = (event.clientY - drag.dy) + "px";
      panel.style.right = "auto";

    });

      if(!drag || drag.pointerId !== event.pointerId) return;
      drag = null;
      LukeAnimate.clampFloatingPanelToViewport(panel);
    }
    header.addEventListener("pointerup", stop);
    header.addEventListener("pointercancel", stop);

  function buildHierarchy(root){
    const clips = LukeAnimate.getClips();
    const nodes = [];
    const nodeMap = new Map();
    const visited = new Set();
    const rootClip = clips[root.clipId];

    const rootNode = {
      id:objectKey(root, ["root"]),
      obj:root,
      clip:rootClip || null,
      depth:0,
      path:[],
      isRoot:true,
      name:clipName(root, rootClip, "Selected Clip")
    };
    nodes.push(rootNode);
    nodeMap.set(rootNode.id, rootNode);
    visited.add(rootNode.id);

    function walkClipDefinition(clip, depth, path){
      if(!clip || depth > 12) return;
      const objects = Array.isArray(clip.drawObjects) ? clip.drawObjects : [];
      objects.forEach(function(obj, index){
        if(!obj || obj.type !== "clipinstance") return;
        const childClip = clips[obj.clipId];
        const childPath = path.concat(index);
        const id = objectKey(obj, childPath);
        if(visited.has(id)) return;
        visited.add(id);
        const node = {
          id:id,
          obj:obj,
          clip:childClip || null,
          depth:depth,
          path:childPath,
          name:clipName(obj, childClip, "Clip Part " + (nodes.length + 1))
        };
        nodes.push(node);
        nodeMap.set(id, node);
        if(childClip) walkClipDefinition(childClip, depth + 1, childPath);
      });
    }

    walkClipDefinition(rootClip, 1, []);
    return { nodes:nodes, nodeMap:nodeMap, rootClip:rootClip };
  }

  function ensureData(obj){

        if(kf.rotation !== undefined){
          data.channels.rotate.keys[String(frame)] = { rotation:round3(num(kf.rotation, data.base.rotation)) };
          data.channels.rotate.easing[String(frame)] = "linear";
        }
        if(kf.opacity !== undefined){
          data.channels.opacity.keys[String(frame)] = { opacity:clamp(num(kf.opacity, data.base.opacity), 0, 1) };
          data.channels.opacity.easing[String(frame)] = "linear";
        }
      });
      obj[DATA_KEY] = data;
    }
    if(!data.base) data.base = readBase(obj);
    if(!data.channels) data.channels = {};
    ["move","scale","rotate","opacity"].forEach(function(channel){
      if(!data.channels[channel]) data.channels[channel] = emptyChannel();
      if(!data.channels[channel].keys) data.channels[channel].keys = {};
      if(!data.channels[channel].easing) data.channels[channel].easing = {};
    });
    data.rotationMin = num(data.rotationMin, -359);
    data.rotationMax = num(data.rotationMax, 360);
    return data;
  }

  function sortedFrames(channel){
    return Object.keys(channel.keys || {}).map(Number).filter(Number.isFinite).sort(function(a,b){ return a-b; });
  }

  function easingValue(t, easing){
    t = clamp(t, 0, 1);
    if(easing === "ease-in") return t * t;
    if(easing === "ease-out") return 1 - Math.pow(1 - t, 2);
    if(easing === "ease-in-out") return t < .5 ? 2*t*t : 1 - Math.pow(-2*t + 2, 2)/2;
    if(easing === "hold") return 0;
    return t;
  }

  function interpolateChannel(data, channelName, frame){
    const channel = data.channels[channelName];
    const frames = sortedFrames(channel);
    const base = data.base;
    const properties = channelName === "move" ? ["x","y"] : channelName === "scale" ? ["scale"] : channelName === "rotate" ? ["rotation"] : ["opacity"];
    const baseValue = {};
    properties.forEach(function(property){ baseValue[property] = base[property]; });
    if(!frames.length) return baseValue;

    if(frame <= frames[0]){
      if(frames[0] <= 1) return Object.assign({}, channel.keys[String(frames[0])]);
      const next = channel.keys[String(frames[0])];
      const ratio = easingValue((frame - 1) / Math.max(1, frames[0] - 1), "linear");
      const out = {};
      properties.forEach(function(property){ out[property] = baseValue[property] + (num(next[property], baseValue[property]) - baseValue[property]) * ratio; });
      return out;
    }
    if(frame >= frames[frames.length - 1]) return Object.assign({}, channel.keys[String(frames[frames.length - 1])]);

    let previousFrame = frames[0];
    let nextFrame = frames[frames.length - 1];
    for(let i=0;i<frames.length-1;i++){
      if(frame >= frames[i] && frame <= frames[i+1]){
        previousFrame = frames[i];
        nextFrame = frames[i+1];
        break;
      }
    }
    const previous = channel.keys[String(previousFrame)];
    const next = channel.keys[String(nextFrame)];
    const easing = channel.easing[String(previousFrame)] || "linear";
    const ratio = easingValue((frame - previousFrame) / Math.max(1, nextFrame - previousFrame), easing);
    const out = {};
    properties.forEach(function(property){
      const from = num(previous[property], baseValue[property]);
      let to = num(next[property], from);
      if(channelName === "rotate"){
        let delta = to - from;
        while(delta > 180) delta -= 360;
        while(delta < -180) delta += 360;
        to = from + delta;
      }
      out[property] = from + (to - from) * ratio;
    });
    return out;
  }

  function evaluateData(data, frame){
    const move = interpolateChannel(data, "move", frame);
    const scale = interpolateChannel(data, "scale", frame);
    const rotate = interpolateChannel(data, "rotate", frame);
    const opacity = interpolateChannel(data, "opacity", frame);
    return {
      x:round3(num(move.x, data.base.x)),
      y:round3(num(move.y, data.base.y)),
      scale:round3(Math.max(.01, num(scale.scale, data.base.scale))),
      rotation:round3(num(rotate.rotation, data.base.rotation)),
      opacity:clamp(num(opacity.opacity, data.base.opacity), 0, 1)
    };
  }

  function channelFramesAt(data, frame){
    const found = [];
    ["move","scale","rotate","opacity"].forEach(function(channel){
      if(data.channels[channel].keys[String(frame)]) found.push(channel);
    });
    return found;
  }

  function allUserFrames(data){
    const set = new Set();
    ["move","scale","rotate","opacity"].forEach(function(channel){
      sortedFrames(data.channels[channel]).forEach(function(frame){ set.add(frame); });
    });
    return Array.from(set).sort(function(a,b){ return a-b; });
  }

  function compileKeys(obj, data){
    const external = Array.isArray(obj.keyframes) ? obj.keyframes.filter(function(kf){ return !kf[KEY_MARKER]; }) : [];
    const compiled = allUserFrames(data).map(function(frame){
      const pose = evaluateData(data, frame);
      return {
        time:frameToTime(frame),
        x:pose.x,
        y:pose.y,
        scale:pose.scale,
        rotation:pose.rotation,
        opacity:pose.opacity,
        [KEY_MARKER]:true,
        _stickRigSourceFrame:frame,
        _stickRigChannels:channelFramesAt(data, frame)
      };
    });
    obj.keyframes = external.concat(compiled).sort(function(a,b){ return num(a.time,0)-num(b.time,0); });
  }

  function evaluator(obj, time, effective){
    const data = obj && obj[DATA_KEY];
    if(!data) return effective;
    const frame = Math.max(1, Math.round(num(time, 0) * Math.max(1, num(LukeAnimate.fps, 30))) + 1);
    const pose = evaluateData(data, frame);
    return Object.assign({}, effective || obj, pose);
  }

  LukeAnimate.registerObjectEvaluator(PLUGIN_ID, evaluator);

  function pivotDelta(dx, dy, scale, rotation){
    const s = num(scale, 1);
    const rad = num(rotation, 0) * Math.PI / 180;
    const cos = Math.cos(rad), sin = Math.sin(rad);
    return { x:(dx * cos - dy * sin) * s, y:(dx * sin + dy * cos) * s };
  }

  function compensateStickRigPivot(instance, change){
    const data = instance && instance[DATA_KEY];
    if(!data || !change) return;
    ensureData(instance);
    const frames = allUserFrames(data);
    const snapshots = frames.map(function(frame){ return { frame:frame, pose:evaluateData(data, frame) }; });
    const baseDelta = pivotDelta(change.dx, change.dy, data.base.scale, data.base.rotation);
    data.base.x = round3(num(data.base.x, 0) + baseDelta.x);
    data.base.y = round3(num(data.base.y, 0) + baseDelta.y);
    snapshots.forEach(function(entry){
      const delta = pivotDelta(change.dx, change.dy, entry.pose.scale, entry.pose.rotation);
      data.channels.move.keys[String(entry.frame)] = {
        x:round3(entry.pose.x + delta.x),
        y:round3(entry.pose.y + delta.y)
      };
      if(!data.channels.move.easing[String(entry.frame)]) data.channels.move.easing[String(entry.frame)] = "linear";
    });
    compileKeys(instance, data);
  }

  if(typeof LukeAnimate.registerClipPivotCompensator === "function"){
    LukeAnimate.registerClipPivotCompensator(PLUGIN_ID, compensateStickRigPivot);
  }

  function syncExternalTimelineChanges(node){
    const obj = node.obj;
    const data = obj && obj[DATA_KEY];
    if(!data) return false;
    const keys = Array.isArray(obj.keyframes) ? obj.keyframes.filter(function(kf){ return kf[KEY_MARKER]; }) : [];
    let changed = false;

    keys.slice().forEach(function(kf){
      const oldFrame = Math.max(1, Math.round(num(kf._stickRigSourceFrame, 1)));
      const newFrame = Math.max(1, Math.round(num(kf.time, 0) * Math.max(1, num(LukeAnimate.fps, 30))) + 1);
      if(oldFrame === newFrame) return;
      const channels = Array.isArray(kf._stickRigChannels) ? kf._stickRigChannels : [];
      channels.forEach(function(channelName){
        const channel = data.channels[channelName];
        const source = channel.keys[String(oldFrame)];
        if(!source) return;
        channel.keys[String(newFrame)] = source;
        channel.easing[String(newFrame)] = channel.easing[String(oldFrame)] || "linear";
        delete channel.keys[String(oldFrame)];
        delete channel.easing[String(oldFrame)];
      });
      kf._stickRigSourceFrame = newFrame;
      changed = true;
    });

    const actualSourceFrames = new Set(keys.map(function(kf){ return Math.max(1, Math.round(num(kf._stickRigSourceFrame, 1))); }));
    allUserFrames(data).forEach(function(frame){
      if(actualSourceFrames.has(frame)) return;
      const channels = channelFramesAt(data, frame);
      if(!channels.length) return;
      channels.forEach(function(channelName){
        delete data.channels[channelName].keys[String(frame)];
        delete data.channels[channelName].easing[String(frame)];
      });
      changed = true;
    });

    if(changed) compileKeys(obj, data);
    return changed;
  }

  function nodeHasAnyKeys(node){
    const data = ensureData(node.obj);
    return allUserFrames(data).length > 0;
  }

  function currentKeyCount(node){
    const data = ensureData(node.obj);
    return channelFramesAt(data, currentFrame()).length;
  }

  function setChannelKey(node, channelName, values, useUndo, rerender){
    if(!node) return;
    if(useUndo && typeof LukeAnimate.pushUndo === "function") LukeAnimate.pushUndo();
    const data = ensureData(node.obj);
    const frame = currentFrame();
    const key = String(frame);
    const current = evaluateData(data, frame);
    if(channelName === "move") data.channels.move.keys[key] = { x:round3(num(values.x, current.x)), y:round3(num(values.y, current.y)) };
    else if(channelName === "scale") data.channels.scale.keys[key] = { scale:round3(Math.max(.01, num(values.scale, current.scale))) };
    else if(channelName === "rotate") data.channels.rotate.keys[key] = { rotation:round3(num(values.rotation, current.rotation)) };
    else data.channels.opacity.keys[key] = { opacity:clamp(num(values.opacity, current.opacity * 100) / 100, 0, 1) };
    if(!data.channels[channelName].easing[key]) data.channels[channelName].easing[key] = "linear";
    compileKeys(node.obj, data);
    LukeAnimate.refreshUI();
    if(rerender !== false) renderCards();
  }

  function deleteChannelKey(node, channelName){
    if(!node) return;
    const data = ensureData(node.obj);
    const frame = currentFrame();
    if(!data.channels[channelName].keys[String(frame)]){
      setStatus("No " + channelName + " key exists at frame " + frame + ".", "warn");
      return;
    }
    LukeAnimate.pushUndo();
    delete data.channels[channelName].keys[String(frame)];
    delete data.channels[channelName].easing[String(frame)];
    compileKeys(node.obj, data);
    LukeAnimate.refreshUI();
    renderCards();
    setStatus("Deleted " + channelName + " key at frame " + frame + ".", "good");
  }

  function clearNodeKeys(node, useUndo){
    if(!node) return;
    if(useUndo) LukeAnimate.pushUndo();
    const data = ensureData(node.obj);
    data.channels.move = emptyChannel();
    data.channels.scale = emptyChannel();
    data.channels.rotate = emptyChannel();
    data.channels.opacity = emptyChannel();
    compileKeys(node.obj, data);
  }

  function inputControl(label, kind, value, min, max, step, field){
    return `
      <div class="sr-control">
        <label>${escapeHtml(label)}</label>
        <input type="range" data-role="range" data-field="${field}" data-kind="${kind}" min="${min}" max="${max}" step="${step}" value="${value}">
        <input type="number" data-role="number" data-field="${field}" data-kind="${kind}" min="${min}" max="${max}" step="${step}" value="${value}">
      </div>`;
  }

  function channelHtml(node, channelName, pose, data){
    const frame = currentFrame();
    const keyed = !!data.channels[channelName].keys[String(frame)];
    const open = state.openChannel === channelName;
    let controls = "";
    if(channelName === "move"){
      const rangeX = Math.max(1000, num(LukeAnimate.width, 960) * 2);
      const rangeY = Math.max(1000, num(LukeAnimate.height, 540) * 2);
      controls += inputControl("X", channelName, pose.x, -rangeX, rangeX, .1, "x");
      controls += inputControl("Y", channelName, pose.y, -rangeY, rangeY, .1, "y");
    }else if(channelName === "scale"){
      controls += inputControl("Scale", channelName, pose.scale, .01, 10, .01, "scale");
    }else if(channelName === "rotate"){
      controls += inputControl("Rotate", channelName, pose.rotation, data.rotationMin, data.rotationMax, .1, "rotation");
      controls += `
        <div class="sr-control"><label>Min</label><span></span><input type="number" data-role="rotation-min" value="${data.rotationMin}" step="1"></div>
        <div class="sr-control"><label>Max</label><span></span><input type="number" data-role="rotation-max" value="${data.rotationMax}" step="1"></div>
        <div class="sr-quick">
          ${[-180,-90,0,90,180].map(function(v){ return `<button type="button" data-role="quick-rotate" data-value="${v}">${v}°</button>`; }).join("")}
        </div>`;
    }else{
      controls += inputControl("Opacity", channelName, Math.round(pose.opacity * 100), 0, 100, 1, "opacity");
    }
    const easing = data.channels[channelName].easing[String(frame)] || "linear";
    controls += `
      <div class="sr-control"><label>Ease</label><span></span>
        <select data-role="easing" data-channel="${channelName}">
          <option value="linear"${easing==="linear"?" selected":""}>Linear</option>
          <option value="ease-in"${easing==="ease-in"?" selected":""}>Ease In</option>
          <option value="ease-out"${easing==="ease-out"?" selected":""}>Ease Out</option>
          <option value="ease-in-out"${easing==="ease-in-out"?" selected":""}>Ease In/Out</option>
          <option value="hold"${easing==="hold"?" selected":""}>Hold</option>
        </select>
      </div>
      <div class="sr-channel-footer">
        <button type="button" class="primary" data-role="set-key" data-channel="${channelName}">${keyed ? "Update Key" : "Set Key"}</button>
        <button type="button" class="danger" data-role="delete-key" data-channel="${channelName}"${keyed?"":" disabled"}>Delete Key</button>
      </div>`;
    return `
      <div class="sr-channel ${channelName}${open?" open":""}" data-channel="${channelName}">
        <div class="sr-channel-head" data-role="channel-head" data-channel="${channelName}">
          <button type="button" tabindex="-1">${open?"−":"+"}</button>
          <div class="sr-channel-title">${channelName}</div>
          <span class="sr-key-badge${keyed?" keyed":""}">${keyed?"Keyed":"No Key"}</span>
          <button type="button" data-role="set-key" data-channel="${channelName}">Set</button>
        </div>
        <div class="sr-channel-body">${controls}</div>
      </div>`;
  }

  function renderCards(){
    const cards = panel.querySelector("#sr-cards");
    const empty = panel.querySelector("#sr-empty");
    const clearButton = panel.querySelector("#sr-clear");
    const exportButton = panel.querySelector("#sr-export");
    cards.innerHTML = "";
    const hasRoot = !!state.root;
    clearButton.disabled = !hasRoot || !state.nodes.some(nodeHasAnyKeys);
    exportButton.disabled = !hasRoot;

    if(!hasRoot){
      empty.style.display = "block";
      empty.innerHTML = "Select a Luke Animate <b>Clip</b> instance, then click <b>Load</b>.<br>Its transform controls and any nested Clip parts will appear here.";
      return;
    }
    if(!state.nodes.length){
      empty.style.display = "block";
      empty.innerHTML = "No animatable Clip was found in the current selection.";
      return;
    }
    empty.style.display = "none";

    const frame = currentFrame();
    state.nodes.forEach(function(node){
      const data = ensureData(node.obj);
      const pose = evaluateData(data, frame);
      const isActive = state.activeId === node.id;
      const hasAny = nodeHasAnyKeys(node);
      const keyCount = currentKeyCount(node);
      const card = document.createElement("div");
      card.className = "sr-card" + (isActive?" active open":"") + (hasAny?" has-key":"");
      card.dataset.nodeId = node.id;
      card.innerHTML = `
        <div class="sr-card-head">
          <span class="sr-depth" style="--sr-depth:${node.depth}"></span>
          <button type="button" class="sr-expand" data-role="open-node">${isActive?"−":"+"}</button>
          <div class="sr-name" title="${escapeHtml(node.name)}">${escapeHtml(node.name)}</div>
          <span class="sr-key-badge${keyCount?" keyed":""}">${keyCount ? keyCount + " key" + (keyCount===1?"":"s") : "No key"}</span>
          <button type="button" class="sr-animate${isActive?" active":""}" data-role="open-node">Animate</button>
        </div>
        <div class="sr-card-body">
          ${channelHtml(node,"move",pose,data)}
          ${channelHtml(node,"scale",pose,data)}
          ${channelHtml(node,"rotate",pose,data)}
          ${channelHtml(node,"opacity",pose,data)}
        </div>`;
      cards.appendChild(card);
    });
  }

  function getNodeFromElement(element){
    const card = element.closest(".sr-card");
    return card ? state.nodeMap.get(card.dataset.nodeId) : null;
  }

  function channelValuesFromCard(card, channelName){
    const values = {};
    card.querySelectorAll('[data-kind="' + channelName + '"][data-role="number"]').forEach(function(input){
      values[input.dataset.field] = num(input.value, 0);
    });
    return values;
  }

  function updatePairedInput(input){
    const card = input.closest(".sr-card");
    if(!card) return;
    const role = input.dataset.role;
    const otherRole = role === "range" ? "number" : "range";
    const other = card.querySelector('[data-kind="' + input.dataset.kind + '"][data-field="' + input.dataset.field + '"][data-role="' + otherRole + '"]');
    if(other) other.value = input.value;
  }

  panel.querySelector("#sr-cards").addEventListener("click", function(event){
    const roleNode = event.target.closest("[data-role]");
    if(!roleNode) return;
    const node = getNodeFromElement(roleNode);
    const role = roleNode.dataset.role;
    if(role === "open-node"){
      state.activeId = state.activeId === node.id ? "" : node.id;
      renderCards();
      return;
    }
    if(role === "channel-head"){
      state.openChannel = roleNode.dataset.channel;
      renderCards();
      return;
    }
    if(role === "set-key"){
      const channel = roleNode.dataset.channel;
      const values = channelValuesFromCard(roleNode.closest(".sr-card"), channel);
      setChannelKey(node, channel, values, true);
      setStatus("Set " + channel + " key for " + node.name + " at frame " + currentFrame() + ".", "good");
      return;
    }
    if(role === "delete-key"){
      deleteChannelKey(node, roleNode.dataset.channel);
      return;
    }
    if(role === "quick-rotate"){
      const value = num(roleNode.dataset.value, 0);
      setChannelKey(node, "rotate", {rotation:value}, true);
      setStatus("Set rotation to " + value + "° at frame " + currentFrame() + ".", "good");
      return;
    }
  });

  panel.querySelector("#sr-cards").addEventListener("input", function(event){
    const input = event.target;
    const role = input.dataset.role;
    const node = getNodeFromElement(input);
    if(!node) return;
    if(role === "range" || role === "number"){
      updatePairedInput(input);
      const channel = input.dataset.kind;
      const values = channelValuesFromCard(input.closest(".sr-card"), channel);
      if(!state.editUndoArmed){
        LukeAnimate.pushUndo();
        state.editUndoArmed = true;
      }
      setChannelKey(node, channel, values, false, false);
      return;
    }
    if(role === "rotation-min" || role === "rotation-max"){
      const data = ensureData(node.obj);
      if(role === "rotation-min") data.rotationMin = num(input.value, -359);
      else data.rotationMax = num(input.value, 360);
      if(data.rotationMin > data.rotationMax){
        const swap = data.rotationMin;
        data.rotationMin = data.rotationMax;
        data.rotationMax = swap;
      }
      renderCards();
    }
  });

  panel.querySelector("#sr-cards").addEventListener("change", function(event){
    const input = event.target;
    state.editUndoArmed = false;
    const node = getNodeFromElement(input);
    if(!node) return;
    if(input.dataset.role === "easing"){
      const data = ensureData(node.obj);
      const frame = currentFrame();
      const channel = input.dataset.channel;
      if(!data.channels[channel].keys[String(frame)]){
        const pose = evaluateData(data, frame);
        const values = channel === "move" ? {x:pose.x,y:pose.y} : channel === "scale" ? {scale:pose.scale} : channel === "rotate" ? {rotation:pose.rotation} : {opacity:pose.opacity * 100};
        setChannelKey(node, channel, values, true);
      }
      data.channels[channel].easing[String(frame)] = input.value;
      compileKeys(node.obj, data);
      LukeAnimate.refreshUI();
      renderCards();
    }
  });

  function loadSelection(){
    const root = selectedRoot();
    if(!root){
      state.root = null;
      state.rootClip = null;
      state.nodes = [];
      state.nodeMap = new Map();
      state.activeId = "";
      renderCards();
      setStatus("Select exactly one Clip instance in Luke Animate, then click Load.", "bad");
      return;
    }
    const result = buildHierarchy(root);
    if(!result.rootClip){
      setStatus("The selected Clip instance refers to a missing Clip definition.", "bad");
      return;
    }
    state.root = root;
    state.rootClip = result.rootClip;
    state.nodes = result.nodes;
    state.nodeMap = result.nodeMap;
    state.activeId = result.nodes.length ? result.nodes[0].id : "";
    state.nodes.forEach(function(node){
      const data = ensureData(node.obj);
      compileKeys(node.obj, data);
    });
    LukeAnimate.refreshUI();
    renderCards();
    const nestedCount = Math.max(0, result.nodes.length - 1);
    setStatus("Loaded the selected Clip" + (nestedCount ? " and " + nestedCount + " nested Clip part" + (nestedCount===1?"":"s") : "") + ". Its controls are now in this floating panel.", "good");
  }

  function refreshHierarchy(){
    if(!state.root){ loadSelection(); return; }
    const result = buildHierarchy(state.root);
    state.rootClip = result.rootClip;
    state.nodes = result.nodes;
    state.nodeMap = result.nodeMap;
    if(!state.nodeMap.has(state.activeId)) state.activeId = result.nodes.length ? result.nodes[0].id : "";
    renderCards();
    setStatus("Refreshed StickRig hierarchy.", "good");
  }

  function exportRig(){
    if(!state.root) return;
    const payload = {
      format:"LukeAnimateStickRig",
      version:1,
      exportedAt:new Date().toISOString(),
      root:{ clipId:state.root.clipId, name:clipName(state.root,state.rootClip,"Clip") },
      nodes:state.nodes.map(function(node){
        return { id:node.id, path:node.path, name:node.name, clipId:node.obj.clipId, data:JSON.parse(JSON.stringify(ensureData(node.obj))) };
      })
    };
    const blob = new Blob([JSON.stringify(payload,null,2)], {type:"application/json"});
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = (payload.root.name || "StickRig").replace(/[^a-z0-9_-]+/gi,"_") + "_StickRig.json";
    document.body.appendChild(a);
    a.click();
    a.remove();
    setTimeout(function(){ URL.revokeObjectURL(url); }, 1000);
    setStatus("Exported StickRig JSON.", "good");
  }

  function importRigFile(file){
    if(!state.root){
      setStatus("Load a Clip before importing a StickRig JSON file.", "bad");
      return;
    }
    const reader = new FileReader();
    reader.onload = function(){
      try{
        const payload = JSON.parse(String(reader.result || ""));
        if(!payload || payload.format !== "LukeAnimateStickRig" || !Array.isArray(payload.nodes)) throw new Error("Not a Luke Animate StickRig JSON file.");
        LukeAnimate.pushUndo();
        let applied = 0;
        payload.nodes.forEach(function(saved){
          let node = state.nodeMap.get(saved.id);
          if(!node && Array.isArray(saved.path)){
            node = state.nodes.find(function(candidate){ return JSON.stringify(candidate.path) === JSON.stringify(saved.path); });
          }
          if(!node) node = state.nodes.find(function(candidate){ return candidate.name === saved.name && candidate.obj.clipId === saved.clipId; });
          if(!node || !saved.data) return;
          node.obj[DATA_KEY] = JSON.parse(JSON.stringify(saved.data));
          compileKeys(node.obj, ensureData(node.obj));
          applied += 1;
        });
        LukeAnimate.refreshUI();
        renderCards();
        setStatus("Imported StickRig data for " + applied + " part" + (applied===1?"":"s") + ".", "good");
      }catch(err){
        setStatus("Import failed: " + (err && err.message ? err.message : err), "bad");
      }
    };
    reader.readAsText(file);
  }

  panel.querySelector("#sr-load").addEventListener("click", loadSelection);
  panel.querySelector("#sr-refresh").addEventListener("click", refreshHierarchy);
  panel.querySelector("#sr-export").addEventListener("click", exportRig);
  panel.querySelector("#sr-import").addEventListener("click", function(){ importInput.value = ""; importInput.click(); });
  importInput.addEventListener("change", function(){ if(importInput.files && importInput.files[0]) importRigFile(importInput.files[0]); });
  panel.querySelector("#sr-clear").addEventListener("click", function(){
    if(!state.root) return;
    LukeAnimate.pushUndo();
    state.nodes.forEach(function(node){ clearNodeKeys(node, false); });
    LukeAnimate.refreshUI();
    renderCards();
    setStatus("Cleared all StickRig keys from the loaded Clip.", "good");
  });
  panel.querySelector("#sr-close").addEventListener("click", function(){
    panel.style.display = "none";
    setToolSelected(false);
    suppressFixedInspector(false);
  });

  function selectionKey(){
    return LukeAnimate.getSelectedDrawObjects().map(function(obj){ return obj && (obj.id || obj._id || obj.clipId || obj.type); }).join("|");
  }

  function poll(){
    const frame = currentFrame();
    const frameValue = panel.querySelector("#sr-frame-value");
    if(frameValue) frameValue.textContent = String(frame);

    let changed = false;
    if(state.root){
      state.nodes.forEach(function(node){ if(syncExternalTimelineChanges(node)) changed = true; });
    }
    if(changed){
      LukeAnimate.refreshUI();
      renderCards();
      setStatus("Synchronized StickRig with timeline key changes.", "good");
    }else if(frame !== state.lastFrame && state.root){
      if(!panel.contains(document.activeElement)) renderCards();
    }
    state.lastFrame = frame;
    state.lastSelectionKey = selectionKey();
  }

  state.pollTimer = window.setInterval(poll, 160);
  window.addEventListener("resize", function(){ LukeAnimate.clampFloatingPanelToViewport(panel); });

  renderCards();
  poll();
  if(selectedRoot()) loadSelection();
  else setStatus("StickRig " + VERSION + " loaded. Select a Clip instance and click Load.", "good");
})();
