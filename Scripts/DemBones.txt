  function objectBounds(o){

    }
    if(o.type==="line"||o.type==="rect"||o.type==="ellipse") return {x1:Math.min(n(o.x1,0),n(o.x2,0)),y1:Math.min(n(o.y1,0),n(o.y2,0)),x2:Math.max(n(o.x1,0),n(o.x2,0)),y2:Math.max(n(o.y1,0),n(o.y2,0))};
    if(o.type==="image"){
      var ix=n(o.x,0),iy=n(o.y,0),iw=Math.max(1,n(o.width,1)),ih=Math.max(1,n(o.height,1));
      return {x1:ix,y1:iy,x2:ix+iw,y2:iy+ih};

      return {x1:n(o.x,0),y1:n(o.y,0),x2:n(o.x,0)+w,y2:n(o.y,0)+h};
    }
    return {x1:0,y1:0,x2:0,y2:0};
  }

  function shiftObject(o,dx,dy){
    /*
      IMPORTANT CHARACTER COORDINATE CONTRACT:
      shiftObject() moves the visible object geometry AND DemBones restPoints
      together. Callers must therefore NOT shift demBones.restPoints again.
    */
    if(o.type==="freehand" && Array.isArray(o.points)) o.points.forEach(function(p){p.x=n(p.x,0)+dx;p.y=n(p.y,0)+dy;});
    else if(o.type==="line"||o.type==="rect"||o.type==="ellipse"){o.x1=n(o.x1,0)+dx;o.y1=n(o.y1,0)+dy;o.x2=n(o.x2,0)+dx;o.y2=n(o.y2,0)+dy;}
    else if(o.type==="text"||o.type==="clipinstance"||o.type==="image"){o.x=n(o.x,0)+dx;o.y=n(o.y,0)+dy;}
    if(o.demBones && Array.isArray(o.demBones.restPoints)) o.demBones.restPoints.forEach(function(p){p.x=n(p.x,0)+dx;p.y=n(p.y,0)+dy;});
    return o;
  }

  function centreImportedSvgObjectsOnStage(objects){
    var list=(objects||[]).filter(Boolean);

    if(!list.length){
      return {dx:0,dy:0};
    }

    /*
      Use the bounds of the actual imported artwork rather than the SVG's
      width/height/viewBox. This is what prevents empty SVG document margins
      from making an apparently centred import look offset.
    */
    var bounds=combinedBounds(list);
    var centre=currentStageCentreScene();

    var dx=centre.x-n(bounds.cx,0);
    var dy=centre.y-n(bounds.cy,0);

    list.forEach(function(object){
      shiftObject(
        object,
        dx,
        dy
      );
    });

    return{
      dx:dx,
      dy:dy,
      x:centre.x,
      y:centre.y
    };
  }

  function sampleSvgPath(el,count){
    var points=[];
    try{
      var len=el.getTotalLength();
      /*
        v8: SVGs loaded directly into DemBones get a deformation mesh fine
        enough that envelope transitions do not show obvious straight kinks.
        Four scene units per sample is still bounded to avoid runaway SVGs.
      */
      var steps=Math.max(16,Math.min(600,Math.ceil(len/4),count||480));
      var ctm=el.getCTM();
      for(var i=0;i<=steps;i++){
        var p=el.getPointAtLength(len*i/steps);
        if(ctm){ var x=p.x*ctm.a+p.y*ctm.c+ctm.e; var y=p.x*ctm.b+p.y*ctm.d+ctm.f; points.push({x:x,y:y}); }
        else points.push({x:p.x,y:p.y});
      }
    }catch(e){}
    return points;
  }

  function objectToSvg(o){
    var fill=(o.fillEnabled===false||o.type==="line")?"none":(o.fillColor||o.color||"#4fe0c4");
    var stroke=o.strokeEnabled===false?"none":(o.strokeColor||o.color||"#1b1f24");
    var common=' fill="'+esc(fill)+'" stroke="'+esc(stroke)+'" stroke-width="'+n(o.strokeWidth,2)+'" opacity="'+n(o.opacity,1)+'"';
    if(o.type==="freehand") return '<'+(o.closedPath?'polygon':'polyline')+' points="'+(o.points||[]).map(function(p){return round3(p.x)+','+round3(p.y);}).join(' ')+'"'+common+' />';
    if(o.type==="line") return '<line x1="'+n(o.x1,0)+'" y1="'+n(o.y1,0)+'" x2="'+n(o.x2,0)+'" y2="'+n(o.y2,0)+'"'+common+' />';
    if(o.type==="rect") return '<rect x="'+Math.min(n(o.x1,0),n(o.x2,0))+'" y="'+Math.min(n(o.y1,0),n(o.y2,0))+'" width="'+Math.abs(n(o.x2,0)-n(o.x1,0))+'" height="'+Math.abs(n(o.y2,0)-n(o.y1,0))+'" rx="'+n(o.cornerRadius,0)+'"'+common+' />';
    if(o.type==="ellipse") return '<ellipse cx="'+((n(o.x1,0)+n(o.x2,0))/2)+'" cy="'+((n(o.y1,0)+n(o.y2,0))/2)+'" rx="'+Math.abs(n(o.x2,0)-n(o.x1,0))/2+'" ry="'+Math.abs(n(o.y2,0)-n(o.y1,0))/2+'"'+common+' />';
    if(o.type==="text") return '<text x="'+n(o.x,0)+'" y="'+n(o.y,0)+'" font-size="'+n(o.fontSize,32)+'" font-family="'+esc(o.fontFamily||"Arial")+'"'+common+'>'+esc(o.text||"")+'</text>';
    return "";
  }

  function roundedRectPoints(o){
    var x1=Math.min(n(o.x1,0),n(o.x2,0)),y1=Math.min(n(o.y1,0),n(o.y2,0));
    var x2=Math.max(n(o.x1,0),n(o.x2,0)),y2=Math.max(n(o.y1,0),n(o.y2,0));
    var w=x2-x1,h=y2-y1,r=clamp(n(o.cornerRadius,0),0,Math.min(w,h)/2);
    if(r<=.001)return[{x:x1,y:y1},{x:x2,y:y1},{x:x2,y:y2},{x:x1,y:y2}];
    var segments=clamp(Math.ceil(r/4),4,18),pts=[];
    function arc(cx,cy,start,end,includeStart){
      for(var i=includeStart?0:1;i<=segments;i++){
        var a=start+(end-start)*(i/segments);
        pts.push({x:cx+Math.cos(a)*r,y:cy+Math.sin(a)*r});
      }
    }
    arc(x2-r,y1+r,-Math.PI/2,0,true);
    arc(x2-r,y2-r,0,Math.PI/2,false);
    arc(x1+r,y2-r,Math.PI/2,Math.PI,false);
    arc(x1+r,y1+r,Math.PI,Math.PI*1.5,false);
    return pts;
  }

  function bakeTransformIntoPoints(o,eff,pts){
    var base=objectBounds(o),px=(base.x1+base.x2)/2,py=(base.y1+base.y2)/2;
    var tx=n(eff&&eff.tx,n(o.tx,0)),ty=n(eff&&eff.ty,n(o.ty,0));
    var scale=n(eff&&eff.tscale,n(o.tscale,1));
    var angle=n(eff&&eff.trotation,n(o.trotation,0))*Math.PI/180;
    var c=Math.cos(angle),si=Math.sin(angle);
    return (pts||[]).map(function(p){
      var dx=(n(p.x,0)-px)*scale,dy=(n(p.y,0)-py)*scale;
      return{x:round3(px+tx+dx*c-dy*si),y:round3(py+ty+dx*si+dy*c)};
    });
  }

  function uniformIngestPoints(points,closedPath,maxStep){
    var source=(points||[]).map(function(p){return{x:n(p.x,0),y:n(p.y,0)};});
    if(source.length<2)return source;
    var out=[];
    var segmentCount=closedPath?source.length:source.length-1;
    var step=Math.max(2,n(maxStep,18));
    for(var i=0;i<segmentCount;i++){
      var a=source[i],b=source[(i+1)%source.length];
      out.push({x:a.x,y:a.y});
      var distance=Math.hypot(b.x-a.x,b.y-a.y);
      var divisions=Math.max(1,Math.ceil(distance/step));
      for(var j=1;j<divisions;j++){
        var t=j/divisions;
        out.push({x:round3(a.x+(b.x-a.x)*t),y:round3(a.y+(b.y-a.y)*t)});
      }
    }
    if(!closedPath)out.push({x:source[source.length-1].x,y:source[source.length-1].y});
    return out;
  }

  function copySkinBezierHandles(handles,count){
    var source=Array.isArray(handles)?handles:[];
    var out=[];
    for(var i=0;i<count;i++){
      var h=source[i]||{};
      out.push({
        inX:n(h.inX,0),
        inY:n(h.inY,0),
        outX:n(h.outX,0),
        outY:n(h.outY,0)
      });
    }
    return out;
  }

  function hasRealBezierGeometry(o){
    if(!o||!Array.isArray(o.points)||!Array.isArray(o.bezierHandles)||o.bezierHandles.length!==o.points.length)return false;
    return o.bezierHandles.some(function(h){
      return Math.abs(n(h&&h.inX,0))>.0001||
             Math.abs(n(h&&h.inY,0))>.0001||
             Math.abs(n(h&&h.outX,0))>.0001||
             Math.abs(n(h&&h.outY,0))>.0001;
    });
  }

  function densifyStraightWarpPath(o,maxStep){
    if(!o||!Array.isArray(o.points)||o.points.length<2||hasRealBezierGeometry(o))return o;
    var source=o.points.map(function(p){return{x:n(p.x,0),y:n(p.y,0)};});
    var closed=!!o.closedPath;
    var step=Math.max(4,n(maxStep,12));
    var out=[];
    var segmentCount=closed?source.length:source.length-1;

    for(var i=0;i<segmentCount;i++){
      var a=source[i],b=source[(i+1)%source.length];
      out.push({x:a.x,y:a.y});
      var distance=Math.hypot(b.x-a.x,b.y-a.y);
      var divisions=Math.max(1,Math.ceil(distance/step));
      for(var j=1;j<divisions;j++){
        var t=j/divisions;
        out.push({
          x:round3(a.x+(b.x-a.x)*t),
          y:round3(a.y+(b.y-a.y)*t)
        });
      }
    }

    if(!closed)out.push(clone(source[source.length-1]));

    if(out.length!==source.length){
      o.points=out;
      if(Array.isArray(o.bezierHandles))o.bezierHandles=copySkinBezierHandles([],out.length);
      if(Array.isArray(o.bezierModes))o.bezierModes=out.map(function(){return"corner";});
      o.demBonesAutoDensified=true;
    }

    return o;
  }

  function limbIngestPoints(o){
    var b=objectBounds(o);
    var x0=b.x1,y0=b.y1,x1=b.x2,y1=b.y2;
    var w=Math.max(0,x1-x0),h=Math.max(0,y1-y0);
    if(w<=.001||h<=.001)return Array.isArray(o&&o.points)?clone(o.points):[];
    var vertical=h>=w;
    var jointFrac=.55;
    if(vertical){
      var gapY=Math.max(2,h*.035);
      var midY=y0+h*jointFrac;
      return[
        {x:x0,y:y0},{x:x1,y:y0},
        {x:x1,y:midY-gapY*.5},{x:x1,y:midY+gapY*.5},
        {x:x1,y:y1},{x:x0,y:y1},{x:x0,y:midY}
      ];
    }
    var gapX=Math.max(2,w*.035);
    var midX=x0+w*jointFrac;
    return[
      {x:x0,y:y0},{x:x0,y:y1},
      {x:midX-gapX*.5,y:y1},{x:midX+gapX*.5,y:y1},
      {x:x1,y:y1},{x:x1,y:y0},{x:midX,y:y0}
    ];
  }

  function applyIngestPointMode(o,mode){
    if(!o||!Array.isArray(o.points)||o.points.length<2)return o;
    mode=mode||"selected";
    if(mode==="uniform"){
      o.points=uniformIngestPoints(o.points,!!o.closedPath,10);
      if(Array.isArray(o.bezierHandles)&&!hasRealBezierGeometry(o))o.bezierHandles=copySkinBezierHandles([],o.points.length);
      o.demBonesPointMode="uniform";
    }else if(mode==="limb"){
      o.points=limbIngestPoints(o);
      o.closedPath=true;
      o.demBonesPointMode="limb";
      o.demBonesLimbPoints=7;
    }else{
      o.points=o.points.map(function(p){return{x:round3(p.x),y:round3(p.y)};});
      /*
        Existing Bezier curves keep their exact anchors/handles. Straight and
        sampled/polyline artwork gets extra collinear anchors only where needed;
        this does not alter the visible source outline but gives the skin enough
        vertices to bend smoothly between bones.
      */
      densifyStraightWarpPath(o,12);
      o.demBonesPointMode="selected";
    }
    return o;
  }

  function convertToWarpPath(o,pointMode){
    if(!supportedWarpObject(o)) return null;
    var originalType=o.type;
    ensureObjectId(o);

    /* Use Luke Animate's own permanent vector conversion first. This is the
       authoritative path for rounded rectangles, ellipses, current transforms
       and native editable points. It prevents Dem Bones from replacing the
       selected artwork with a crude bounding-box polygon. */
    try{
      if(typeof API.convertDrawObjectToEditablePath==="function"){
        var converted=API.convertDrawObjectToEditablePath(o);
        if(converted && Array.isArray(converted.points) && converted.points.length>1){
          converted.demBonesOriginalType=converted.demBonesOriginalType||originalType;
          converted.keyframes=[];
          ensureObjectId(converted);
          return applyIngestPointMode(converted,pointMode||"selected");
        }
      }
    }catch(eNative){}

    var eff=typeof API.getEffectiveObject==="function"?API.getEffectiveObject(o,currentTime()):o;
    var pts=[],closed=false,fill=false;
    if(Array.isArray((eff&&eff.points)||o.points)){
      pts=((eff&&eff.points)||o.points||[]).map(function(p){return{x:n(p.x,0),y:n(p.y,0)};});
      closed=eff&&eff.closedPath!==undefined?!!eff.closedPath:!!o.closedPath;
      fill=eff&&eff.fillEnabled!==undefined?!!eff.fillEnabled:!!o.fillEnabled;
    }else if(originalType==="line") pts=[{x:n(eff.x1,o.x1),y:n(eff.y1,o.y1)},{x:n(eff.x2,o.x2),y:n(eff.y2,o.y2)}];
    else if(originalType==="rect"){pts=roundedRectPoints({x1:n(eff.x1,o.x1),y1:n(eff.y1,o.y1),x2:n(eff.x2,o.x2),y2:n(eff.y2,o.y2),cornerRadius:n(eff.cornerRadius,o.cornerRadius)});closed=true;fill=eff.fillEnabled!==undefined?!!eff.fillEnabled:true;}
    else if(originalType==="ellipse"){pts=ellipsePoints({x1:n(eff.x1,o.x1),y1:n(eff.y1,o.y1),x2:n(eff.x2,o.x2),y2:n(eff.y2,o.y2)},64);closed=true;fill=eff.fillEnabled!==undefined?!!eff.fillEnabled:true;}
    if(pts.length<2) return null;
    pts=bakeTransformIntoPoints(o,eff,pts);
    o.demBonesOriginalType=o.demBonesOriginalType||originalType;
    if(originalType==="rect")o.demBonesOriginalCornerRadius=n(eff.cornerRadius,o.cornerRadius||0);
    o.type="freehand";o.points=pts;o.closedPath=closed;o.fillEnabled=fill;
    ["fillColor","strokeColor","strokeWidth","opacity","strokeEnabled"].forEach(function(key){if(eff&&eff[key]!==undefined)o[key]=eff[key];});
    o.tx=0;o.ty=0;o.tscale=1;o.trotation=0;o.keyframes=[];
    if(o.strokeEnabled===undefined)o.strokeEnabled=true;if(o.strokeWidth===undefined)o.strokeWidth=3;
    ensureObjectId(o);return applyIngestPointMode(o,pointMode||"selected");
  }

  }

  }

  function liveRigById(rigId,fallbackRig){

  }

  function demBonesEvaluationTime(data,fallbackTime,api){

    var part=data?characterPartForRigId(data.rigId):null;
    if(part&&api&&isFinite(Number(api.mainTimelineTime))){
      return Number(api.mainTimelineTime);
    }
    return time;
  }

  function syncCharacterPartRigRefs(part){
    if(!part||!part.rig||!part.clipId)return false;
    var clip=clips()[part.clipId];
    if(!clip)return false;
    var rig=part.rig,changed=false;
    function visit(object){
      if(!object)return;
      if(object.demBones&&String(object.demBones.rigId)===String(rig.id)){
        if(object.demBones.rig!==rig){object.demBones.rig=rig;changed=true;}
      }
      if(object.demBonesExpression&&String(object.demBonesExpression.rigId)===String(rig.id)){
        if(object.demBonesExpression.rig!==rig){object.demBonesExpression.rig=rig;changed=true;}
      }
      if(object.type==='group'&&Array.isArray(object.children))object.children.forEach(visit);
    }
    (clip.drawObjects||[]).forEach(visit);
    (clip.sceneFrames||[]).forEach(function(frame){(frame&&frame.objects||[]).forEach(visit);});
    (clip.drawingLayers||[]).forEach(function(layer){(layer&&layer.frames||[]).forEach(function(frame){if(frame&&frame.object)visit(frame.object);});});
    return changed;
  }

  function syncAllCharacterRigRefs(){
    (state.character&&state.character.parts||[]).forEach(syncCharacterPartRigRefs);
  }

  function isRasterWarpObject(o){
    return !!(o&&o.type==="image"&&o.src);
  }

  function createRasterSkinData(o,rig){
    ensureObjectId(o);
    var x=n(o.x,0),y=n(o.y,0),w=Math.max(1,n(o.width,1)),h=Math.max(1,n(o.height,1));
    var cols=clamp(Math.round(w/36),6,22);
    var rows=clamp(Math.round(h/36),6,22);
    var points=[];
    for(var row=0;row<=rows;row++){
      for(var col=0;col<=cols;col++){
        points.push({x:x+w*(col/cols),y:y+h*(row/rows)});
      }
    }
    o.tx=0;o.ty=0;o.tscale=1;o.trotation=0;o.keyframes=[];
    o.demBones={
      version:5,
      rigId:rig.id,
      rig:rig,
      raster:true,
      rasterCols:cols,
      rasterRows:rows,
      sourceX:x,
      sourceY:y,
      sourceWidth:w,
      sourceHeight:h,
      restPoints:points,
      weights:[],
      closedPath:false,
      weightModel:""
    };
    return o;
  }

  function triangleMatrix(s0,s1,s2,d0,d1,d2){
    var sx0=s0.x,sy0=s0.y,sx1=s1.x,sy1=s1.y,sx2=s2.x,sy2=s2.y;
    var dx0=d0.x,dy0=d0.y,dx1=d1.x,dy1=d1.y,dx2=d2.x,dy2=d2.y;
    var denom=sx0*(sy1-sy2)+sx1*(sy2-sy0)+sx2*(sy0-sy1);
    if(Math.abs(denom)<.0000001)return null;
    return{
      a:(dx0*(sy1-sy2)+dx1*(sy2-sy0)+dx2*(sy0-sy1))/denom,
      b:(dy0*(sy1-sy2)+dy1*(sy2-sy0)+dy2*(sy0-sy1))/denom,
      c:(dx0*(sx2-sx1)+dx1*(sx0-sx2)+dx2*(sx1-sx0))/denom,
      d:(dy0*(sx2-sx1)+dy1*(sx0-sx2)+dy2*(sx1-sx0))/denom,
      e:(dx0*(sx1*sy2-sx2*sy1)+dx1*(sx2*sy0-sx0*sy2)+dx2*(sx0*sy1-sx1*sy0))/denom,
      f:(dy0*(sx1*sy2-sx2*sy1)+dy1*(sx2*sy0-sx0*sy2)+dy2*(sx0*sy1-sx1*sy0))/denom
    };
  }

  var rasterImageCache={};
  function rasterImageForSource(source){
    var src=String(source||"");
    if(!src)return null;
    var record=rasterImageCache[src];
    if(record)return record.image;
    var image=new Image();
    record={image:image,loaded:false,failed:false};
    rasterImageCache[src]=record;
    image.onload=function(){record.loaded=true;try{API.refreshUI();}catch(ignore){}};
    image.onerror=function(){record.failed=true;};
    image.src=src;
    return image;
  }

  function drawRasterTriangle(ctx,image,s0,s1,s2,d0,d1,d2){
    var m=triangleMatrix(s0,s1,s2,d0,d1,d2);
    if(!m)return;
    ctx.save();
    ctx.beginPath();
    ctx.moveTo(d0.x,d0.y);ctx.lineTo(d1.x,d1.y);ctx.lineTo(d2.x,d2.y);ctx.closePath();
    ctx.clip();
    ctx.transform(m.a,m.b,m.c,m.d,m.e,m.f);
    ctx.drawImage(image,0,0);
    ctx.restore();
  }

  function rasterObjectRenderer(ctx,object,effective,api){
    if(!object||object.type!=="image"||!object.demBones||!object.demBones.raster)return false;
    var data=object.demBones;
    var liveRig=liveRigForDemBonesData(data);
    if(!liveRig)return false;
    var image=rasterImageForSource(effective&&effective.src!==undefined?effective.src:object.src);
    if(!image||!image.complete||!image.naturalWidth||!image.naturalHeight)return true;
    var evalTime=demBonesEvaluationTime(data,n(api&&api.time,currentTime()),api);
    var geometry=deformedObjectGeometry(object,evalTime);
    if(!geometry||!Array.isArray(geometry.points))return true;
    var cols=Math.max(1,Math.round(n(data.rasterCols,8))),rows=Math.max(1,Math.round(n(data.rasterRows,8)));
    var expected=(cols+1)*(rows+1);
    if(geometry.points.length!==expected)return true;
    var iw=image.naturalWidth,ih=image.naturalHeight;
    function idx(row,col){return row*(cols+1)+col;}
    function src(row,col){return{x:iw*(col/cols),y:ih*(row/rows)};}
    for(var row=0;row<rows;row++){
      for(var col=0;col<cols;col++){
        var d00=geometry.points[idx(row,col)],d10=geometry.points[idx(row,col+1)],d01=geometry.points[idx(row+1,col)],d11=geometry.points[idx(row+1,col+1)];
        var s00=src(row,col),s10=src(row,col+1),s01=src(row+1,col),s11=src(row+1,col+1);
        drawRasterTriangle(ctx,image,s00,s10,s11,d00,d10,d11);
        drawRasterTriangle(ctx,image,s00,s11,s01,d00,d11,d01);
      }
    }
    return true;
  }

  function loadPngFile(file){
    if(!file)return;
    if(typeof API.registerObjectRenderer!=="function"){
      notify("PNG bone deformation needs the supplied v173 index file.",true);
      return;
    }
    var reader=new FileReader();
    reader.onload=function(){
      var dataUrl=reader.result;
      var image=new Image();
      image.onload=function(){
        var topLeft=API.stageToScenePoint({x:0,y:0});
        var bottomRight=API.stageToScenePoint({x:API.width,y:API.height});
        var viewW=Math.abs(bottomRight.x-topLeft.x),viewH=Math.abs(bottomRight.y-topLeft.y);
        var scale=Math.min(1,(viewW*.62)/Math.max(1,image.naturalWidth),(viewH*.62)/Math.max(1,image.naturalHeight));
        var w=Math.max(1,image.naturalWidth*scale),h=Math.max(1,image.naturalHeight*scale);
        var centre=currentStageCentreScene();
        var object={
          type:"image",id:uid("db-png"),name:(file.name||"PNG Part").replace(/\.[^.]+$/,"")||"PNG Part",
          src:dataUrl,x:centre.x-w/2,y:centre.y-h/2,width:w,height:h,
          naturalWidth:image.naturalWidth,naturalHeight:image.naturalHeight,
          opacity:1,tx:0,ty:0,tscale:1,trotation:0,keyframes:[],mediaStartTime:0
        };
        API.pushUndo();
        var rig=createRigFromObjects([object],object.name,state.pointMode);
        if(!rig)return;
        API.addDrawObject(object,"animate");
        state.lastStageSelectionIds=[object.id];
        state.tab="bones";state.mode="add-bone";
        try{API.selectDrawObject(object);}catch(ignoreSelect){}
        renderAll();API.refreshUI();
        notify("PNG loaded. Add bones, then Bind.",false);
      };
      image.onerror=function(){notify("PNG load failed.",true);};
      image.src=dataUrl;
    };
    reader.readAsDataURL(file);
  }

  var lastCreatedRigObjects=[];
  function createRigFromObjects(objects,name,pointMode){
    var supplied=(objects||[]).filter(supportedWarpObject);
    if(!supplied.length){notify("No usable vector or PNG artwork was supplied.",true);return null;}
    var rig=createEmptyRig(name),chosen=[];
    supplied.forEach(function(source){
      var o;
      if(isRasterWarpObject(source)){
        o=createRasterSkinData(source,rig);
      }else{
        o=convertToWarpPath(source,pointMode||"selected");
      }
      if(!o)return;
      if(!isRasterWarpObject(o)&&(!Array.isArray(o.points)||o.points.length<2))return;
      chosen.push(o);
      rig.targetIds.push(o.id);
      if(!o.demBones||!o.demBones.raster){
        o.demBones={version:4,rigId:rig.id,rig:rig,pointMode:o.demBonesPointMode||pointMode||"selected",restPoints:(o.points||[]).map(function(p){return{x:n(p.x,0),y:n(p.y,0)};}),weights:[],closedPath:!!o.closedPath};
      }
    });
    if(!chosen.length){notify("The supplied artwork could not be ingested.",true);return null;}
    state.activeRig=rig;state.selectedBoneId="";state.selectedCurveId=chosen[0].type==="image"?"":chosen[0].id;state.selectedPointIndex=0;syncRigRefs(rig);
    lastCreatedRigObjects=chosen;
    registerLooseBuildRig(rig,chosen,[]);
    return rig;
  }

  function useStageSelection(){
    /* Read the real Luke Animate selection at the moment the button is pressed.
       If opening the floating panel temporarily changed focus, fall back to the
       object ids captured immediately before the panel suppressed the inspector. */
    var chosen=selectedObjects(true).filter(supportedWarpObject);
    if(!chosen.length){
      notify("No selected Luke Animate vector or PNG was found. Select the artwork, then click Use Stage Selection.",true);
      return;
    }
    API.pushUndo();
    var working=[],toStage=[];
    chosen.forEach(function(source){
      var target=source;
      if(!state.removeOriginals){
        target=clone(source);
        target.id=uid("db-copy");
        target.keyframes=[];
        toStage.push(target);
      }
      working.push(target);
    });
    var rig=createRigFromObjects(working,"Stage Part",state.pointMode);
    if(rig){
      toStage.forEach(function(o){API.addDrawObject(o,"animate");});
      state.lastStageSelectionIds=working.map(function(o){return o.id;});
      try{API.selectDrawObject(working[0]);}catch(eSelect){}
      state.tab="bones";state.mode="add-bone";renderAll();
      API.refreshUI();
      var modeLabel=state.pointMode==="limb"?"Limb":(state.pointMode==="uniform"?"Uniform":"Selected points");
      notify("Ingested "+working.length+" selected vector"+(working.length===1?"":"s")+" using "+modeLabel+". Drag to create the first bone; keep dragging to extend the chain.",false);
    }
  }

  function pointToSegmentInfo(p,a,b){
    var dx=b.x-a.x,dy=b.y-a.y,l=dx*dx+dy*dy;
    if(l<.000001)return{d:Math.hypot(p.x-a.x,p.y-a.y),t:0,x:a.x,y:a.y};
    var t=clamp(((p.x-a.x)*dx+(p.y-a.y)*dy)/l,0,1),x=a.x+dx*t,y=a.y+dy*t;
    return{d:Math.hypot(p.x-x,p.y-y),t:t,x:x,y:y};
  }
  function distancePointToSegment(p,a,b){return pointToSegmentInfo(p,a,b).d;}
  function angleOf(a,b){return Math.atan2(b.y-a.y,b.x-a.x);}
  function rotateVector(x,y,a){var c=Math.cos(a),s=Math.sin(a);return{x:x*c-y*s,y:x*s+y*c};}
  function inverseRotateVector(x,y,a){return rotateVector(x,y,-a);}
  function easingValue(v,e){var t=clamp(v,0,1);if(e==="ease-in")return t*t;if(e==="ease-out")return 1-Math.pow(1-t,2);if(e==="ease-in-out")return t<.5?2*t*t:1-Math.pow(-2*t+2,2)/2;if(e==="hold")return 0;return t;}

  function defaultBoneEnvelope(rig,bone){
    var b=rigTargetBounds(rig),w=Math.max(1,b.x2-b.x1),h=Math.max(1,b.y2-b.y1);
    var cross=Math.min(w,h),len=Math.max(1,Math.hypot(bone.restTail.x-bone.restHead.x,bone.restTail.y-bone.restHead.y));
    return clamp(Math.max(12,cross*.62,len*.12),12,140);
  }

  function normaliseRigBones(rig){
    if(!rig)return;
    var by={};
    (rig.bones||[]).forEach(function(b){
      b.restHead=b.restHead||clone(b.head)||{x:0,y:0};
      b.restTail=b.restTail||clone(b.tail)||{x:b.restHead.x+40,y:b.restHead.y};
      b.restLength=Math.max(.001,n(b.restLength,Math.hypot(b.restTail.x-b.restHead.x,b.restTail.y-b.restHead.y)));
      b.restAngle=n(b.restAngle,angleOf(b.restHead,b.restTail));
      var defaultEnv=defaultBoneEnvelope(rig,b);
      b.envStart=n(b.envStart,0)>0?Math.max(2,n(b.envStart,defaultEnv)):defaultEnv;
      b.envEnd=n(b.envEnd,0)>0?Math.max(2,n(b.envEnd,b.envStart)):b.envStart;
      b.strength=Math.max(.001,n(b.strength,1));
      b.basePose=b.basePose||{dx:0,dy:0,rotation:0,scale:1,easing:"linear"};
      b.keys=Array.isArray(b.keys)?b.keys:[];
      by[b.id]=b;
    });
    (rig.bones||[]).forEach(function(b){
      if(b.parentId&&by[b.parentId]){
        var parent=by[b.parentId];
        b.restAngle=angleOf(b.restHead,b.restTail);
        b.restLength=Math.max(.001,Math.hypot(b.restTail.x-b.restHead.x,b.restTail.y-b.restHead.y));
        b.localRestAngle=b.restAngle-angleOf(parent.restHead,parent.restTail);
      }else{
        b.parentId="";
        b.restAngle=angleOf(b.restHead,b.restTail);
        b.restLength=Math.max(.001,Math.hypot(b.restTail.x-b.restHead.x,b.restTail.y-b.restHead.y));
        b.localRestAngle=b.restAngle;
      }
    });
  }

  function smoothEnvelopeResponse(value){
    /*
      Quintic smootherstep. Both the centre and the envelope edge have a zero
      first derivative, which removes the visible "crease" created by a sudden
      influence slope change at the edge of a bone envelope.
    */
    var x=clamp(n(value,0),0,1);
    return x*x*x*(x*(x*6-15)+10);
  }

  function envelopeWeightForPoint(point,bone){
    var info=pointToSegmentInfo(point,bone.restHead,bone.restTail);
    var radius=Math.max(
      1,
      n(bone.envStart,30)+
      (
        n(bone.envEnd,bone.envStart)-
        n(bone.envStart,30)
      )*
      info.t
    );

    if(info.d>=radius)return 0;

    var influence=1-info.d/radius;

    return smoothEnvelopeResponse(influence)*
      Math.max(.001,n(bone.strength,1));
  }

  function buildPointWeights(point,rig){
    var list=[];

    (rig.bones||[]).forEach(function(b){
      var w=envelopeWeightForPoint(point,b);
      if(w>.000001)list.push({
        boneId:b.id,
        weight:w
      });
    });

    if(!list.length){
      var nearest=null;

      (rig.bones||[]).forEach(function(b){
        var d=distancePointToSegment(
          point,
          b.restHead,
          b.restTail
        );

        if(!nearest||d<nearest.d)nearest={
          boneId:b.id,
          d:d
        };
      });

      if(nearest)list=[{
        boneId:nearest.boneId,
        weight:1
      }];
    }

    return normaliseWeightList(list);
  }

  function rebuildObjectSkinData(o,rig){
    if(!o||!o.demBones)return;

    var d=o.demBones;
    var points=(
      d.raster
        ?(d.restPoints||[])
        :(o.points&&o.points.length?o.points:(d.restPoints||[]))
    ).map(function(p){
      return{
        x:n(p.x,0),
        y:n(p.y,0)
      };
    });

    d.version=4;
    d.restPoints=points;
    d.weights=points.map(function(p){
      return buildPointWeights(p,rig);
    });

    /*
      Native Luke Animate Bezier handles are offsets from each anchor. Skin
      their absolute control points separately, then reconstruct the offsets
      after deformation. This is the critical v8 change that stops imported
      SVG/native curves from developing lumpy or wrongly oriented bends.
    */
    if(
      !d.raster&&
      Array.isArray(o.bezierHandles)&&
      o.bezierHandles.length===points.length
    ){
      d.restBezierHandles=copySkinBezierHandles(
        o.bezierHandles,
        points.length
      );

      d.handleWeights=d.restBezierHandles.map(function(handle,index){
        var anchor=points[index];

        var inPoint={
          x:anchor.x+n(handle.inX,0),
          y:anchor.y+n(handle.inY,0)
        };

        var outPoint={
          x:anchor.x+n(handle.outX,0),
          y:anchor.y+n(handle.outY,0)
        };

        return{
          in:buildPointWeights(inPoint,rig),
          out:buildPointWeights(outPoint,rig)
        };
      });
    }else{
      d.restBezierHandles=[];
      d.handleWeights=[];
    }

    d.weightModel="envelope-v2-smooth";
  }

  function rebuildRigSkinData(rig){

    rigObjects(rig).forEach(function(o){

    syncRigRefs(rig);
  }

  function nearestBoneEndpoint(point,ignoreId){
    var rig=state.activeRig,best=null;
    if(!rig)return null;
    (rig.bones||[]).forEach(function(b){
      if(ignoreId&&String(b.id)===String(ignoreId))return;
      [["head",b.restHead],["tail",b.restTail]].forEach(function(entry){
        var d=Math.hypot(point.x-entry[1].x,point.y-entry[1].y);
        if(!best||d<best.distance)best={bone:b,type:entry[0],point:entry[1],distance:d};
      });
    });
    return best;
  }

  function autoConnectNewBone(a,b){
    var out={head:clone(a),tail:clone(b),parentId:"",connected:false};
    var rig=state.activeRig;
    if(!rig||!rig.bones.length)return out;
    var threshold=scenePixels(18);
    var headHit=nearestBoneEndpoint(a,null);
    var tailHit=nearestBoneEndpoint(b,null);
    var parent=null;
    if(headHit&&headHit.type==="tail"&&headHit.distance<=threshold){
      parent=headHit.bone;
      var dx=headHit.point.x-out.head.x,dy=headHit.point.y-out.head.y;
      out.head={x:headHit.point.x,y:headHit.point.y};
      out.tail={x:out.tail.x+dx,y:out.tail.y+dy};
    }else if(tailHit&&tailHit.type==="tail"&&tailHit.distance<=threshold){
      parent=tailHit.bone;
      out.head={x:tailHit.point.x,y:tailHit.point.y};
      out.tail=clone(a);
    }else{
      parent=selectedBone();
      if(parent){
        var sx=parent.restTail.x-out.head.x,sy=parent.restTail.y-out.head.y;
        out.head={x:parent.restTail.x,y:parent.restTail.y};
        out.tail={x:out.tail.x+sx,y:out.tail.y+sy};
      }
    }
    if(parent){out.parentId=parent.id;out.connected=true;}
    if(Math.hypot(out.tail.x-out.head.x,out.tail.y-out.head.y)<5){
      out.tail={x:out.head.x+20,y:out.head.y};
    }
    return out;
  }

  function enforceRestChains(){
    var rig=state.activeRig;
    if(!rig)return;
    var by={};
    (rig.bones||[]).forEach(function(b){by[b.id]=b;});
    var visiting={};
    function solve(parent){
      if(!parent||visiting[parent.id])return;
      visiting[parent.id]=true;
      (rig.bones||[]).forEach(function(child){
        if(!child||String(child.parentId)!==String(parent.id))return;
        var dx=parent.restTail.x-child.restHead.x,dy=parent.restTail.y-child.restHead.y;
        child.restHead={x:child.restHead.x+dx,y:child.restHead.y+dy};
        child.restTail={x:child.restTail.x+dx,y:child.restTail.y+dy};
        solve(child);
      });
      delete visiting[parent.id];
    }
    (rig.bones||[]).forEach(function(b){if(!b.parentId||!by[b.parentId])solve(b);});
  }

  function deleteBoneById(boneId){
    var rig=state.activeRig;
    if(!rig||!Array.isArray(rig.bones))return false;

    var bone=rig.bones.find(function(item){
      return String(item.id)===String(boneId);
    });

    if(!bone)return false;

    API.pushUndo();

    /*
      v9:
      Direct children survive deletion. They are detached as root bones and
      keep their existing rest geometry so removing an unwanted parent does not
      make the remaining rig jump to another joint.
    */
    rig.bones.forEach(function(child){
      if(String(child.parentId)===String(bone.id)){
        child.parentId="";
        child.restAngle=angleOf(
          child.restHead,
          child.restTail
        );
        child.localRestAngle=child.restAngle;
      }
    });

    rig.bones=rig.bones.filter(function(item){
      return String(item.id)!==String(bone.id);
    });

    if(String(state.selectedBoneId)===String(bone.id)){
      state.selectedBoneId=rig.bones.length
        ?rig.bones[rig.bones.length-1].id
        :"";
    }

    state.mode=state.tab==="bones"
      ?"add-bone"
      :"select";

    normaliseRigBones(rig);

    if(rig.bones.length){
      rebuildRigSkinData(rig);
    }else{
      rigObjects(rig).forEach(function(o){
        if(o&&o.demBones){
          o.demBones.weights=[];
          o.demBones.handleWeights=[];
        }
      });
      syncRigRefs(rig);
    }

    renderAll();
    API.refreshUI();
    notify(
      (bone.name||"Bone")+" deleted.",
      false
    );

    return true;
  }

  function interpolatePose(bone,time){
    var base=bone.basePose||{dx:0,dy:0,rotation:0,scale:1,easing:"linear"},keys=(bone.keys||[]).slice().sort(function(a,b){return a.time-b.time;});
    function copyPose(p){return{dx:n(p.dx,0),dy:n(p.dy,0),rotation:n(p.rotation,0),scale:Math.max(.01,n(p.scale,1)),easing:p.easing||"linear"};}
    if(!keys.length)return copyPose(base);
    if(time<=keys[0].time){if(keys[0].time<=.0001)return copyPose(keys[0]);var r=easingValue(time/Math.max(.0001,keys[0].time),base.easing||"linear");return{dx:n(base.dx,0)+(n(keys[0].dx,0)-n(base.dx,0))*r,dy:n(base.dy,0)+(n(keys[0].dy,0)-n(base.dy,0))*r,rotation:n(base.rotation,0)+(n(keys[0].rotation,0)-n(base.rotation,0))*r,scale:n(base.scale,1)+(n(keys[0].scale,1)-n(base.scale,1))*r,easing:keys[0].easing||"linear"};}
    if(time>=keys[keys.length-1].time)return copyPose(keys[keys.length-1]);
    for(var i=0;i<keys.length-1;i++){var a=keys[i],b=keys[i+1];if(time>=a.time&&time<=b.time){var q=easingValue((time-a.time)/Math.max(.0001,b.time-a.time),a.easing||"linear");return{dx:n(a.dx,0)+(n(b.dx,0)-n(a.dx,0))*q,dy:n(a.dy,0)+(n(b.dy,0)-n(a.dy,0))*q,rotation:n(a.rotation,0)+(n(b.rotation,0)-n(a.rotation,0))*q,scale:n(a.scale,1)+(n(b.scale,1)-n(a.scale,1))*q,easing:a.easing||"linear"};}}
    return copyPose(base);
  }

  function posedTransformMap(rig,time){
    normaliseRigBones(rig);
    var by={},rest=restTransformMap(rig),out={},resolving={};
    (rig.bones||[]).forEach(function(b){by[b.id]=b;});
    function resolve(b){
      if(!b)return null;if(out[b.id])return out[b.id];if(resolving[b.id])return rest[b.id];
      resolving[b.id]=true;
      var r=rest[b.id],pose=interpolatePose(b,time),head,angle;
      var localScale=Math.max(.01,n(pose.scale,1));
      if(b.parentId&&by[b.parentId]){
        var parent=resolve(by[b.parentId]);
        head=clone(parent.tail);
        angle=parent.angle+n(b.localRestAngle,r.angle-rest[b.parentId].angle)+n(pose.rotation,0)*Math.PI/180;
      }else{
        head={x:r.head.x+n(pose.dx,0),y:r.head.y+n(pose.dy,0)};
        angle=r.angle+n(pose.rotation,0)*Math.PI/180;
      }
      var length=r.length*localScale,off=rotateVector(length,0,angle);
      var result={head:head,tail:{x:head.x+off.x,y:head.y+off.y},angle:angle,length:length,scale:localScale};
      out[b.id]=result;delete resolving[b.id];return result;
    }
    (rig.bones||[]).forEach(resolve);return out;
  }

  function deformPoint(point,weights,rest,posed){
    if(!weights||!weights.length)return{x:point.x,y:point.y};
    var sumX=0,sumY=0,sumW=0;
    weights.forEach(function(w){
      var r=rest[w.boneId],p=posed[w.boneId],ww=Math.max(0,n(w.weight,0));
      if(!r||!p||ww<=0)return;
      var restDir={x:(r.tail.x-r.head.x)/r.length,y:(r.tail.y-r.head.y)/r.length};
      var restPerp={x:-restDir.y,y:restDir.x};
      var posedDir={x:(p.tail.x-p.head.x)/Math.max(.001,p.length),y:(p.tail.y-p.head.y)/Math.max(.001,p.length)};
      var posedPerp={x:-posedDir.y,y:posedDir.x};
      var rel={x:point.x-r.head.x,y:point.y-r.head.y};
      var u=rel.x*restDir.x+rel.y*restDir.y;
      var v=rel.x*restPerp.x+rel.y*restPerp.y;
      var alongScale=p.length/Math.max(.001,r.length);
      var tx=p.head.x+posedDir.x*u*alongScale+posedPerp.x*v;
      var ty=p.head.y+posedDir.y*u*alongScale+posedPerp.y*v;
      sumX+=tx*ww;sumY+=ty*ww;sumW+=ww;
    });
    return sumW>.000001?{x:sumX/sumW,y:sumY/sumW}:{x:point.x,y:point.y};
  }

  function deformBezierHandles(basePoints,deformedPoints,data,rest,posed){
    if(
      !data||
      !Array.isArray(data.restBezierHandles)||
      data.restBezierHandles.length!==basePoints.length||
      !Array.isArray(data.handleWeights)||
      data.handleWeights.length!==basePoints.length
    ){
      return null;
    }

    return data.restBezierHandles.map(function(handle,index){
      var sourceAnchor=basePoints[index];
      var deformedAnchor=deformedPoints[index];
      var weights=data.handleWeights[index]||{};

      var inSource={
        x:sourceAnchor.x+n(handle.inX,0),
        y:sourceAnchor.y+n(handle.inY,0)
      };

      var outSource={
        x:sourceAnchor.x+n(handle.outX,0),
        y:sourceAnchor.y+n(handle.outY,0)
      };

      var inDeformed=deformPoint(
        inSource,
        weights.in||data.weights[index],
        rest,
        posed
      );

      var outDeformed=deformPoint(
        outSource,
        weights.out||data.weights[index],
        rest,
        posed
      );

      return{
        inX:inDeformed.x-deformedAnchor.x,
        inY:inDeformed.y-deformedAnchor.y,
        outX:outDeformed.x-deformedAnchor.x,
        outY:outDeformed.y-deformedAnchor.y
      };
    });
  }

  }

  function characterRootPartInstances(root){
    if(!root)return[];

    var found=[];
    var seenPartIds={};
    var seenObjectIds={};
    var seenRefs=[];

    function add(object){
      if(!object||object.__dbPartId===undefined||object.__dbPartId===null)return;

      var partKey=String(object.__dbPartId);
      var objectKey=object.id!==undefined&&object.id!==null?String(object.id):"";

      /*
        Prefer the first live representation for a part, but inspect every
        storage model. A newly-added second part can exist in drawObjects while
        an older part is already represented by canonical drawingLayers.
        v16.7 stopped after finding any drawing-layer part, which could make the
        later part invisible to finalisation and therefore get pruned.
      */
      if(seenPartIds[partKey])return;
      if(objectKey&&seenObjectIds[objectKey])return;
      if(!objectKey&&seenRefs.indexOf(object)>=0)return;

      seenPartIds[partKey]=true;
      if(objectKey)seenObjectIds[objectKey]=true;
      else seenRefs.push(object);
      found.push(object);
    }

    if(Array.isArray(root.drawingLayers)){
      root.drawingLayers.forEach(function(layer){
        (layer.frames||[]).forEach(function(frame){
          if(frame&&!frame.blank&&frame.object)add(frame.object);
        });
      });
    }

    if(Array.isArray(root.sceneFrames)){
      root.sceneFrames.forEach(function(frame){
        (frame&&frame.objects||[]).forEach(add);
      });
    }

    if(Array.isArray(root.drawObjects)){
      root.drawObjects.forEach(add);
    }

    return found;
  }

  function partInstance(part){
    if(!part||!state.character.rootClipId)return null;
    var root=clips()[state.character.rootClipId];
    if(!root)return null;

    return characterRootPartInstances(root).filter(function(object){
      return Number(object.__dbPartId)===Number(part.id);
    })[0]||null;
  }

  function normalisePartTransform(value){
    value=value||{};
    return{
      x:n(value.x,0),
      y:n(value.y,0),
      scale:Math.max(.01,n(value.scale,1)),
      scaleX:Math.max(.01,n(value.scaleX,1)),
      scaleY:Math.max(.01,n(value.scaleY,1)),
      rotation:n(value.rotation,0),
      opacity:clamp(n(value.opacity,1),0,1)
    };
  }

  function transformFromPartInstance(instance,fallback){
    var base=normalisePartTransform(fallback);
    if(!instance)return base;

    return{
      x:n(instance.x,base.x),
      y:n(instance.y,base.y),
      scale:Math.max(.01,n(instance.scale,base.scale)),
      scaleX:Math.max(.01,n(instance.scaleX,base.scaleX)),
      scaleY:Math.max(.01,n(instance.scaleY,base.scaleY)),
      rotation:n(instance.rotation,base.rotation),
      opacity:clamp(n(instance.opacity,base.opacity),0,1)
    };
  }

  function syncPartTransformFromInstance(part){
    if(!part||part.pin)return false;
    var instance=partInstance(part);
    if(!instance)return false;

    part.transform=transformFromPartInstance(instance,part.transform);
    return true;
  }

  function syncAllUnpinnedCharacterTransforms(){
    state.character.parts.forEach(function(part){
      syncPartTransformFromInstance(part);
    });
  }

  function partLocalPointToCharacterScene(partTransform,point){
    var transform=normalisePartTransform(partTransform);
    var scaled={
      x:n(point&&point.x,0)*transform.scale*transform.scaleX,
      y:n(point&&point.y,0)*transform.scale*transform.scaleY
    };

    var rotated=rotateVector(
      scaled.x,
      scaled.y,
      transform.rotation*Math.PI/180
    );

    return{
      x:transform.x+rotated.x,
      y:transform.y+rotated.y
    };
  }

  function resolvedPartTransform(part,time,visited){
    if(!part)return normalisePartTransform(null);

    visited=visited||{};

    if(visited[part.id]){
      return normalisePartTransform(part.transform);
    }

    visited[part.id]=true;

    var instance=partInstance(part);
    var adjusting=Number(state.adjustingPartId)===Number(part.id);
    var base=part.pin
      ?normalisePartTransform(part.transform)
      :transformFromPartInstance(instance,part.transform);

    /*
      Adjust is a placement edit, not animation. While it is active, the native
      Luke Animate Clip instance is the authority even if this part is pinned.
      That makes the artwork and every displayed bone travel, rotate and scale
      together under the normal stage transform gizmo.
    */
    if(adjusting){
      var liveSource=instance;
      if(instance&&typeof API.getEffectiveObject==="function"){
        try{liveSource=API.getEffectiveObject(instance,time)||instance;}catch(ignoreAdjustedEffective){}
      }
      delete visited[part.id];
      return transformFromPartInstance(liveSource,part.transform);
    }

    if(!part.pin||!part.pin.targetPartId||!part.pin.boneId){
      delete visited[part.id];
      return base;
    }

    var parent=partById(part.pin.targetPartId);
    if(!parent||!parent.rig){
      delete visited[part.id];
      return base;
    }

    var parentTransform=resolvedPartTransform(parent,time,visited);
    var poseMap=posedTransformMap(parent.rig,time);
    var bonePose=poseMap[part.pin.boneId];

    if(!bonePose){
      delete visited[part.id];
      return base;
    }

    var headWorld=partLocalPointToCharacterScene(parentTransform,bonePose.head);
    var tailWorld=partLocalPointToCharacterScene(parentTransform,bonePose.tail);

    var attachPoint=part.pin.attachPoint==="head"?"head":"tail";
    var anchor=attachPoint==="head"?headWorld:tailWorld;

    var targetAngle=Math.atan2(
      tailWorld.y-headWorld.y,
      tailWorld.x-headWorld.x
    )*180/Math.PI;

    var localOffset=part.pin.offsetLocal||{
      x:n(part.pin.offsetX,0),
      y:n(part.pin.offsetY,0)
    };

    var parentScaleNow=Math.max(.0001,n(parentTransform.scale,1));

    var rotatedOffset=rotateVector(
      n(localOffset.x,0)*parentScaleNow,
      n(localOffset.y,0)*parentScaleNow,
      targetAngle*Math.PI/180
    );

    var parentScaleAtPin=Math.max(
      .0001,
      n(part.pin.parentScaleAtPin,parentScaleNow)
    );

    var childScaleAtPin=Math.max(
      .0001,
      n(part.pin.childScaleAtPin,base.scale)
    );

    var scaleRatio=parentScaleNow/parentScaleAtPin;

    var solved={
      x:anchor.x+rotatedOffset.x,
      y:anchor.y+rotatedOffset.y,
      scale:childScaleAtPin*scaleRatio,
      scaleX:Math.max(.01,n(part.pin.childScaleXAtPin,base.scaleX)),
      scaleY:Math.max(.01,n(part.pin.childScaleYAtPin,base.scaleY)),
      rotation:part.pin.inheritRotation===false
        ?n(part.pin.fixedRotation,base.rotation)
        :targetAngle+n(part.pin.rotationOffset,0),
      opacity:base.opacity
    };

    delete visited[part.id];
    return normalisePartTransform(solved);
  }

  function characterRootPointToMainScene(point,time){
    var root=state.character.rootClipId
      ?clips()[state.character.rootClipId]
      :null;

    var instance=characterRootInstance();

    if(!root||!instance){
      return{
        x:n(point&&point.x,0),
        y:n(point&&point.y,0)
      };
    }

    var effective=instance;

    if(typeof API.getEffectiveObject==="function"){
      try{
        effective=API.getEffectiveObject(
          instance,
          time
        )||instance;
      }catch(ignore){}
    }

    var scale=Math.max(.0001,n(effective.scale,1));
    var scaleX=Math.max(.0001,n(effective.scaleX,1));
    var scaleY=Math.max(.0001,n(effective.scaleY,1));

    var local={
      x:(
        n(point&&point.x,0)-
        n(root.pivotX,0)
      )*scale*scaleX,
      y:(
        n(point&&point.y,0)-
        n(root.pivotY,0)
      )*scale*scaleY
    };

    var rotated=rotateVector(
      local.x,
      local.y,
      n(effective.rotation,0)*Math.PI/180
    );

    return{
      x:n(effective.x,0)+rotated.x,
      y:n(effective.y,0)+rotated.y
    };
  }

  function characterBoneDisplayRecords(time){
    if(!state.character.rootClipId||!state.character.parts.length){
      return[];
    }

    var editing=typeof API.getEditingClipId==="function"
      ?API.getEditingClipId()
      :null;

    /*
      Show the complete assembled skeleton in Main Scene or while directly
      editing MyCharacter. Inside an individual part Clip the normal active-rig
      overlay takes over instead.
    */
    if(
      editing&&
      String(editing)!==String(state.character.rootClipId)
    ){
      return[];
    }

    var records=[];

    state.character.parts.forEach(function(part){
      characterPartBoneWorldRecords(
        part,
        time
      ).forEach(function(record){
        var head=record.head;
        var tail=record.tail;

        if(!editing){
          head=characterRootPointToMainScene(
            head,
            time
          );

          tail=characterRootPointToMainScene(
            tail,
            time
          );
        }

        records.push({
          part:part,
          bone:record.bone,
          head:head,
          tail:tail
        });
      });
    });

    return records;
  }

  function characterRestBoneDisplayRecords(time){
    if(!state.character.rootClipId||!state.character.parts.length)return[];

    var editing=typeof API.getEditingClipId==="function"?API.getEditingClipId():null;
    if(editing&&String(editing)!==String(state.character.rootClipId))return[];

    var records=[];
    state.character.parts.forEach(function(part){
      if(!part||!part.rig)return;
      var transform=resolvedPartTransform(part,time,{});
      var map=restTransformMap(part.rig);
      (part.rig.bones||[]).forEach(function(bone){
        var pose=map[bone.id];
        if(!pose)return;
        var head=partLocalPointToCharacterScene(transform,pose.head);
        var tail=partLocalPointToCharacterScene(transform,pose.tail);
        if(!editing){
          head=characterRootPointToMainScene(head,time);
          tail=characterRootPointToMainScene(tail,time);
        }
        records.push({part:part,bone:bone,head:head,tail:tail});
      });
    });
    return records;
  }

  function characterAllEditBoneHit(event){
    var point=eventStagePoint(event);
    var nodeThreshold=stagePixelsFromClient(14);
    var lineThreshold=stagePixelsFromClient(9);
    var best=null;
    characterRestBoneDisplayRecords(currentTime()).forEach(function(record){
      var head=stagePointFromScene(record.head);
      var tail=stagePointFromScene(record.tail);
      var headDistance=Math.hypot(point.x-head.x,point.y-head.y);
      var tailDistance=Math.hypot(point.x-tail.x,point.y-tail.y);
      var lineDistance=distancePointToSegment(point,head,tail);
      var handle="line",distance=lineDistance;
      if(headDistance<=tailDistance&&headDistance<=lineDistance){handle="head";distance=headDistance;}
      else if(tailDistance<=lineDistance){handle="tail";distance=tailDistance;}
      if((headDistance<=nodeThreshold||tailDistance<=nodeThreshold||lineDistance<=lineThreshold)&&(!best||distance<best.distance)){
        best={part:record.part,bone:record.bone,handle:handle,head:record.head,tail:record.tail,distance:distance};
      }
    });
    return best;
  }

  function currentScenePointToPartLocal(part,point,time){
    if(!part)return{x:n(point&&point.x,0),y:n(point&&point.y,0)};
    time=n(time,currentTime());
    var editing=typeof API.getEditingClipId==="function"?API.getEditingClipId():null;
    var charPoint=(editing&&String(editing)===String(state.character.rootClipId))
      ?{x:n(point&&point.x,0),y:n(point&&point.y,0)}
      :mainScenePointToCharacterRoot(point,time);
    var transform=resolvedPartTransform(part,time,{});
    var q=inverseRotateVector(
      charPoint.x-transform.x,
      charPoint.y-transform.y,
      n(transform.rotation,0)*Math.PI/180
    );
    var scale=Math.max(.0001,n(transform.scale,1));
    var sx=Math.max(.0001,n(transform.scaleX,1));
    var sy=Math.max(.0001,n(transform.scaleY,1));
    return{x:q.x/(scale*sx),y:q.y/(scale*sy)};
  }

  function partLocalPointToCurrentCharacterScene(part,point,time){
    time=n(time,currentTime());
    var transform=resolvedPartTransform(part,time,{});
    var charPoint=partLocalPointToCharacterScene(transform,point);
    var editing=typeof API.getEditingClipId==="function"?API.getEditingClipId():null;
    if(!editing)return characterRootPointToMainScene(charPoint,time);
    return charPoint;
  }

  function characterAllEditEnvelopeGeometry(part,bone,time){
    if(!part||!part.rig||!bone)return null;
    var map=restTransformMap(part.rig),q=map[bone.id];
    if(!q)return null;
    var dx=q.tail.x-q.head.x,dy=q.tail.y-q.head.y;
    var len=Math.max(.0001,Math.hypot(dx,dy));
    var perp={x:-dy/len,y:dx/len};
    var r0=Math.max(2,n(bone.envStart,30));
    var r1=Math.max(2,n(bone.envEnd,r0));
    function scene(local){return partLocalPointToCurrentCharacterScene(part,local,time);}
    return{
      head:scene(q.head),tail:scene(q.tail),
      hPlus:scene({x:q.head.x+perp.x*r0,y:q.head.y+perp.y*r0}),
      hMinus:scene({x:q.head.x-perp.x*r0,y:q.head.y-perp.y*r0}),
      tPlus:scene({x:q.tail.x+perp.x*r1,y:q.tail.y+perp.y*r1}),
      tMinus:scene({x:q.tail.x-perp.x*r1,y:q.tail.y-perp.y*r1})
    };
  }

  function characterAllEditEnvelopeHandleHit(event){
    if(!state.characterAllBonesEdit||state.tab!=="editbone"||!state.showEnvelopes)return null;
    var part=partById(state.character.selectedPartId),bone=selectedBone();
    if(!part||!bone)return null;
    var geom=characterAllEditEnvelopeGeometry(part,bone,currentTime());
    if(!geom)return null;
    var point=eventStagePoint(event),threshold=stagePixelsFromClient(10),best=null;
    [
      {end:"start",side:1,p:stagePointFromScene(geom.hPlus)},
      {end:"start",side:-1,p:stagePointFromScene(geom.hMinus)},
      {end:"end",side:1,p:stagePointFromScene(geom.tPlus)},
      {end:"end",side:-1,p:stagePointFromScene(geom.tMinus)}
    ].forEach(function(handle){
      var d=Math.hypot(point.x-handle.p.x,point.y-handle.p.y);
      if(d<=threshold&&(!best||d<best.distance))best={part:part,bone:bone,end:handle.end,side:handle.side,distance:d};
    });
    return best;
  }

  function mainScenePointToCharacterRoot(point,time){
    var root=state.character.rootClipId?clips()[state.character.rootClipId]:null;
    var instance=characterRootInstance();
    if(!root||!instance)return{x:n(point&&point.x,0),y:n(point&&point.y,0)};

    if(typeof API.getEffectiveObject==="function")try{effective=API.getEffectiveObject(instance,time)||instance;}catch(ignore){}
    var q={x:n(point&&point.x,0)-n(effective.x,0),y:n(point&&point.y,0)-n(effective.y,0)};
    q=inverseRotateVector(n(point&&point.x,0)-n(effective.x,0),n(point&&point.y,0)-n(effective.y,0),n(effective.rotation,0)*Math.PI/180);
    var scale=Math.max(.0001,n(effective.scale,1)),sx=Math.max(.0001,n(effective.scaleX,1)),sy=Math.max(.0001,n(effective.scaleY,1));
    return{x:n(root.pivotX,0)+q.x/(scale*sx),y:n(root.pivotY,0)+q.y/(scale*sy)};
  }

  function stageRiggedShapeHit(event){
    var point=selectedScenePoint(event), pad=scenePixels(7), best=null;
    var list=sceneObjects();
    for(var i=list.length-1;i>=0;i--){
      var object=list[i];
      if(!object||!object.demBones||!object.demBones.rig)continue;
      var bounds=objectBounds(object);
      if(!(bounds&&isFinite(bounds.x1)&&isFinite(bounds.y1)&&isFinite(bounds.x2)&&isFinite(bounds.y2)))continue;
      var x1=bounds.x1-pad,x2=bounds.x2+pad,y1=bounds.y1-pad,y2=bounds.y2+pad;
      if(point.x>=x1&&point.x<=x2&&point.y>=y1&&point.y<=y2){
        var area=Math.max(1,(x2-x1)*(y2-y1));
        if(!best||area<best.area)best={object:object,rig:object.demBones.rig,area:area};
      }
    }
    return best;
  }

  function activeRigBoneHitAtTime(event){
    var rig=state.activeRig;if(!rig)return null;
    var p=selectedScenePoint(event),map=posedTransformMap(rig,currentTime());
    var best=null,bestD=scenePixels(14),lineD=scenePixels(9);
    if(state.closedBoneId){
      var selected=(rig.bones||[]).filter(function(b){return String(b.id)===String(state.closedBoneId);})[0],sq=selected&&map[selected.id];
      if(selected&&sq){
        var hd=Math.hypot(p.x-sq.head.x,p.y-sq.head.y),td=Math.hypot(p.x-sq.tail.x,p.y-sq.tail.y);
        if(Math.min(hd,td)<bestD)return{bone:selected,handle:hd<=td?"head":"tail",point:p,distance:Math.min(hd,td)};
      }
    }
    (rig.bones||[]).forEach(function(b){
      var q=map[b.id];if(!q)return;
      [["head",q.head],["tail",q.tail]].forEach(function(pair){var d=Math.hypot(p.x-pair[1].x,p.y-pair[1].y);if(d<bestD&&(!best||d<best.distance))best={bone:b,handle:pair[0],point:p,distance:d};});
      var ld=distancePointToSegment(p,q.head,q.tail);if(ld<lineD&&(!best||ld<best.distance-scenePixels(2)))best={bone:b,handle:"line",point:p,distance:ld};
    });
    return best;
  }

  function characterPartShapeHit(event){
    if(!state.character.parts.length)return null;
    var p=selectedScenePoint(event),best=null;
    state.character.parts.forEach(function(part){
      if(!part||!part.rig)return;
      var localObjects=rigObjects(part.rig),localBounds=combinedBounds(localObjects);
      if(!(localBounds.x2>localBounds.x1&&localBounds.y2>localBounds.y1))return;
      var transform=resolvedPartTransform(part,currentTime(),{});
      var corners=[
        {x:localBounds.x1,y:localBounds.y1},{x:localBounds.x2,y:localBounds.y1},
        {x:localBounds.x2,y:localBounds.y2},{x:localBounds.x1,y:localBounds.y2}
      ].map(function(pt){return partLocalPointToCharacterScene(transform,pt);});
      var editing=typeof API.getEditingClipId==="function"?API.getEditingClipId():null;
      if(!editing)corners=corners.map(function(pt){return characterRootPointToMainScene(pt,currentTime());});
      var xs=corners.map(function(pt){return pt.x;}),ys=corners.map(function(pt){return pt.y;});
      var pad=scenePixels(7),x1=Math.min.apply(null,xs)-pad,x2=Math.max.apply(null,xs)+pad,y1=Math.min.apply(null,ys)-pad,y2=Math.max.apply(null,ys)+pad;
      if(p.x>=x1&&p.x<=x2&&p.y>=y1&&p.y<=y2){var area=(x2-x1)*(y2-y1);if(!best||area<best.area)best={part:part,area:area};}
    });
    return best;
  }

  function characterPartBoneWorldRecords(part,time){
    if(!part||!part.rig)return[];

    var transform=resolvedPartTransform(part,time,{});
    var map=posedTransformMap(part.rig,time);

    return(part.rig.bones||[]).map(function(bone){
      var pose=map[bone.id];
      if(!pose)return null;

      var head=partLocalPointToCharacterScene(transform,pose.head);
      var tail=partLocalPointToCharacterScene(transform,pose.tail);

      return{
        part:part,
        bone:bone,
        head:head,
        tail:tail,
        angle:Math.atan2(
          tail.y-head.y,
          tail.x-head.x
        )*180/Math.PI
      };
    }).filter(Boolean);
  }

  function deformedObjectGeometry(object,time){
    if(!object||!object.demBones)return null;

    var data=object.demBones;
    var rig=liveRigForDemBonesData(data);
    if(!rig)return null;

    if(!Array.isArray(data.restPoints)||!data.restPoints.length)return null;

    normaliseRigBones(rig);

    if(
      data.weightModel!=="envelope-v2-smooth"||
      !Array.isArray(data.weights)||
      data.weights.length!==data.restPoints.length||
      (
        Array.isArray(object.bezierHandles)&&
        object.bezierHandles.length===data.restPoints.length&&
        (
          !Array.isArray(data.restBezierHandles)||
          data.restBezierHandles.length!==data.restPoints.length||
          !Array.isArray(data.handleWeights)||
          data.handleWeights.length!==data.restPoints.length
        )
      )
    ){
      rebuildObjectSkinData(object,rig);
      data=object.demBones;
    }

    var rest=restTransformMap(rig);
    var editingRestOnly=state.tab==="editbone";
    var posed=editingRestOnly?rest:posedTransformMap(rig,time);
    var base=editingRestOnly?data.restPoints:interpolatePointKey(rig,object.id,time,data.restPoints);

    var points=base.map(function(point,index){
      return deformPoint(point,data.weights[index],rest,posed);
    });

    var handles=deformBezierHandles(base,points,data,rest,posed);

    return{
      points:points,
      closedPath:!!data.closedPath,
      bezierHandles:handles
    };
  }

  function syncRigDisplayGeometry(rig,time){
    if(!rig)return false;

    var changed=false;

    rigObjects(rig).forEach(function(object){
      if(object&&object.demBones&&object.demBones.raster)return;
      var geometry=deformedObjectGeometry(object,time);
      if(!geometry)return;

      object.points=geometry.points.map(function(point){
        return{x:n(point.x,0),y:n(point.y,0)};
      });

      object.closedPath=!!geometry.closedPath;

      if(geometry.bezierHandles){
        object.bezierHandles=copySkinBezierHandles(
          geometry.bezierHandles,
          geometry.bezierHandles.length
        );
      }

      changed=true;
    });

    return changed;
  }

  function evaluator(object,time,effective,api){
    if(object&&object.__dbPartId){
      var part=partById(object.__dbPartId);
      if(!part)return effective;

      /* Placement Adjust must never be overwritten by the pin evaluator. */
      if(Number(state.adjustingPartId)===Number(part.id)){
        return effective;
      }

      if(!part.pin){
        return effective;
      }

      var solved=resolvedPartTransform(part,time,{});
      var result=Object.assign({},effective);

      result.x=solved.x;
      result.y=solved.y;
      result.scale=solved.scale;
      result.scaleX=solved.scaleX;
      result.scaleY=solved.scaleY;
      result.rotation=solved.rotation;
      result.opacity=solved.opacity;

      return result;
    }

    if(object&&object.demBonesExpression){
      var ex=object.demBonesExpression;
      ex.rig=liveRigById(ex.rigId,ex.rig);
      if(!ex.rig)return effective;
      var exTime=demBonesEvaluationTime({rigId:ex.rigId},time,api);
      var slot=activeExpressionSlot(ex.rig,ex.type,exTime);
      var r=Object.assign({},effective);
      r.opacity=slot===ex.slot?n(ex.baseOpacity,1):0;
      return r;
    }

    if(object&&object.demBones&&object.demBones.raster){
      var rasterOut=Object.assign({},effective);
      rasterOut.tx=0;rasterOut.ty=0;rasterOut.tscale=1;rasterOut.tscaleX=1;rasterOut.tscaleY=1;rasterOut.trotation=0;
      return rasterOut;
    }
    var geometryTime=object&&object.demBones
      ?demBonesEvaluationTime(object.demBones,time,api)
      :time;
    var geometry=deformedObjectGeometry(object,geometryTime);
    if(!geometry)return effective;

    var out=Object.assign({},effective);
    out.points=geometry.points;
    out.closedPath=!!geometry.closedPath;
    if(geometry.bezierHandles)out.bezierHandles=geometry.bezierHandles;
    out.x=0;out.y=0;out.tx=0;out.ty=0;
    out.scale=1;out.scaleX=1;out.scaleY=1;
    out.tscale=1;out.tscaleX=1;out.tscaleY=1;
    out.rotation=0;out.trotation=0;
    return out;
  }

  function uniqueBuildPartObjects(list){
    var seenIds={},seenRefs=[],out=[];
    (list||[]).forEach(function(object){
      if(!object)return;
      var id=object.id!=null?String(object.id):"";
      if(id){
        if(seenIds[id])return;
        seenIds[id]=true;
      }else{
        if(seenRefs.indexOf(object)>=0)return;
        seenRefs.push(object);
      }
      out.push(object);
    });
    return out;
  }

  function availableBuildPartCandidates(){
    var groups={};

    function ensureGroup(rigId){
      rigId=String(rigId==null?"":rigId);
      if(!rigId)return null;
      return groups[rigId]||(groups[rigId]={rigId:rigId,rig:null,objects:[],expressionObjects:[]});
    }

    /* First use the real main-stage objects when Luke Animate can enumerate
       them. This remains the best source for current live object references. */
    mainSceneObjectsForPartPicker().forEach(function(object){
      if(!object)return;
      var data=object.demBones||object.demBonesExpression;
      var rigId=data&&data.rigId!=null?String(data.rigId):"";
      if(!rigId||sourceRigIdAlreadyInCharacter(rigId))return;
      var group=ensureGroup(rigId);
      if(!group)return;
      if(data&&data.rig&&!group.rig)group.rig=data.rig;
      if(object.demBones)group.objects.push(object);
      else if(object.demBonesExpression)group.expressionObjects.push(object);
    });

    /* v16.10: merge the explicit loose-rig registry. The second imported part
       used to disappear from Add Part because entering Build Character calls
       returnToCharacterRoot(), which intentionally clears state.activeRig.
       The imported artwork still existed, but the picker could then have no
       reliable live rig object to associate with it. The registry survives
       that context switch and is keyed by the unique rig id, not the name. */
    Object.keys(state.looseBuildParts||{}).forEach(function(rigId){
      if(sourceRigIdAlreadyInCharacter(rigId)){
        unregisterLooseBuildRig(rigId);
        return;
      }
      var entry=registeredLooseBuildRig(rigId);
      if(!entry||!entry.rig)return;
      var group=ensureGroup(rigId);
      if(!group)return;
      if(!group.rig)group.rig=entry.rig;
      group.objects=group.objects.concat(entry.objects||[]);
      group.expressionObjects=group.expressionObjects.concat(entry.expressionObjects||[]);
    });

    var candidates=[];
    Object.keys(groups).forEach(function(rigId){
      var group=groups[rigId];
      if(!group)return;
      group.objects=uniqueBuildPartObjects(group.objects);
      group.expressionObjects=uniqueBuildPartObjects(group.expressionObjects);
      if(!group.objects.length)return;
      var rig=group.rig;
      if(!rig&&state.activeRig&&String(state.activeRig.id)===String(rigId))rig=state.activeRig;
      var registry=registeredLooseBuildRig(rigId);
      if(!rig&&registry&&registry.rig)rig=registry.rig;
      if(!rig)return;
      registerLooseBuildRig(rig,group.objects,group.expressionObjects);
      var bounds=combinedBounds(group.objects);
      var name=String(rig.name||"Part");
      candidates.push({
        rigId:rigId,
        rig:rig,
        name:name,
        objects:group.objects.slice(),
        expressionObjects:group.expressionObjects.slice(),
        bounds:bounds,
        label:name+' · x '+Math.round(n(bounds.cx,0))+' y '+Math.round(n(bounds.cy,0))
      });
    });

    candidates.sort(function(a,b){
      var nameCompare=String(a.name).localeCompare(String(b.name));
      if(nameCompare)return nameCompare;
      var yCompare=n(a.bounds&&a.bounds.cy,0)-n(b.bounds&&b.bounds.cy,0);
      if(Math.abs(yCompare)>.001)return yCompare;
      return n(a.bounds&&a.bounds.cx,0)-n(b.bounds&&b.bounds.cx,0);
    });
    return candidates;
  }

  function characterContentOrigin(){

  function migrateCharacterOrigin(root){
    if(!root)return;
    var parts=state.character.parts||[];
    var origin=null;
    if(state.character.originSet&&isFinite(Number(state.character.originX))&&isFinite(Number(state.character.originY))){
      origin={x:Number(state.character.originX),y:Number(state.character.originY)};
    }else{
      var first=parts[0],firstTransform=first&&normalisePartTransform(first.transform);
      if(firstTransform)origin={x:firstTransform.x,y:firstTransform.y};
    }
    if(!origin)origin=characterContentOrigin();
    parts.forEach(function(part){
      var transform=normalisePartTransform(part.transform);
      var relativeX=transform.x-origin.x,relativeY=transform.y-origin.y;
      part.transform=Object.assign({},transform,{x:relativeX,y:relativeY});
      var instance=partInstance(part);
      if(instance){instance.x=relativeX;instance.y=relativeY;}
    });
    state.character.originX=origin.x;state.character.originY=origin.y;state.character.originSet=true;
    root.pivotX=0;root.pivotY=0;
    var rootInstance=characterRootInstance();
    if(rootInstance){rootInstance.x=origin.x;rootInstance.y=origin.y;}
  }

  function createPartClip(part,build){
    var map=clips();
    var bounds=combinedBounds(build.objects);
    var cx=bounds.cx;
    var cy=bounds.cy;
    var rig=clone(build.rig);

    rig.id=uid("db-part-rig");
    shiftRig(rig,-cx,-cy);
    shiftRigPointKeys(rig,-cx,-cy);

    var idMap={};

    var objects=build.objects.map(function(raw){
      var o=clone(raw);
      var old=o.id;

      o.id=uid("db-child-object");
      idMap[old]=o.id;

      /*
        v12:
        shiftObject() already moves BOTH o.points and o.demBones.restPoints.
        Do not subtract cx/cy from restPoints a second time.
      */
      shiftObject(o,-cx,-cy);

      if(o.demBones){
        o.demBones.rigId=rig.id;
        o.demBones.rig=rig;
      }

      return o;
    });

    rig.targetIds=(rig.targetIds||[]).map(function(id){
      return idMap[id]||id;
    });

    if(rig.pointKeys){
      var remapped={};
      Object.keys(rig.pointKeys).forEach(function(id){
        remapped[idMap[id]||id]=rig.pointKeys[id];
      });
      rig.pointKeys=remapped;
    }

    var expressionObjects=(build.expressionObjects||[]).map(function(raw){
      var o=clone(raw);
      o.id=uid("db-child-expression");
      shiftObject(o,-cx,-cy);

      if(o.demBonesExpression){
        o.demBonesExpression.rigId=rig.id;
        o.demBonesExpression.rig=rig;
      }

      return o;
    });

    var clipId=uid("db-part-clip");

    var clip={
      id:clipId,
      name:part.name,
      points:[],
      closed:false,
      targetPoints:[],
      layers:[],
      drawObjects:objects.concat(expressionObjects),
      pivotX:0,
      pivotY:0,
      timelineDuration:Math.max(3,n(API.getTimelineDuration(),6)),
      loopTimeline:true,
      __demBonesPart:true,
      __dbPartId:part.id
    };

    map[clipId]=clip;

    part.clipId=clipId;
    part.rig=rig;
    part.source=clone(build);
    part.localOrigin={x:cx,y:cy};

    if(!part.transform){
      if(!state.character.originSet){state.character.originX=cx;state.character.originY=cy;state.character.originSet=true;}
      part.transform={
        x:cx-n(state.character.originX,0),
        y:cy-n(state.character.originY,0),
        scale:1,scaleX:1,scaleY:1,rotation:0,opacity:1
      };
    }else{
      part.transform=normalisePartTransform(part.transform);
    }

    validateCharacterPartRegistration(
      part,
      clip
    );
    syncCharacterPartRigRefs(part);

    return clip;
  }

  function validateCharacterPartRegistration(part,clip){
    if(!part||!part.rig||!clip)return true;

    var vectorObjects=(clip.drawObjects||[]).filter(function(object){
      return object&&object.demBones&&Array.isArray(object.demBones.restPoints);
    });

    if(!vectorObjects.length||!(part.rig.bones||[]).length){
      return true;
    }

    var allRestPoints=[];

    vectorObjects.forEach(function(object){
      (object.demBones.restPoints||[]).forEach(function(point){
        allRestPoints.push({
          x:n(point.x,0),
          y:n(point.y,0)
        });
      });
    });

    if(!allRestPoints.length){
      return true;
    }

    var minX=Math.min.apply(null,allRestPoints.map(function(point){return point.x;}));
    var maxX=Math.max.apply(null,allRestPoints.map(function(point){return point.x;}));
    var minY=Math.min.apply(null,allRestPoints.map(function(point){return point.y;}));
    var maxY=Math.max.apply(null,allRestPoints.map(function(point){return point.y;}));

    var artCentre={
      x:(minX+maxX)/2,
      y:(minY+maxY)/2
    };

    var bonePoints=[];

    (part.rig.bones||[]).forEach(function(bone){
      bonePoints.push(bone.restHead,bone.restTail);
    });

    var boneMinX=Math.min.apply(null,bonePoints.map(function(point){return n(point.x,0);}));
    var boneMaxX=Math.max.apply(null,bonePoints.map(function(point){return n(point.x,0);}));
    var boneMinY=Math.min.apply(null,bonePoints.map(function(point){return n(point.y,0);}));
    var boneMaxY=Math.max.apply(null,bonePoints.map(function(point){return n(point.y,0);}));

    var boneCentre={
      x:(boneMinX+boneMaxX)/2,
      y:(boneMinY+boneMaxY)/2
    };

    var separation=Math.hypot(
      artCentre.x-boneCentre.x,
      artCentre.y-boneCentre.y
    );

    /*
      This is diagnostic only. Irregular rigs do not have to share an exact
      visual centre with the artwork, so only report very large divergence.
    */
    if(separation>Math.max(400,n(API.width,0)*.65)){
      console.warn(
        "DemBones Character registration looks suspicious:",
        {
          part:part.name,
          artworkCentre:artCentre,
          boneCentre:boneCentre,
          separation:separation
        }
      );
    }

    return true;
  }

  function makeCharacterPartInstance(part){
    var transform=normalisePartTransform(part&&part.transform);

    var instance=typeof API.createClipInstance==="function"
      ?API.createClipInstance(part.clipId,{
          name:part.name,
          x:transform.x,
          y:transform.y,
          scale:transform.scale,
          scaleX:transform.scaleX,
          scaleY:transform.scaleY,
          rotation:transform.rotation,
          opacity:transform.opacity
        })
      :{
          type:"clipinstance",
          clipId:part.clipId,
          name:part.name,
          x:transform.x,
          y:transform.y,
          scale:transform.scale,
          scaleX:transform.scaleX,
          scaleY:transform.scaleY,
          rotation:transform.rotation,
          timeOffset:0,
          opacity:transform.opacity,
          keyframes:[],
          id:uid("db-part-instance")
        };

    instance.__dbPartId=part.id;
    instance.__dbCharacterId=state.character.id;
    if(!Array.isArray(instance.keyframes))instance.keyframes=[];
    return instance;
  }

  function updateCharacterInstances(){
    /*
      Metadata-only reconciliation. This function must never rewrite native
      x/y/scale/rotation, sceneFrames, drawingLayers or object collections.
    */
    var root=state.character.rootClipId?clips()[state.character.rootClipId]:null;
    if(!root)return;

    var existing={};
    characterRootPartInstances(root).forEach(function(instance){
      existing[String(instance.__dbPartId)]=instance;
    });

    state.character.parts.forEach(function(part){
      var instance=existing[String(part.id)];
      if(!instance)return;
      instance.clipId=part.clipId;
      instance.name=part.name;
      instance.__dbPartId=part.id;
      instance.__dbCharacterId=state.character.id;
      if(!Array.isArray(instance.keyframes))instance.keyframes=[];
    });
  }

  function installCharacterPartInstance(part){
    if(!part||!part.clipId)return null;
    if(!returnToCharacterRoot())return null;

    var existing=partInstance(part);
    if(existing){
      updateCharacterInstances();
      return existing;
    }

    var instance=makeCharacterPartInstance(part);

    /*
      This is an explicit structural operation, so use Luke Animate's own
      addDrawObject path while the Character Clip is the active context.
    */
    API.addDrawObject(instance,"animate");

    part.transform=transformFromPartInstance(instance,part.transform);
    API.refreshUI();
    return instance;
  }

  function ensureAllCharacterPartInstances(){
    if(!state.character.parts.length)return false;
    if(!returnToCharacterRoot())return false;

    var changed=false;

    state.character.parts.forEach(function(part){
      if(!part||!part.clipId||!clips()[part.clipId])return;
      if(partInstance(part))return;

      var instance=makeCharacterPartInstance(part);
      API.addDrawObject(instance,"animate");
      part.transform=transformFromPartInstance(instance,part.transform);
      changed=true;
    });

    updateCharacterInstances();
    if(changed)API.refreshUI();
    return true;
  }

  function returnToCharacterRoot(){
    if(!state.character.parts.length)return false;

    var root=ensureCharacterRoot();
    var guard=0;
    var editing=typeof API.getEditingClipId==="function"
      ?API.getEditingClipId()
      :null;

    while(editing&&editing!==root.id&&guard<12){
      if(!API.exitClip())break;
      editing=API.getEditingClipId();
      guard++;
    }

    if(editing!==root.id){
      API.enterClip(root.id);
    }

    /* Navigation only. No Character child reconstruction is allowed here.
       Preserve any loose build rig before clearing the active editing rig so
       the Add Part picker can still offer a newly imported second/later part. */
    if(state.activeRig&&!characterPartForRigId(state.activeRig.id)){
      registerLooseBuildRig(state.activeRig,sourceObjectsForRig(state.activeRig),[]);
    }
    state.activeRig=null;
    state.selectedBoneId="";
    state.selectedCurveId="";

    return API.getEditingClipId()===root.id;
  }

    })[0]||partInstance(part);

    API.selectDrawObject(instance);
    API.refreshUI();

    return true;
  }

  function liveAdjustedPartTransform(part){
    if(!part)return normalisePartTransform(null);
    var instance=partInstance(part);
    if(!instance)return normalisePartTransform(part.transform);
    var source=instance;
    if(typeof API.getEffectiveObject==="function"){
      try{source=API.getEffectiveObject(instance,currentTime())||instance;}catch(ignoreLiveAdjusted){}
    }
    return transformFromPartInstance(source,part.transform);
  }

  function rewritePinForAdjustedPlacement(part,childTransform){
    if(!part||!part.pin)return false;

    var target=partById(part.pin.targetPartId);
    if(!target||!target.rig)return false;

    var bone=(target.rig.bones||[]).filter(function(candidate){
      return String(candidate.id)===String(part.pin.boneId);
    })[0];
    if(!bone)return false;

    var parentTransform=resolvedPartTransform(target,currentTime(),{});
    var boneMap=posedTransformMap(target.rig,currentTime());
    var pose=boneMap[bone.id];
    if(!pose)return false;

    var headWorld=partLocalPointToCharacterScene(parentTransform,pose.head);
    var tailWorld=partLocalPointToCharacterScene(parentTransform,pose.tail);
    var attachPoint=part.pin.attachPoint==="head"?"head":"tail";
    var anchor=attachPoint==="head"?headWorld:tailWorld;
    var boneAngle=Math.atan2(tailWorld.y-headWorld.y,tailWorld.x-headWorld.x)*180/Math.PI;
    var stageOffset={x:childTransform.x-anchor.x,y:childTransform.y-anchor.y};
    var unrotated=rotateVector(stageOffset.x,stageOffset.y,-boneAngle*Math.PI/180);
    var parentScale=Math.max(.0001,n(parentTransform.scale,1));

    part.pin.offsetLocal={
      x:unrotated.x/parentScale,
      y:unrotated.y/parentScale
    };
    part.pin.parentScaleAtPin=parentScale;
    part.pin.childScaleAtPin=Math.max(.01,n(childTransform.scale,1));
    part.pin.childScaleXAtPin=Math.max(.01,n(childTransform.scaleX,1));
    part.pin.childScaleYAtPin=Math.max(.01,n(childTransform.scaleY,1));

    if(part.pin.inheritRotation===false){
      part.pin.fixedRotation=n(childTransform.rotation,0);
    }else{
      part.pin.rotationOffset=n(childTransform.rotation,0)-boneAngle;
    }

    return true;
  }

  function syncAdjustedPartPlacement(){
    var part=partById(state.adjustingPartId);
    if(!part)return false;

    var transform=liveAdjustedPartTransform(part);
    part.transform=normalisePartTransform(transform);

    if(part.pin){
      rewritePinForAdjustedPlacement(part,part.transform);
    }

    return true;
  }

  function finishPartPlacementAdjust(silent){
    var part=partById(state.adjustingPartId);
    if(part)syncAdjustedPartPlacement();

    state.adjustingPartId=null;
    state.mode="select";

    if(state.adjustRestoreAutoKeyframe!==null){
      setAdjustAutoKeyframe(!!state.adjustRestoreAutoKeyframe);
    }
    state.adjustRestoreAutoKeyframe=null;

    API.refreshUI();

    if(!silent&&part){
      renderAll();
      notify('Adjusted "'+part.name+'" placement. Bones and pin move with the part.',false);
    }

    return !!part;
  }

  function beginPartPlacementAdjust(part){
    state.characterAllBonesEdit=false;
    if(!part||!part.clipId||!clips()[part.clipId]){
      notify("This character part has no editable Clip.",true);
      return false;
    }

    if(state.adjustingPartId&&Number(state.adjustingPartId)!==Number(part.id)){
      finishPartPlacementAdjust(true);
    }

    if(!selectCharacterPartOnStage(part)){
      notify("Could not select this character part on the stage.",true);
      return false;
    }

    var instance=partInstance(part);
    if(!instance){
      notify("This character part has no stage instance.",true);
      return false;
    }

    /*
      A pinned instance is normally evaluator-driven. Seed its real native
      transform with the currently solved placement before temporarily allowing
      the native transform gizmo to control it.
    */
    if(part.pin){
      var solved=resolvedPartTransform(part,currentTime(),{});
      instance.x=solved.x;
      instance.y=solved.y;
      instance.scale=solved.scale;
      instance.scaleX=solved.scaleX;
      instance.scaleY=solved.scaleY;
      instance.rotation=solved.rotation;
      instance.opacity=solved.opacity;
      part.transform=normalisePartTransform(solved);
    }

    state.adjustingPartId=part.id;
    state.character.selectedPartId=part.id;
    state.activeRig=null;
    state.selectedBoneId="";
    state.selectedCurveId="";
    state.tab="character";
    state.mode="adjust-part";

    if(state.adjustRestoreAutoKeyframe===null){
      try{state.adjustRestoreAutoKeyframe=typeof API.getAutoKeyframe==="function"?!!API.getAutoKeyframe():false;}catch(ignoreReadAutoKey){state.adjustRestoreAutoKeyframe=false;}
    }
    setAdjustAutoKeyframe(false);

    API.selectDrawObject(instance);
    API.refreshUI();
    renderAll();
    notify('Adjusting "'+part.name+'". Move, rotate or scale the whole part; no animation keys are added.',false);
    return true;
  }

  function syncCharacterSelectionFromStage(){
    if(
      !state.open||
      state.tab!=="character"||
      !state.character.rootClipId||
      API.getEditingClipId()!==state.character.rootClipId
    ){
      return false;
    }

    var selected=[];

    try{
      selected=API.getSelectedDrawObjects()||[];
    }catch(ignore){}

    var instance=selected.filter(function(object){
      return object&&object.__dbPartId!==undefined;
    })[0];

    if(!instance)return false;

    var part=partById(instance.__dbPartId);
    if(!part)return false;

    var changed=Number(state.character.selectedPartId)!==Number(part.id);
    state.character.selectedPartId=part.id;

    if(Number(state.adjustingPartId)===Number(part.id)){
      part.transform=liveAdjustedPartTransform(part);
    }else if(!part.pin){
      part.transform=transformFromPartInstance(instance,part.transform);
    }

    return changed;
  }

  function activateCharacterPartMode(part,mode){
    state.characterAllBonesEdit=false;
    if(!part||!part.clipId||!clips()[part.clipId]){
      notify("This character part has no editable Clip.",true);
      return;
    }

    if(state.adjustingPartId)finishPartPlacementAdjust(true);

    /*
      Character part editing and animation stay in the Character root Clip so
      every other part remains visible while the chosen rig is being edited.
      The selected part's own rig becomes active, but we do not enter the
      child part Clip here.
    */
    if(!selectCharacterPartOnStage(part)){
      notify("Could not select this character part on the stage.",true);
      return;
    }

    state.character.selectedPartId=part.id;
    state.activeRig=part.rig;
    syncCharacterPartRigRefs(part);

    if(state.activeRig){
      syncRigRefs(state.activeRig);
      state.selectedBoneId=
        state.activeRig.bones&&state.activeRig.bones.length
          ?state.activeRig.bones[0].id
          :"";
    }

    state.tab=mode==="animate"?"animate":"editbone";
    state.mode="select";
    state.showEnvelopes=mode!=="animate";

    API.refreshUI();
    renderAll();

    notify(
      mode==="animate"
        ?'Animating "'+part.name+'" while all other Character parts stay visible.'
        :'Editing bones for "'+part.name+'" while all other Character parts stay visible.',
      false
    );
  }

  function addBuildToCharacter(options){
    options=options&&options.__dbInternalOptions?options:{};

    var build=options.build||serializeBuildPart();

    if(!build){
      if(!options.silent)notify("Build or import a part first.",true);
      return null;
    }
    if(!state.character.parts.length&&!state.character.nameConfirmed){
      if(!options.silent){
        state.tab="character";renderAll();
        var nameField=state.panel&&state.panel.querySelector("[data-db-character-name]");
        if(nameField){nameField.focus();nameField.select();}
        notify("Name the character before adding the first part.",true);
      }
      return null;
    }

    var sourceRig=options.sourceRig||state.activeRig;
    var sourceRigId=sourceRig&&sourceRig.id!=null?String(sourceRig.id):"";
    var part=state.editingPartId
      ?partById(state.editingPartId)
      :null;

    if(!part&&sourceRigId){
      var existingSourcePart=sourceRigIdAlreadyInCharacter(sourceRigId);
      if(existingSourcePart){
        state.character.selectedPartId=existingSourcePart.id;
        if(!options.keepTab)state.tab="character";
        if(!options.skipStageSelect)selectCharacterPartOnStage(existingSourcePart);
        renderAll();
        if(!options.silent)notify('"'+existingSourcePart.name+'" is already in this Character.',true);
        return existingSourcePart;
      }
    }

    if(!part){
      var nextId=state.character.nextPartId++;

      part={
        id:nextId,
        name:(build&&build.name)||(sourceRig&&sourceRig.name)||(state.activeRig&&state.activeRig.name)||("Part "+nextId),
        clipId:"",
        source:null,
        rig:null,
        transform:null,
        pin:null,
        localOrigin:null,
        buildSourceIds:[],
        buildSourceRigId:"",
        buildSourceContextId:"",
        collapsed:false
      };

      state.character.parts.push(part);
    }else if(part.clipId&&clips()[part.clipId]){
      delete clips()[part.clipId];
    }

    /*
      Remember the temporary Build Parts objects. The Character Clip receives
      its own independent copy, so these loose source objects can be removed
      safely when the Character is finalised.
    */
    part.buildSourceIds=Array.isArray(options.sourceObjectIds)
      ?options.sourceObjectIds.slice()
      :(sourceRig
      ?sourceObjectsForRig(sourceRig).map(function(object){
          return object.id;
        })
      :[]);

    part.buildSourceRigId=sourceRig
      ?sourceRig.id
      :"";

    part.buildSourceContextId=options.sourceContextId!==undefined
      ?String(options.sourceContextId||"")
      :(typeof API.getEditingClipId==="function"?(API.getEditingClipId()||""):"");

    createPartClip(part,build);
    state.character.selectedPartId=part.id;
    state.editingPartId=null;

    ensureCharacterRoot();
    if(!options.keepTab)state.tab="character";

    var installedInstance=installCharacterPartInstance(part);
    finaliseCharacterOnStage(true);
    if(sourceRigId)unregisterLooseBuildRig(sourceRigId);
    installedInstance=partInstance(part)||installedInstance;

    renderAll();

    if(!options.skipStageSelect&&installedInstance){
      API.selectDrawObject(installedInstance);
      API.refreshUI();
    }

    if(!options.silent){
      notify(
        'Added "'+part.name+'" to MyCharacter at its current stage position. Use Select / Move, Adjust or Animate.',
        false
      );
    }

    return part;
  }

  function captureCharacterPartRuntime(part){
    if(!part||!part.rig||!part.clipId)return;

    var clip=clips()[part.clipId];
    if(!clip)return;

    var origin=part.localOrigin||{
      x:n(part.transform&&part.transform.x,0),
      y:n(part.transform&&part.transform.y,0)
    };

    var rig=clone(part.rig);

    shiftRig(rig,origin.x,origin.y);
    shiftRigPointKeys(rig,origin.x,origin.y);

    var objects=[];
    var expressionObjects=[];

    (clip.drawObjects||[]).forEach(function(raw){
      var object=clone(raw);

      /*
        v12 reverse conversion:
        shiftObject() restores visible geometry and restPoints together.
        Adding origin to restPoints again here would corrupt the saved/re-edited
        part by the same amount in the opposite direction.
      */
      shiftObject(object,origin.x,origin.y);

      if(object.demBones){
        delete object.demBones.rig;
      }

      if(object.demBonesExpression){
        delete object.demBonesExpression.rig;
        expressionObjects.push(object);
      }else{
        objects.push(object);
      }
    });

    part.source={
      app:"LukeAnimateDemBonesPart",
      version:3,
      name:part.name,
      rig:rig,
      objects:objects,
      expressionObjects:expressionObjects
    };
  }

  function editCharacterPart(part){
    if(!part||!part.source)return;

    captureCharacterPartRuntime(part);

    state.editingPartId=part.id;
    loadBuildPart(part.source,true);
    state.tab="source";
    renderAll();

    notify(
      'Editing "'+part.name+'" in Build Parts. Add To Character updates this part.',
      false
    );
  }

  function deleteCharacterPart(part){
    if(!part)return;
    if(state.adjustingPartId)finishPartPlacementAdjust(true);

    if(state.character.rootClipId){
      returnToCharacterRoot();

      var instance=partInstance(part);
      var objects=sceneObjects();
      var objectIndex=objects.indexOf(instance);
      if(objectIndex>=0)objects.splice(objectIndex,1);

      if(typeof API.getDrawingLayers==="function"){
        var drawingLayers=API.getDrawingLayers();
        for(var i=drawingLayers.length-1;i>=0;i--){
          var layer=drawingLayers[i];
          var owns=(layer.frames||[]).some(function(frame){
            return !!(frame&&frame.object&&Number(frame.object.__dbPartId)===Number(part.id));
          });
          if(owns)drawingLayers.splice(i,1);
        }
      }
    }

    if(part.clipId&&clips()[part.clipId])delete clips()[part.clipId];

    state.character.parts.forEach(function(candidate){
      if(candidate!==part&&candidate.pin&&Number(candidate.pin.targetPartId)===Number(part.id)){
        clearPartPin(candidate,true);
      }
    });

    state.character.parts=state.character.parts.filter(function(candidate){
      return candidate.id!==part.id;
    });

    if(state.character.selectedPartId===part.id){
      state.character.selectedPartId=state.character.parts[0]?state.character.parts[0].id:null;
    }

    if(Number(state.character.pinPickSourceId)===Number(part.id)){
      state.character.pinPickSourceId=null;
    }

    API.refreshUI();
    renderAll();
  }

  function movePart(part,dir){
    if(state.adjustingPartId)finishPartPlacementAdjust(true);
    var i=state.character.parts.indexOf(part);
    var j=i+dir;
    if(i<0||j<0||j>=state.character.parts.length)return;

    var temp=state.character.parts[i];
    state.character.parts[i]=state.character.parts[j];
    state.character.parts[j]=temp;

    if(returnToCharacterRoot()&&typeof API.getDrawingLayers==="function"){
      var model=API.getDrawingLayers();
      var byPart={};

      model.forEach(function(layer){
        (layer.frames||[]).some(function(frame){
          if(frame&&frame.object&&frame.object.__dbPartId!==undefined){
            byPart[String(frame.object.__dbPartId)]=layer;
            return true;
          }
          return false;
        });
      });

      var characterLayers=Object.keys(byPart).map(function(key){return byPart[key];});
      var otherLayers=model.filter(function(layer){
        return characterLayers.indexOf(layer)<0;
      });

      var ordered=state.character.parts.map(function(candidate){
        return byPart[String(candidate.id)];
      }).filter(Boolean);

      model.splice.apply(model,[0,model.length].concat(otherLayers,ordered));

      model.forEach(function(layer,index){
        layer.order=index+1;
        (layer.frames||[]).forEach(function(frame){
          if(frame&&frame.object)frame.object.timelineLayerOrder=layer.order;
        });
      });
    }

    API.refreshUI();
    renderAll();
  }

  function saveCharacter(){
    syncAllUnpinnedCharacterTransforms();

    state.character.parts.forEach(function(part){
      captureCharacterPartRuntime(part);
    });

    var data={
      app:"LukeAnimateMyCharacter",
      version:3,
      savedAt:new Date().toISOString(),
      character:{
        name:state.character.name,
        id:state.character.id,
        originX:n(state.character.originX,0),
        originY:n(state.character.originY,0),
        originSet:!!state.character.originSet,

  function pruneCharacterRootToPartInstances(){
    var root=state.character.rootClipId
      ?clips()[state.character.rootClipId]
      :null;

    if(!root)return;

    function keep(object){
      return !!(
        object&&
        object.type==="clipinstance"&&
        object.__dbPartId!==undefined&&
        object.__dbPartId!==null&&
        partById(object.__dbPartId)
      );
    }

    if(Array.isArray(root.drawObjects)){
      root.drawObjects=root.drawObjects.filter(keep);
    }

    if(Array.isArray(root.sceneFrames)){
      root.sceneFrames.forEach(function(frame){
        if(frame&&Array.isArray(frame.objects)){
          frame.objects=frame.objects.filter(keep);
        }
      });
    }

    if(Array.isArray(root.drawingLayers)){
      root.drawingLayers=root.drawingLayers.filter(function(layer){
        var hasRealPart=false;

        (layer.frames||[]).forEach(function(frame){
          if(frame&&frame.object&&keep(frame.object)){
            hasRealPart=true;
          }
        });

        return hasRealPart;
      });

      root.drawingLayers.forEach(function(layer,index){
        layer.order=index+1;

        (layer.frames||[]).forEach(function(frame){
          if(frame&&frame.object){
            frame.object.timelineLayerOrder=layer.order;
          }
        });
      });
    }
  }

  function finaliseCharacterOnStage(silent){
    if(state.adjustingPartId)finishPartPlacementAdjust(true);
    if(!state.character.parts.length){
      return false;
    }

    syncAllUnpinnedCharacterTransforms();
    syncAllCharacterRigRefs();
    ensureCharacterRoot();

    /*
      v16.8: before pruning anything, verify that every Character part has a
      real child Clip instance. This is especially important for the second
      and later parts when Luke Animate has already canonicalised the first
      part into drawingLayers.
    */
    ensureAllCharacterPartInstances();
    updateCharacterInstances();

    /*
      The Character root is an assembly container. It should contain only its
      child part Clips, never a loose Build Parts SVG.
    */
    pruneCharacterRootToPartInstances();

    exitToMainScene();

    /*
      Now we are safely in Main Scene, remove the temporary construction
      artwork/layers. The cloned vectors inside each part Clip are unaffected.
    */
    cleanupLooseBuildPartsFromMainScene();

    var root=clips()[state.character.rootClipId];
    var instance=ensureCharacterRootInstanceOnMainStage();

    if(root){
      root.name=state.character.name||"MyCharacter";
    }

    if(instance){
      instance.name=state.character.name||"MyCharacter";
      API.selectDrawObject(
        instance
      );
    }

    state.activeRig=null;
    state.selectedBoneId="";
    state.selectedCurveId="";
    state.editingPartId=null;
    state.character.pinPickSourceId=null;
    state.character.committed=true;
    state.character.showBones=true;
    state.tab="character";
    state.mode="select";

    API.refreshUI();

    if(!silent){
      notify(
        'MyCharacter is now the only DemBones artwork on the main stage. Its bones remain visible for animation.',
        false
      );
    }

    return true;
  }

  function pinWouldCreateCycle(source,target){
    if(!source||!target)return true;
    if(Number(source.id)===Number(target.id))return true;

    var cursor=target;
    var seen={};
    var guard=0;

    while(cursor&&guard<100){
      if(Number(cursor.id)===Number(source.id))return true;
      if(seen[String(cursor.id)])return true;
      seen[String(cursor.id)]=true;
      if(!cursor.pin||!cursor.pin.targetPartId)break;
      cursor=partById(cursor.pin.targetPartId);
      guard++;
    }

    return false;
  }

  function characterBoneRecordInCurrentScene(record,time){
    if(!record)return null;
    time=n(time,currentTime());

    var editing=typeof API.getEditingClipId==="function"
      ?API.getEditingClipId()
      :null;

    var head={x:n(record.head&&record.head.x,0),y:n(record.head&&record.head.y,0)};
    var tail={x:n(record.tail&&record.tail.x,0),y:n(record.tail&&record.tail.y,0)};

    /*
      characterPartBoneWorldRecords() returns Character-root scene coordinates.
      Convert those to Main Scene only when the Character root is not the active
      editing context. This keeps pin targeting registered with the bones that
      are actually drawn on screen.
    */
    if(!editing){
      head=characterRootPointToMainScene(head,time);
      tail=characterRootPointToMainScene(tail,time);
    }

    return{head:head,tail:tail};
  }

  function eventStagePoint(event){
    var rect=API.canvas.getBoundingClientRect();
    return{
      x:(n(event&&event.clientX,rect.left)-rect.left)*(API.width/Math.max(1,rect.width)),
      y:(n(event&&event.clientY,rect.top)-rect.top)*(API.height/Math.max(1,rect.height))
    };
  }

  function stagePixelsFromClient(px){
    var rect=API.canvas.getBoundingClientRect();
    var sx=API.width/Math.max(1,rect.width);
    var sy=API.height/Math.max(1,rect.height);
    return Math.max(1,n(px,1)*Math.max(sx,sy));
  }

  function completePartPin(source,target,bone,attachPoint){
    if(!source||!target||!bone)return false;

    if(pinWouldCreateCycle(source,target)){
      notify('That pin would create a circular Character dependency.',true);
      return false;
    }

    /*
      Preserve the child exactly where it is now. This matters when changing an
      existing pin: the native child Clip instance can contain the old stored
      transform while its visible placement is being supplied by the evaluator.
    */
    var childTransform=resolvedPartTransform(source,currentTime(),{});

    syncAllUnpinnedCharacterTransforms();

    var parentTransform=resolvedPartTransform(
      target,
      currentTime(),
      {}
    );

    var boneMap=posedTransformMap(
      target.rig,
      currentTime()
    );

    var pose=boneMap[bone.id];
    if(!pose)return false;

    var headWorld=partLocalPointToCharacterScene(parentTransform,pose.head);
    var tailWorld=partLocalPointToCharacterScene(parentTransform,pose.tail);

    attachPoint=attachPoint==="head"?"head":"tail";
    var anchor=attachPoint==="head"?headWorld:tailWorld;

    var boneAngle=Math.atan2(
      tailWorld.y-headWorld.y,
      tailWorld.x-headWorld.x
    )*180/Math.PI;

    var stageOffset={
      x:childTransform.x-anchor.x,
      y:childTransform.y-anchor.y
    };

    var unrotated=rotateVector(
      stageOffset.x,
      stageOffset.y,
      -boneAngle*Math.PI/180
    );

    var parentScale=Math.max(.0001,n(parentTransform.scale,1));

    source.transform=normalisePartTransform(childTransform);

    source.pin={
      version:2,
      targetPartId:target.id,
      boneId:bone.id,
      attachPoint:attachPoint,
      offsetLocal:{
        x:unrotated.x/parentScale,
        y:unrotated.y/parentScale
      },
      rotationOffset:childTransform.rotation-boneAngle,
      parentScaleAtPin:parentScale,
      childScaleAtPin:childTransform.scale,
      childScaleXAtPin:childTransform.scaleX,
      childScaleYAtPin:childTransform.scaleY,
      inheritRotation:true
    };

    state.character.pinPickSourceId=null;
    state.character.selectedPartId=source.id;
    state.mode="select";

    updateCharacterInstances();
    renderAll();
    API.refreshUI();

    notify(
      source.name+
      " pinned to "+
      target.name+
      " / "+
      bone.name+
      " without changing its current position.",
      false
    );

    return true;
  }

  function characterPinHit(event){
    var source=partById(state.character.pinPickSourceId);
    if(!source)return null;

    /* Compare in the same stage/display coordinate system used to draw targets. */
    var point=eventStagePoint(event);
    var nodeThreshold=stagePixelsFromClient(16);
    var lineThreshold=stagePixelsFromClient(11);
    var best=null;

    state.character.parts.forEach(function(target){
      if(Number(target.id)===Number(source.id)||!target.rig)return;
      if(pinWouldCreateCycle(source,target))return;

      characterPartBoneWorldRecords(target,currentTime()).forEach(function(record){
        var display=characterBoneRecordDisplayPoints(record,currentTime());
        if(!display)return;

        [["head",display.head],["tail",display.tail]].forEach(function(pair){
          var distance=Math.hypot(
            point.x-pair[1].x,
            point.y-pair[1].y
          );

          if(distance<=nodeThreshold&&(!best||distance<best.distance)){
            best={
              target:target,
              bone:record.bone,
              attachPoint:pair[0],
              distance:distance
            };
          }
        });

        var lineDistance=distancePointToSegment(
          point,
          display.head,
          display.tail
        );

        if(lineDistance<=lineThreshold&&(!best||lineDistance<best.distance)){
          var headDistance=Math.hypot(
            point.x-display.head.x,
            point.y-display.head.y
          );
          var tailDistance=Math.hypot(
            point.x-display.tail.x,
            point.y-display.tail.y
          );

          best={
            target:target,
            bone:record.bone,
            attachPoint:headDistance<tailDistance?"head":"tail",
            distance:lineDistance
          };
        }
      });
    });

    return best;
  }

        ctx.lineTo(tail.x,tail.y);
        ctx.stroke();

        ctx.beginPath();
        ctx.arc(head.x,head.y,7,0,Math.PI*2);
        ctx.fillStyle="#34b887";
        ctx.fill();
        ctx.strokeStyle="#153f34";
        ctx.lineWidth=2;

        );
      });
    });

  }

  function normaliseWeightList(list){
    var total=0;(list||[]).forEach(function(w){total+=Math.max(0,n(w.weight,0));});
    if(total<=.000001)return list||[];
    return(list||[]).map(function(w){return{boneId:w.boneId,weight:n(w.weight,0)/total};});
  }

  function blendWeightLists(a,b,ratio){
    var map={},t=clamp(n(ratio,.5),0,1);
    (a||[]).forEach(function(w){map[w.boneId]=(map[w.boneId]||0)+n(w.weight,0)*(1-t);});
    (b||[]).forEach(function(w){map[w.boneId]=(map[w.boneId]||0)+n(w.weight,0)*t;});
    return normaliseWeightList(Object.keys(map).map(function(id){return{boneId:id,weight:map[id]};}).filter(function(w){return w.weight>.00001;}));
  }

  function segmentProjection(point,a,b){
    var dx=b.x-a.x,dy=b.y-a.y,len2=dx*dx+dy*dy;
    var t=len2>.000001?clamp(((point.x-a.x)*dx+(point.y-a.y)*dy)/len2,0,1):0;
    var q={x:a.x+dx*t,y:a.y+dy*t};
    return{point:q,ratio:t,distance:Math.hypot(point.x-q.x,point.y-q.y)};
  }

  function closestCurveSegment(o,scenePoint){
    if(!o||!Array.isArray(o.points)||o.points.length<2)return null;
    var eff=typeof API.getEffectiveObject==="function"?API.getEffectiveObject(o,currentTime()):o;
    var shown=Array.isArray(eff.points)&&eff.points.length===o.points.length?eff.points:o.points;
    var count=o.closedPath?shown.length:shown.length-1,best=null;
    for(var i=0;i<count;i++){
      var j=(i+1)%shown.length,hit=segmentProjection(scenePoint,shown[i],shown[j]);
      if(!best||hit.distance<best.distance)best={startIndex:i,endIndex:j,insertIndex:i+1,ratio:hit.ratio,displayPoint:hit.point,distance:hit.distance};
    }
    return best;
  }

  function insertPointInCurveObject(o,segment,recordUndo){
    if(!o||!segment||!Array.isArray(o.points)||o.points.length<2)return false;
    if(recordUndo!==false)API.pushUndo();
    var oldLength=o.points.length,i=segment.startIndex,j=segment.endIndex,t=segment.ratio;
    var rawA=o.points[i],rawB=o.points[j],rawPoint={x:round3(rawA.x+(rawB.x-rawA.x)*t),y:round3(rawA.y+(rawB.y-rawA.y)*t)};
    var insertIndex=segment.insertIndex;
    if(o.closedPath&&i===oldLength-1)insertIndex=oldLength;
    o.points.splice(insertIndex,0,rawPoint);
    var d=o.demBones;
    if(d){
      d.restPoints=d.restPoints||clone(o.points);
      if(d.restPoints.length===oldLength){
        var ra=d.restPoints[i],rb=d.restPoints[j];
        d.restPoints.splice(insertIndex,0,{x:round3(ra.x+(rb.x-ra.x)*t),y:round3(ra.y+(rb.y-ra.y)*t)});
      }else d.restPoints=clone(o.points);
      if(Array.isArray(d.weights)&&d.weights.length===oldLength)d.weights.splice(insertIndex,0,blendWeightLists(d.weights[i],d.weights[j],t));
      else d.weights=[];
    }
    var rig=state.activeRig;
    if(rig&&rig.pointKeys&&Array.isArray(rig.pointKeys[o.id])){
      rig.pointKeys[o.id].forEach(function(key){
        if(!Array.isArray(key.points)||key.points.length!==oldLength)return;
        var a=key.points[i],b=key.points[j];
        key.points.splice(insertIndex,0,{x:round3(a.x+(b.x-a.x)*t),y:round3(a.y+(b.y-a.y)*t)});
      });
    }
    state.selectedPointIndex=insertIndex;
    if(d)rebuildObjectSkinData(o,state.activeRig);
    API.refreshUI();
    return true;
  }

  function loadSvgFile(file,asExpression){
    if(!file)return;

    var reader=new FileReader();

    reader.onload=function(){
      try{
        if(asExpression){
          /*
            Expression SVGs are intentionally NOT stage-centred. Their existing
            behaviour places them at the selected Lip_Bone / Frame_Bone anchor.
          */
          importExpressionSvg(
            reader.result,
            state.expressionType,
            state.expressionSlot||"Default"
          );
          return;
        }

        var objects=svgTextToObjects(
          reader.result
        );

        if(!objects.length){
          throw new Error(
            "No supported vector shapes were found."
          );
        }

        /*
          v10:
          Centre the actual imported artwork before the objects are registered
          with Luke Animate and before the rig rest points are captured.
        */
        var placement=centreImportedSvgObjectsOnStage(
          objects
        );

        API.pushUndo();

        var rig=createRigFromObjects(
          objects,
          file.name.replace(/\.[^.]+$/,"")||"SVG Part"
        );
        var staged=rig?lastCreatedRigObjects:objects;
        staged.forEach(function(o){
          API.addDrawObject(o,"animate");
        });

        state.tab="bones";
        state.mode="add-bone";

        renderAll();
        API.refreshUI();

        notify(
          "SVG loaded into Build Parts and centred on the stage.",
          false
        );

      }catch(e){
        notify(
          "SVG load failed: "+
          (e.message||e),
          true
        );
      }
    };

    reader.readAsText(file);
  }

  function loadSavedPartForCharacter(data){
    if(!data||data.app!=="LukeAnimateDemBonesPart"||!data.rig||!Array.isArray(data.objects)){
      throw new Error("Not a Dem Bones part file.");
    }

    /*
      Load Part in Build Character must create a loose Build Part on the main
      scene. If the Character root/part is currently being edited, temporarily
      leave nested Clip editing, load the saved part on Main Scene, then restore
      the Character root so all existing Character parts remain visible.
    */
    var originalEditingClipId=null;
    try{originalEditingClipId=API.getEditingClipId?API.getEditingClipId():null;}catch(ignoreReadEditingClip){}

    var guard=0;
    try{
      while(API.getEditingClipId&&API.getEditingClipId()&&guard<24){
        if(!API.exitClip())break;
        guard++;
      }
    }catch(ignoreExitToMain){}

    var loaded=loadBuildPart(data,true);
    if(!loaded||!loaded.rig)throw new Error("The saved part could not be loaded.");

    state.activeRig=loaded.rig;
    state.character.addCandidateRigId=String(loaded.rig.id);
    state.character.addPickerOpen=true;
    state.tab="character";
    state.mode="select";

    /* Existing Character parts should remain visible while choosing the newly
       loaded loose part. Restore the Character root when it already exists. */
    if(state.character.parts.length&&state.character.rootClipId&&clips()[state.character.rootClipId]){
      try{API.enterClip(state.character.rootClipId);}catch(ignoreRestoreCharacterRoot){}
    }else if(originalEditingClipId&&clips()[originalEditingClipId]&&!state.character.rootClipId){
      try{API.enterClip(originalEditingClipId);}catch(ignoreRestoreOriginalClip){}
    }

    renderAll();
    API.refreshUI();
    notify('Loaded "'+String(loaded.rig.name||data.name||"Part")+'". Choose Add Selected Part to add that exact part to the Character.',false);
    return loaded;
  }

  function drawBoneEnvelope(ctx,bone,q,selected){
    if(!bone||!q)return;

    var dx=q.tail.x-q.head.x;
    var dy=q.tail.y-q.head.y;
    var length=Math.max(.0001,Math.hypot(dx,dy));
    var perp={
      x:-dy/length,
      y:dx/length
    };

    var r0=Math.max(2,n(bone.envStart,30));
    var r1=Math.max(2,n(bone.envEnd,r0));

    var hPlus=stagePointFromScene({
      x:q.head.x+perp.x*r0,
      y:q.head.y+perp.y*r0
    });

    var hMinus=stagePointFromScene({
      x:q.head.x-perp.x*r0,
      y:q.head.y-perp.y*r0
    });

    var tPlus=stagePointFromScene({
      x:q.tail.x+perp.x*r1,
      y:q.tail.y+perp.y*r1
    });

    var tMinus=stagePointFromScene({
      x:q.tail.x-perp.x*r1,
      y:q.tail.y-perp.y*r1
    });

    var h=stagePointFromScene(q.head);
    var t=stagePointFromScene(q.tail);
    var pixelsPerScene=1/Math.max(.0001,scenePixels(1));

    ctx.save();

    /*
      The side quad plus circular end caps visually matches the actual
      point-to-segment envelope test closely enough to make influence obvious.
    */
    ctx.fillStyle=selected
      ?"rgba(14,156,134,.18)"
      :"rgba(80,116,150,.08)";

    ctx.strokeStyle=selected
      ?"rgba(14,156,134,.78)"
      :"rgba(82,103,124,.34)";

    ctx.lineWidth=selected?2:1;

    ctx.beginPath();
    ctx.moveTo(hPlus.x,hPlus.y);
    ctx.lineTo(tPlus.x,tPlus.y);
    ctx.lineTo(tMinus.x,tMinus.y);
    ctx.lineTo(hMinus.x,hMinus.y);
    ctx.closePath();
    ctx.fill();
    ctx.stroke();

    ctx.beginPath();
    ctx.arc(
      h.x,
      h.y,
      r0*pixelsPerScene,
      0,
      Math.PI*2
    );
    ctx.fill();
    ctx.stroke();

    ctx.beginPath();
    ctx.arc(
      t.x,
      t.y,
      r1*pixelsPerScene,
      0,
      Math.PI*2
    );
    ctx.fill();
    ctx.stroke();

    if(selected){
      ctx.fillStyle="#ffffff";
      ctx.strokeStyle="#0e9c86";
      ctx.lineWidth=2;
      [hPlus,hMinus,tPlus,tMinus].forEach(function(handle){ctx.beginPath();ctx.rect(handle.x-5,handle.y-5,10,10);ctx.fill();ctx.stroke();});
    }

    ctx.restore();
  }

  function drawBone(ctx,bone,q,selected){
    var h=stagePointFromScene(q.head);
    var t=stagePointFromScene(q.tail);

    ctx.save();

    ctx.lineWidth=selected?5:3;
    ctx.strokeStyle=selected?"#0e9c86":"#626a74";
    ctx.fillStyle=selected?"#0e9c86":"#ffffff";

    ctx.beginPath();
    ctx.moveTo(h.x,h.y);
    ctx.lineTo(t.x,t.y);
    ctx.stroke();

    ctx.beginPath();
    ctx.arc(h.x,h.y,selected?7:5,0,Math.PI*2);
    ctx.fill();
    ctx.stroke();

    ctx.beginPath();
    ctx.arc(t.x,t.y,selected?8:6,0,Math.PI*2);
    ctx.fill();
    ctx.stroke();

    ctx.fillStyle="#26303b";
    ctx.font="11px Arial";
    ctx.fillText(
      bone.name||"Bone",
      (h.x+t.x)/2+6,
      (h.y+t.y)/2-6
    );

    ctx.restore();
  }

  function overlayRenderer(ctx){
    /*
      v15:
      A finalised Character keeps its skeleton visible even with the DemBones
      panel closed. Those are Character-part bones, not the discarded Build
      Parts rig.
    */
    if(!state.open){
      if(state.character.committed&&state.character.showBones){
        drawCharacterAnimationBones(ctx);
      }
      if(state.activeRig){
        var closedMap=posedTransformMap(state.activeRig,currentTime());
        (state.activeRig.bones||[]).forEach(function(b){if(closedMap[b.id])drawBone(ctx,b,closedMap[b.id],String(b.id)===String(state.closedBoneId));});
      }
      return;
    }

    if(state.tab==="character"){
      drawCharacterAnimationBones(
        ctx
      );

      if(state.character.pinPickSourceId){
        drawCharacterPinTargets(
          ctx
        );
      }

      return;
    }

    if(state.characterAllBonesEdit&&state.tab==="editbone"){
      drawCharacterAllEditBones(ctx);
      return;
    }

    if(!state.activeRig)return;

    var map=state.tab==="animate"?posedTransformMap(
      state.activeRig,currentTime()
    ):restTransformMap(state.activeRig);

    /*
      Envelopes are painted first so bones and control points stay readable.
    */
    if(state.showEnvelopes){
      (state.activeRig.bones||[]).forEach(function(b){
        if(map[b.id]){
          drawBoneEnvelope(
            ctx,
            b,
            map[b.id],
            b.id===state.selectedBoneId
          );
        }
      });
    }

    (state.activeRig.bones||[]).forEach(function(b){
      if(map[b.id]){
        drawBone(
          ctx,
          b,
          map[b.id],
          b.id===state.selectedBoneId
        );
      }
    });

    if(state.addPreview){
      drawBone(
        ctx,
        {name:"New Bone"},
        {
          head:state.addPreview.head,
          tail:state.addPreview.tail
        },
        true
      );
    }

    if(state.tab==="curves"){
      var o=findObjectById(state.selectedCurveId)||
        (rigObjects(state.activeRig)[0]||null);

      if(o){
        var eff=typeof API.getEffectiveObject==="function"
          ?API.getEffectiveObject(o,currentTime())
          :o;

        var pts=Array.isArray(eff.points)
          ?eff.points
          :o.points||[];

        ctx.save();

        pts.forEach(function(p,i){
          var q=stagePointFromScene(p);

          ctx.beginPath();
          ctx.arc(
            q.x,
            q.y,
            i===state.selectedPointIndex?6:4,
            0,
            Math.PI*2
          );

          ctx.fillStyle=i===state.selectedPointIndex
            ?"#0e9c86"
            :"#ffffff";

          ctx.strokeStyle="#18212b";
          ctx.lineWidth=2;
          ctx.fill();
          ctx.stroke();
        });

        ctx.restore();
      }
    }
  }

    setInterval(function(){if(state.activeRig)syncRigDisplayGeometry(state.activeRig,currentTime());hideDemBonesTimelineRows();if(state.adjustingPartId){setAdjustAutoKeyframe(false);}if(state.open&&state.panel){var f=state.panel.querySelector('[data-db-frame]');if(f&&f.textContent!==String(currentFrame())){f.textContent=currentFrame();if(state.tab==='animate')renderBody();}if(state.tab==='character'){var changed=syncCharacterSelectionFromStage();if(changed)renderBody();}}},160);
  }

  function open(){
    /* Capture the stage selection before changing focus or suppressing the fixed
       inspector. This mirrors LukeTools, where the selected curve is already
       known when the floating tool opens. */
    selectedObjects(false);
    if(!state.panel)createPanel();
    state.open=true;state.panel.classList.add('open');
    API.setSelectionInspectorSuppressed(TOOL_ID,true);
    API.setConfiguredToolActive(GROUP_ID,ITEM_ID,true);
    API.bringFloatingPanelToFront(state.panel);API.clampFloatingPanelToViewport(state.panel);
    renderAll();notify('Dem Bones ready.',false);
  }
  function close(){
    if(state.adjustingPartId)finishPartPlacementAdjust(true);
    /*
      Closing DemBones after a Character exists is a final assembly handoff:
      leave one Character on Main Scene, discard loose Build Parts artwork,
      but keep the Character skeleton overlay available.
    */
    if(state.character.parts.length){
      finaliseCharacterOnStage(
        true
      );
    }

    state.open=false;
    state.characterAllBonesEdit=false;
    state.mode='select';
    state.curvePointAddMode=false;
    state.drag=null;
    state.addPreview=null;

    if(state.panel){
      state.panel.classList.remove('open');
    }

    API.setSelectionInspectorSuppressed(
      TOOL_ID,
      false
    );

    API.setConfiguredToolActive(
      GROUP_ID,
      ITEM_ID,
      false
    );
    updateStageCursor();

    API.refreshUI();
  }

  var plugin={version:TOOL_VERSION,open:open,close:close,get panel(){return state.panel;},get state(){return state;},refresh:function(){syncRigRefs(state.activeRig);renderAll();API.refreshUI();}};
  window[SINGLETON_KEY]=plugin;
  API.registerObjectEvaluator(EVALUATOR_ID,evaluator);
  if(typeof API.registerObjectRenderer==="function")API.registerObjectRenderer(RASTER_RENDERER_ID,rasterObjectRenderer);
  API.registerStageOverlayRenderer(OVERLAY_ID,overlayRenderer);
  bindGlobal();
  open();
})();
