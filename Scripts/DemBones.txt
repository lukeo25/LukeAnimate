  function ellipsePoints(o,count){
    var pts=[],cx=(n(o.x1,0)+n(o.x2,0))/2,cy=(n(o.y1,0)+n(o.y2,0))/2,rx=Math.abs(n(o.x2,0)-n(o.x1,0))/2,ry=Math.abs(n(o.y2,0)-n(o.y1,0))/2;
    for(var i=0;i<count;i++){var a=i/count*Math.PI*2;pts.push({x:cx+Math.cos(a)*rx,y:cy+Math.sin(a)*ry});} return pts;
  }

  function roundedRectPoints(o){
    var x1=Math.min(n(o.x1,0),n(o.x2,0)),y1=Math.min(n(o.y1,0),n(o.y2,0));
    var x2=Math.max(n(o.x1,0),n(o.x2,0)),y2=Math.max(n(o.y1,0),n(o.y2,0));
    var w=x2-x1,h=y2-y1,r=clamp(n(o.cornerRadius,0),0,Math.min(w,h)/2);
    if(r<=.001)return[{x:x1,y:y1},{x:x2,y:y1},{x:x2,y:y2},{x:x1,y:y2}];
    var segments=clamp(Math.ceil(r/4),4,18),pts=[];
    function arc(cx,cy,start,end,includeStart){
      for(var i=includeStart?0:1;i<=segments;i++){
        var a=start+(end-start)*(i/segments);
        pts.push({x:cx+Math.cos(a)*r,y:cy+Math.sin(a)*r});
      }
    }
    arc(x2-r,y1+r,-Math.PI/2,0,true);
    arc(x2-r,y2-r,0,Math.PI/2,false);
    arc(x1+r,y2-r,Math.PI/2,Math.PI,false);
    arc(x1+r,y1+r,Math.PI,Math.PI*1.5,false);
    return pts;
  }

  function bakeTransformIntoPoints(o,eff,pts){
    var base=objectBounds(o),px=(base.x1+base.x2)/2,py=(base.y1+base.y2)/2;
    var tx=n(eff&&eff.tx,n(o.tx,0)),ty=n(eff&&eff.ty,n(o.ty,0));
    var scale=n(eff&&eff.tscale,n(o.tscale,1));
    var angle=n(eff&&eff.trotation,n(o.trotation,0))*Math.PI/180;
    var c=Math.cos(angle),si=Math.sin(angle);
    return (pts||[]).map(function(p){
      var dx=(n(p.x,0)-px)*scale,dy=(n(p.y,0)-py)*scale;
      return{x:round3(px+tx+dx*c-dy*si),y:round3(py+ty+dx*si+dy*c)};
    });
  }

  function uniformIngestPoints(points,closedPath,maxStep){
    var source=(points||[]).map(function(p){return{x:n(p.x,0),y:n(p.y,0)};});
    if(source.length<2)return source;
    var out=[];
    var segmentCount=closedPath?source.length:source.length-1;
    var step=Math.max(2,n(maxStep,18));
    for(var i=0;i<segmentCount;i++){
      var a=source[i],b=source[(i+1)%source.length];
      out.push({x:a.x,y:a.y});
      var distance=Math.hypot(b.x-a.x,b.y-a.y);
      var divisions=Math.max(1,Math.ceil(distance/step));
      for(var j=1;j<divisions;j++){
        var t=j/divisions;
        out.push({x:round3(a.x+(b.x-a.x)*t),y:round3(a.y+(b.y-a.y)*t)});
      }
    }
    if(!closedPath)out.push({x:source[source.length-1].x,y:source[source.length-1].y});
    return out;
  }

  function limbIngestPoints(o){
    var b=objectBounds(o);
    var x0=b.x1,y0=b.y1,x1=b.x2,y1=b.y2;
    var w=Math.max(0,x1-x0),h=Math.max(0,y1-y0);
    if(w<=.001||h<=.001)return Array.isArray(o&&o.points)?clone(o.points):[];
    var vertical=h>=w;
    var jointFrac=.55;
    if(vertical){
      var gapY=Math.max(2,h*.035);
      var midY=y0+h*jointFrac;
      return[
        {x:x0,y:y0},{x:x1,y:y0},
        {x:x1,y:midY-gapY*.5},{x:x1,y:midY+gapY*.5},
        {x:x1,y:y1},{x:x0,y:y1},{x:x0,y:midY}
      ];
    }
    var gapX=Math.max(2,w*.035);
    var midX=x0+w*jointFrac;
    return[
      {x:x0,y:y0},{x:x0,y:y1},
      {x:midX-gapX*.5,y:y1},{x:midX+gapX*.5,y:y1},
      {x:x1,y:y1},{x:x1,y:y0},{x:midX,y:y0}
    ];
  }

  function applyIngestPointMode(o,mode){
    if(!o||!Array.isArray(o.points)||o.points.length<2)return o;
    mode=mode||"selected";
    if(mode==="uniform"){
      o.points=uniformIngestPoints(o.points,!!o.closedPath,18);
      o.demBonesPointMode="uniform";
    }else if(mode==="limb"){
      o.points=limbIngestPoints(o);
      o.closedPath=true;
      o.demBonesPointMode="limb";
      o.demBonesLimbPoints=7;
    }else{
      o.points=o.points.map(function(p){return{x:round3(p.x),y:round3(p.y)};});
      o.demBonesPointMode="selected";
    }
    return o;
  }

  function convertToWarpPath(o,pointMode){
    if(!supportedWarpObject(o)) return null;
    var originalType=o.type;
    ensureObjectId(o);

    /* Use Luke Animate's own permanent vector conversion first. This is the
       authoritative path for rounded rectangles, ellipses, current transforms
       and native editable points. It prevents Dem Bones from replacing the
       selected artwork with a crude bounding-box polygon. */
    try{
      if(typeof API.convertDrawObjectToEditablePath==="function"){
        var converted=API.convertDrawObjectToEditablePath(o);
        if(converted && Array.isArray(converted.points) && converted.points.length>1){
          converted.demBonesOriginalType=converted.demBonesOriginalType||originalType;
          converted.keyframes=[];
          ensureObjectId(converted);
          return applyIngestPointMode(converted,pointMode||"selected");
        }
      }
    }catch(eNative){}

    var eff=typeof API.getEffectiveObject==="function"?API.getEffectiveObject(o,currentTime()):o;
    var pts=[],closed=false,fill=false;
    if(Array.isArray((eff&&eff.points)||o.points)){
      pts=((eff&&eff.points)||o.points||[]).map(function(p){return{x:n(p.x,0),y:n(p.y,0)};});
      closed=eff&&eff.closedPath!==undefined?!!eff.closedPath:!!o.closedPath;
      fill=eff&&eff.fillEnabled!==undefined?!!eff.fillEnabled:!!o.fillEnabled;
    }else if(originalType==="line") pts=[{x:n(eff.x1,o.x1),y:n(eff.y1,o.y1)},{x:n(eff.x2,o.x2),y:n(eff.y2,o.y2)}];
    else if(originalType==="rect"){pts=roundedRectPoints({x1:n(eff.x1,o.x1),y1:n(eff.y1,o.y1),x2:n(eff.x2,o.x2),y2:n(eff.y2,o.y2),cornerRadius:n(eff.cornerRadius,o.cornerRadius)});closed=true;fill=eff.fillEnabled!==undefined?!!eff.fillEnabled:true;}
    else if(originalType==="ellipse"){pts=ellipsePoints({x1:n(eff.x1,o.x1),y1:n(eff.y1,o.y1),x2:n(eff.x2,o.x2),y2:n(eff.y2,o.y2)},64);closed=true;fill=eff.fillEnabled!==undefined?!!eff.fillEnabled:true;}
    if(pts.length<2) return null;
    pts=bakeTransformIntoPoints(o,eff,pts);
    o.demBonesOriginalType=o.demBonesOriginalType||originalType;
    if(originalType==="rect")o.demBonesOriginalCornerRadius=n(eff.cornerRadius,o.cornerRadius||0);
    o.type="freehand";o.points=pts;o.closedPath=closed;o.fillEnabled=fill;

  function distancePointToSegment(p,a,b){var dx=b.x-a.x,dy=b.y-a.y,l=dx*dx+dy*dy;if(l<.000001)return Math.hypot(p.x-a.x,p.y-a.y);var t=clamp(((p.x-a.x)*dx+(p.y-a.y)*dy)/l,0,1),x=a.x+dx*t,y=a.y+dy*t;return Math.hypot(p.x-x,p.y-y);}
  function angleOf(a,b){return Math.atan2(b.y-a.y,b.x-a.x);}
  function rotateVector(x,y,a){var c=Math.cos(a),s=Math.sin(a);return{x:x*c-y*s,y:x*s+y*c};}
  function inverseRotateVector(x,y,a){return rotateVector(x,y,-a);}
  function easingValue(v,e){var t=clamp(v,0,1);if(e==="ease-in")return t*t;if(e==="ease-out")return 1-Math.pow(1-t,2);if(e==="ease-in-out")return t<.5?2*t*t:1-Math.pow(-2*t+2,2)/2;if(e==="hold")return 0;return t;}

  function bindRig(recordUndo){
    var rig=state.activeRig;
    if(!rig){notify("Build or load a part first.",true);return;}
    if(!rig.bones.length){notify("Add at least one bone before binding.",true);return;}
    if(recordUndo!==false) API.pushUndo();
    rigObjects(rig).forEach(function(o){
      var d=o.demBones;
      d.restPoints=((o.points&&o.points.length)?o.points:(d.restPoints||[])).map(function(p){return{x:n(p.x,0),y:n(p.y,0)};});
      d.weights=d.restPoints.map(function(p){
        var ds=rig.bones.map(function(b){
          return{boneId:b.id,distance:distancePointToSegment(p,b.restHead,b.restTail)};
        }).sort(function(a,b){return a.distance-b.distance;}).slice(0,Math.min(3,rig.bones.length));
        var total=0;
        ds.forEach(function(x){x.weight=1/Math.max(1,x.distance*x.distance);total+=x.weight;});
        ds.forEach(function(x){x.weight=total?x.weight/total:1/ds.length;});
        return ds;
      });
    });
    syncRigRefs(rig);
    API.refreshUI();
    notify("Part bound to "+rig.bones.length+" bone"+(rig.bones.length===1?"":"s")+".",false);
  }

  function scenePixels(pixels){
    var a=API.stageToScenePoint({x:0,y:0});
    var b=API.stageToScenePoint({x:pixels,y:0});
    return Math.max(.001,Math.hypot(b.x-a.x,b.y-a.y));
  }

  function nearestBoneEndpoint(point,ignoreId){
    var rig=state.activeRig,best=null;
    if(!rig)return null;
    (rig.bones||[]).forEach(function(b){
      if(ignoreId&&String(b.id)===String(ignoreId))return;
      [["head",b.restHead],["tail",b.restTail]].forEach(function(entry){
        var d=Math.hypot(point.x-entry[1].x,point.y-entry[1].y);
        if(!best||d<best.distance)best={bone:b,type:entry[0],point:entry[1],distance:d};
      });
    });
    return best;
  }

  function autoConnectNewBone(a,b){
    var out={head:clone(a),tail:clone(b),parentId:"",connected:false};
    var rig=state.activeRig;
    if(!rig||!rig.bones.length)return out;
    var threshold=scenePixels(18);
    var headHit=nearestBoneEndpoint(a,null);
    var tailHit=nearestBoneEndpoint(b,null);
    var parent=null;
    if(headHit&&headHit.type==="tail"&&headHit.distance<=threshold){
      parent=headHit.bone;
      var dx=headHit.point.x-out.head.x,dy=headHit.point.y-out.head.y;
      out.head={x:headHit.point.x,y:headHit.point.y};
      out.tail={x:out.tail.x+dx,y:out.tail.y+dy};
    }else if(tailHit&&tailHit.type==="tail"&&tailHit.distance<=threshold){
      parent=tailHit.bone;
      out.head={x:tailHit.point.x,y:tailHit.point.y};
      out.tail=clone(a);
    }else{
      parent=selectedBone();
      if(parent){
        var sx=parent.restTail.x-out.head.x,sy=parent.restTail.y-out.head.y;
        out.head={x:parent.restTail.x,y:parent.restTail.y};
        out.tail={x:out.tail.x+sx,y:out.tail.y+sy};
      }
    }
    if(parent){out.parentId=parent.id;out.connected=true;}
    if(Math.hypot(out.tail.x-out.head.x,out.tail.y-out.head.y)<5){
      out.tail={x:out.head.x+20,y:out.head.y};
    }
    return out;
  }

  function enforceRestChains(){
    var rig=state.activeRig;
    if(!rig)return;
    var by={};
    (rig.bones||[]).forEach(function(b){by[b.id]=b;});
    var visiting={};
    function solve(parent){
      if(!parent||visiting[parent.id])return;
      visiting[parent.id]=true;
      (rig.bones||[]).forEach(function(child){
        if(!child||String(child.parentId)!==String(parent.id))return;
        var dx=parent.restTail.x-child.restHead.x,dy=parent.restTail.y-child.restHead.y;
        child.restHead={x:child.restHead.x+dx,y:child.restHead.y+dy};
        child.restTail={x:child.restTail.x+dx,y:child.restTail.y+dy};
        solve(child);
      });
      delete visiting[parent.id];
    }
    (rig.bones||[]).forEach(function(b){if(!b.parentId||!by[b.parentId])solve(b);});
  }

  function interpolatePose(bone,time){
    var base=bone.basePose||{dx:0,dy:0,rotation:0,scale:1,easing:"linear"},keys=(bone.keys||[]).slice().sort(function(a,b){return a.time-b.time;});
    function copyPose(p){return{dx:n(p.dx,0),dy:n(p.dy,0),rotation:n(p.rotation,0),scale:Math.max(.01,n(p.scale,1)),easing:p.easing||"linear"};}
    if(!keys.length)return copyPose(base);
    if(time<=keys[0].time){if(keys[0].time<=.0001)return copyPose(keys[0]);var r=easingValue(time/Math.max(.0001,keys[0].time),base.easing||"linear");return{dx:n(base.dx,0)+(n(keys[0].dx,0)-n(base.dx,0))*r,dy:n(base.dy,0)+(n(keys[0].dy,0)-n(base.dy,0))*r,rotation:n(base.rotation,0)+(n(keys[0].rotation,0)-n(base.rotation,0))*r,scale:n(base.scale,1)+(n(keys[0].scale,1)-n(base.scale,1))*r,easing:keys[0].easing||"linear"};}
    if(time>=keys[keys.length-1].time)return copyPose(keys[keys.length-1]);
    for(var i=0;i<keys.length-1;i++){var a=keys[i],b=keys[i+1];if(time>=a.time&&time<=b.time){var q=easingValue((time-a.time)/Math.max(.0001,b.time-a.time),a.easing||"linear");return{dx:n(a.dx,0)+(n(b.dx,0)-n(a.dx,0))*q,dy:n(a.dy,0)+(n(b.dy,0)-n(a.dy,0))*q,rotation:n(a.rotation,0)+(n(b.rotation,0)-n(a.rotation,0))*q,scale:n(a.scale,1)+(n(b.scale,1)-n(a.scale,1))*q,easing:a.easing||"linear"};}}
    return copyPose(base);
  }

  function restTransformMap(rig){var map={};(rig.bones||[]).forEach(function(b){var a=angleOf(b.restHead,b.restTail),l=Math.max(.001,Math.hypot(b.restTail.x-b.restHead.x,b.restTail.y-b.restHead.y));map[b.id]={head:{x:b.restHead.x,y:b.restHead.y},tail:{x:b.restTail.x,y:b.restTail.y},angle:a,length:l,scale:1};});return map;}
  function posedTransformMap(rig,time){
    var by={},rest=restTransformMap(rig),out={},resolving={};(rig.bones||[]).forEach(function(b){by[b.id]=b;});
    function resolve(b){if(!b)return null;if(out[b.id])return out[b.id];if(resolving[b.id])return rest[b.id];resolving[b.id]=true;var r=rest[b.id],p=interpolatePose(b,time),head,angle,parentScale=1;
      if(b.parentId&&by[b.parentId]){var pb=by[b.parentId],pr=rest[pb.id],pp=resolve(pb),lh=inverseRotateVector(r.head.x-pr.head.x,r.head.y-pr.head.y,pr.angle);lh.x*=pp.scale;lh.y*=pp.scale;var rh=rotateVector(lh.x,lh.y,pp.angle);head={x:pp.head.x+rh.x+n(p.dx,0),y:pp.head.y+rh.y+n(p.dy,0)};angle=pp.angle+(r.angle-pr.angle)+n(p.rotation,0)*Math.PI/180;parentScale=pp.scale;}else{head={x:r.head.x+n(p.dx,0),y:r.head.y+n(p.dy,0)};angle=r.angle+n(p.rotation,0)*Math.PI/180;}
      var scale=parentScale*Math.max(.01,n(p.scale,1)),off=rotateVector(r.length*scale,0,angle),q={head:head,tail:{x:head.x+off.x,y:head.y+off.y},angle:angle,length:r.length*scale,scale:scale};out[b.id]=q;delete resolving[b.id];return q;}
    (rig.bones||[]).forEach(resolve);return out;
  }

  function deformPoint(point,weights,rest,posed){if(!weights||!weights.length)return{x:point.x,y:point.y};var x=0,y=0,total=0;weights.forEach(function(w){var r=rest[w.boneId],p=posed[w.boneId];if(!r||!p)return;var local=inverseRotateVector(point.x-r.head.x,point.y-r.head.y,r.angle);local.x*=p.scale;local.y*=p.scale;var tr=rotateVector(local.x,local.y,p.angle),ww=n(w.weight,0);x+=(p.head.x+tr.x)*ww;y+=(p.head.y+tr.y)*ww;total+=ww;});return total?{x:x/total,y:y/total}:{x:point.x,y:point.y};}

  function interpolatePointKey(rig,objectId,time,basePoints){
    var keys=((rig.pointKeys||{})[objectId]||[]).slice().sort(function(a,b){return a.time-b.time;});if(!keys.length)return basePoints;
    if(time<=keys[0].time)return clone(keys[0].points);if(time>=keys[keys.length-1].time)return clone(keys[keys.length-1].points);
    for(var i=0;i<keys.length-1;i++){var a=keys[i],b=keys[i+1];if(time>=a.time&&time<=b.time){var r=easingValue((time-a.time)/Math.max(.0001,b.time-a.time),a.easing||"linear");return a.points.map(function(p,j){var q=b.points[j]||p;return{x:n(p.x,0)+(n(q.x,0)-n(p.x,0))*r,y:n(p.y,0)+(n(q.y,0)-n(p.y,0))*r};});}}
    return basePoints;
  }

  function evaluator(object,time,effective){
    if(object&&object.__dbPartId){
      var part=partById(object.__dbPartId);if(!part)return effective;var result=Object.assign({},effective),base=part.transform||{};result.x=n(base.x,0);result.y=n(base.y,0);result.scale=Math.max(.01,n(base.scale,1));result.rotation=n(base.rotation,0);result.opacity=n(base.opacity,1);
      if(part.pin&&part.pin.targetPartId&&part.pin.boneId){var target=partById(part.pin.targetPartId),ti=partInstance(target);if(target&&ti&&target.rig){var map=posedTransformMap(target.rig,time),bone=map[part.pin.boneId];if(bone){var tx=n(target.transform.x,0),ty=n(target.transform.y,0),ts=Math.max(.01,n(target.transform.scale,1)),tr=n(target.transform.rotation,0)*Math.PI/180;var loc=rotateVector(bone.tail.x*ts,bone.tail.y*ts,tr);result.x=tx+loc.x+n(part.pin.offsetX,0);result.y=ty+loc.y+n(part.pin.offsetY,0);if(part.pin.inheritRotation)result.rotation=n(target.transform.rotation,0)+bone.angle*180/Math.PI+n(part.pin.rotationOffset,0);}}
      }
      return result;
    }
    if(object&&object.demBonesExpression&&object.demBonesExpression.rig){var ex=object.demBonesExpression,slot=activeExpressionSlot(ex.rig,ex.type,time),r=Object.assign({},effective);r.opacity=slot===ex.slot?n(ex.baseOpacity,1):0;return r;}
    if(!object||!object.demBones||!object.demBones.rig)return effective;
    var data=object.demBones,rig=data.rig;if(!Array.isArray(data.restPoints)||!Array.isArray(data.weights)||!data.weights.length)return effective;
    var rest=restTransformMap(rig),posed=posedTransformMap(rig,time),base=interpolatePointKey(rig,object.id,time,data.restPoints),pts=base.map(function(p,i){return deformPoint(p,data.weights[i],rest,posed);});var out=Object.assign({},effective);out.points=pts;out.closedPath=!!data.closedPath;return out;
  }

  function normaliseWeightList(list){
    var total=0;(list||[]).forEach(function(w){total+=Math.max(0,n(w.weight,0));});
    if(total<=.000001)return list||[];
    return(list||[]).map(function(w){return{boneId:w.boneId,weight:n(w.weight,0)/total};});
  }

  function blendWeightLists(a,b,ratio){
    var map={},t=clamp(n(ratio,.5),0,1);
    (a||[]).forEach(function(w){map[w.boneId]=(map[w.boneId]||0)+n(w.weight,0)*(1-t);});
    (b||[]).forEach(function(w){map[w.boneId]=(map[w.boneId]||0)+n(w.weight,0)*t;});
    return normaliseWeightList(Object.keys(map).map(function(id){return{boneId:id,weight:map[id]};}).filter(function(w){return w.weight>.00001;}));
  }

  function segmentProjection(point,a,b){
    var dx=b.x-a.x,dy=b.y-a.y,len2=dx*dx+dy*dy;
    var t=len2>.000001?clamp(((point.x-a.x)*dx+(point.y-a.y)*dy)/len2,0,1):0;
    var q={x:a.x+dx*t,y:a.y+dy*t};
    return{point:q,ratio:t,distance:Math.hypot(point.x-q.x,point.y-q.y)};
  }

  function closestCurveSegment(o,scenePoint){
    if(!o||!Array.isArray(o.points)||o.points.length<2)return null;
    var eff=typeof API.getEffectiveObject==="function"?API.getEffectiveObject(o,currentTime()):o;
    var shown=Array.isArray(eff.points)&&eff.points.length===o.points.length?eff.points:o.points;
    var count=o.closedPath?shown.length:shown.length-1,best=null;
    for(var i=0;i<count;i++){
      var j=(i+1)%shown.length,hit=segmentProjection(scenePoint,shown[i],shown[j]);
      if(!best||hit.distance<best.distance)best={startIndex:i,endIndex:j,insertIndex:i+1,ratio:hit.ratio,displayPoint:hit.point,distance:hit.distance};
    }
    return best;
  }

  function insertPointInCurveObject(o,segment,recordUndo){
    if(!o||!segment||!Array.isArray(o.points)||o.points.length<2)return false;
    if(recordUndo!==false)API.pushUndo();
    var oldLength=o.points.length,i=segment.startIndex,j=segment.endIndex,t=segment.ratio;
    var rawA=o.points[i],rawB=o.points[j],rawPoint={x:round3(rawA.x+(rawB.x-rawA.x)*t),y:round3(rawA.y+(rawB.y-rawA.y)*t)};
    var insertIndex=segment.insertIndex;
    if(o.closedPath&&i===oldLength-1)insertIndex=oldLength;
    o.points.splice(insertIndex,0,rawPoint);
    var d=o.demBones;
    if(d){
      d.restPoints=d.restPoints||clone(o.points);
      if(d.restPoints.length===oldLength){
        var ra=d.restPoints[i],rb=d.restPoints[j];
        d.restPoints.splice(insertIndex,0,{x:round3(ra.x+(rb.x-ra.x)*t),y:round3(ra.y+(rb.y-ra.y)*t)});
      }else d.restPoints=clone(o.points);
      if(Array.isArray(d.weights)&&d.weights.length===oldLength)d.weights.splice(insertIndex,0,blendWeightLists(d.weights[i],d.weights[j],t));
      else d.weights=[];
    }
    var rig=state.activeRig;
    if(rig&&rig.pointKeys&&Array.isArray(rig.pointKeys[o.id])){
      rig.pointKeys[o.id].forEach(function(key){
        if(!Array.isArray(key.points)||key.points.length!==oldLength)return;
        var a=key.points[i],b=key.points[j];
        key.points.splice(insertIndex,0,{x:round3(a.x+(b.x-a.x)*t),y:round3(a.y+(b.y-a.y)*t)});
      });
    }
    state.selectedPointIndex=insertIndex;
    if(d&&(!d.weights||d.weights.length!==o.points.length))bindRig(false);
    API.refreshUI();
    return true;
  }

  function boneHit(event){
    var rig=state.activeRig;
    if(!rig)return null;
    var p=selectedScenePoint(event);
    var map=state.tab==="animate"?posedTransformMap(rig,currentTime()):restTransformMap(rig);
    var best=null,bestD=scenePixels(14),lineD=scenePixels(9);
    (rig.bones||[]).forEach(function(b){
      var q=map[b.id];
      if(!q)return;
      [["head",q.head],["tail",q.tail]].forEach(function(pair){
        var d=Math.hypot(p.x-pair[1].x,p.y-pair[1].y);
        if(d<bestD&&(!best||d<best.distance)){best={bone:b,handle:pair[0],point:p,distance:d};}
      });
      var ld=distancePointToSegment(p,q.head,q.tail);
      if(ld<lineD&&(!best||ld<best.distance-scenePixels(2))){best={bone:b,handle:"line",point:p,distance:ld};}
    });
    return best;
  }

  function onPointerDown(event){
    if(!state.open||!["bones","editbone","animate","curves","expression"].includes(state.tab))return;
    if(event.target&&event.target.closest&&event.target.closest(".la-dembones-full"))return;
    var point=selectedScenePoint(event);

    if(state.tab==="curves"){
      var curve=findObjectById(state.selectedCurveId);
      if(!curve&&state.activeRig){var curveObjects=rigObjects(state.activeRig);curve=curveObjects[0]||null;if(curve)state.selectedCurveId=curve.id;}
      if(!curve)return;
      var effCurve=typeof API.getEffectiveObject==="function"?API.getEffectiveObject(curve,currentTime()):curve;
      var shownPoints=Array.isArray(effCurve.points)?effCurve.points:curve.points;
      var pointThreshold=scenePixels(10),bestPoint=-1,bestDistance=Infinity;
      (shownPoints||[]).forEach(function(q,i){var d=Math.hypot(point.x-q.x,point.y-q.y);if(d<bestDistance){bestDistance=d;bestPoint=i;}});
      if(state.curvePointAddMode){
        var addSegment=closestCurveSegment(curve,point);
        if(addSegment&&addSegment.distance<=scenePixels(18)){
          insertPointInCurveObject(curve,addSegment,true);state.curvePointAddMode=false;state.mode="select";renderAll();notify('Curve point inserted without changing the outline.',false);
        }else notify('Click closer to a curve segment.',true);
        event.preventDefault();event.stopImmediatePropagation();return;
      }
      if(bestPoint>=0&&bestDistance<=pointThreshold){state.selectedPointIndex=bestPoint;state.curvePointAddMode=false;state.mode="select";renderAll();event.preventDefault();event.stopImmediatePropagation();return;}
      if(event.ctrlKey){
        var segment=closestCurveSegment(curve,point);
        if(segment&&segment.distance<=scenePixels(18)){
          insertPointInCurveObject(curve,segment,true);state.curvePointAddMode=false;state.mode="select";renderAll();notify('Curve point inserted without changing the outline.',false);
        }else notify('Click closer to a curve segment.',true);
        event.preventDefault();event.stopImmediatePropagation();return;
      }
      return;
    }

    var hit=boneHit(event);

    if(state.tab==="bones"){
      if(hit){
        event.preventDefault();event.stopImmediatePropagation();
        state.selectedBoneId=hit.bone.id;
        state.mode="add-bone";
        state.drag={kind:"maybe-add",pointerId:event.pointerId,start:point,current:point};
        renderAll();
        return;
      }
      if(state.mode!=="add-bone")return;
      event.preventDefault();event.stopImmediatePropagation();
      state.addPreview={head:point,tail:point};
      state.drag={kind:"add",pointerId:event.pointerId,start:point,current:point};
      return;
    }

    if(!hit)return;
    event.preventDefault();event.stopImmediatePropagation();
    state.selectedBoneId=hit.bone.id;

    if(state.tab==="editbone"){
      API.pushUndo();
      state.drag={
        kind:"edit-rest",
        pointerId:event.pointerId,
        bone:hit.bone,
        handle:hit.handle,
        startPointer:point,
        startHead:clone(hit.bone.restHead),
        startTail:clone(hit.bone.restTail)
      };
      renderAll();
      return;
    }

    if(state.tab==="animate"){
      API.pushUndo();
      var map=posedTransformMap(state.activeRig,currentTime());
      state.drag={
        kind:"animate-bone",
        pointerId:event.pointerId,
        bone:hit.bone,
        handle:hit.handle,
        startPointer:point,
        startPose:clone(interpolatePose(hit.bone,currentTime())),
        startTransform:clone(map[hit.bone.id]),
        key:ensureBoneKeyForDrag(hit.bone,currentTime())
      };
      renderAll();
      return;
    }

    renderAll();
  }

  function onPointerMove(event){
    if(!state.open||!state.drag)return;
    if(state.drag.pointerId!==undefined&&event.pointerId!==state.drag.pointerId)return;
    var p=selectedScenePoint(event);
    event.preventDefault();event.stopImmediatePropagation();

    if(state.drag.kind==="maybe-add"){
      state.drag.current=p;
      if(Math.hypot(p.x-state.drag.start.x,p.y-state.drag.start.y)>=scenePixels(8)){
        state.drag.kind="add";
        state.addPreview={head:clone(state.drag.start),tail:clone(p)};
      }
      API.refreshUI();
      return;
    }

    if(state.drag.kind==="add"){
      state.drag.current=p;
      if(state.addPreview)state.addPreview.tail=clone(p);
      API.refreshUI();
      return;
    }

    if(state.drag.kind==="edit-rest"){
      var b=state.drag.bone;
      var dx=p.x-state.drag.startPointer.x,dy=p.y-state.drag.startPointer.y;
      if(state.drag.handle==="tail"){
        b.restTail={x:round3(p.x),y:round3(p.y)};
      }else{
        b.restHead={x:round3(state.drag.startHead.x+dx),y:round3(state.drag.startHead.y+dy)};
        b.restTail={x:round3(state.drag.startTail.x+dx),y:round3(state.drag.startTail.y+dy)};
      }
      enforceRestChains();
      API.refreshUI();
      return;
    }

    if(state.drag.kind==="animate-bone"){
      var key=state.drag.key;
      var startPose=state.drag.startPose;
      if(state.drag.handle==="tail"){
        var desired=Math.atan2(p.y-state.drag.startTransform.head.y,p.x-state.drag.startTransform.head.x);
        var delta=(desired-state.drag.startTransform.angle)*180/Math.PI;
        while(delta>180)delta-=360;
        while(delta<-180)delta+=360;
        key.rotation=round3(n(startPose.rotation,0)+delta);
      }else{
        key.dx=round3(n(startPose.dx,0)+(p.x-state.drag.startPointer.x));
        key.dy=round3(n(startPose.dy,0)+(p.y-state.drag.startPointer.y));
      }
      API.refreshUI();
      return;
    }
  }

  function onPointerUp(event){
    if(!state.drag)return;
    if(state.drag.pointerId!==undefined&&event.pointerId!==state.drag.pointerId)return;
    event.preventDefault();event.stopImmediatePropagation();
    var finished=state.drag;
    if(finished.kind==="add"&&state.addPreview){
      var type=state.panel.querySelector("[data-db-bone-type]")?state.panel.querySelector("[data-db-bone-type]").value:"bone";
      addBone(state.addPreview.head,state.addPreview.tail,type);
    }else if(finished.kind==="edit-rest"){
      bindRig(false);
      notify("Rest bone edited. Child joints remain connected.",false);
    }else if(finished.kind==="animate-bone"){
      API.refreshUI();
      notify("Bone pose keyed on frame "+currentFrame()+".",false);
    }
    state.drag=null;
    state.addPreview=null;
    state.mode=state.tab==="bones"?"add-bone":"select";
    renderAll();
  }
