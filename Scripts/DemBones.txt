  function makePanelDraggable(panel, handle){
    if (!panel || !handle) return;
    handle.addEventListener("pointerdown", function(event){
      if (event.target.closest("button")) return;
      event.preventDefault();
      API.bringFloatingPanelToFront(panel);
      var rect = panel.getBoundingClientRect();
      var offsetX = event.clientX - rect.left;
      var offsetY = event.clientY - rect.top;
      function move(moveEvent){
        panel.style.left = (moveEvent.clientX - offsetX) + "px";
        panel.style.top = (moveEvent.clientY - offsetY) + "px";
        panel.style.right = "auto";
        panel.style.bottom = "auto";
      }
      function up(){
        window.removeEventListener("pointermove", move, true);
        window.removeEventListener("pointerup", up, true);
        API.clampFloatingPanelToViewport(panel);
      }
      window.addEventListener("pointermove", move, true);
      window.addEventListener("pointerup", up, true);
    });
  }

  function supportedObject(object){
    return object && ["freehand", "line", "rect", "ellipse"].indexOf(object.type) !== -1;
  }

  function ellipsePoints(object, count){
    var points = [];
    var cx = (n(object.x1, 0) + n(object.x2, 0)) / 2;
    var cy = (n(object.y1, 0) + n(object.y2, 0)) / 2;
    var rx = Math.abs(n(object.x2, 0) - n(object.x1, 0)) / 2;
    var ry = Math.abs(n(object.y2, 0) - n(object.y1, 0)) / 2;
    for (var i = 0; i < count; i += 1) {
      var angle = (i / count) * Math.PI * 2;
      points.push({ x: cx + Math.cos(angle) * rx, y: cy + Math.sin(angle) * ry });
    }
    return points;
  }

  function convertToWarpPath(object){
    if (!object || !supportedObject(object)) return null;
    var points = [];
    var closedPath = false;
    var fillEnabled = false;
    if (object.type === "freehand") {
      points = (object.points || []).map(function(point){ return { x:n(point.x,0), y:n(point.y,0) }; });
      closedPath = !!object.closedPath;
      fillEnabled = !!object.fillEnabled;
    } else if (object.type === "line") {
      points = [
        { x:n(object.x1,0), y:n(object.y1,0) },
        { x:n(object.x2,0), y:n(object.y2,0) }
      ];
    } else if (object.type === "rect") {
      var x1 = Math.min(n(object.x1,0), n(object.x2,0));
      var y1 = Math.min(n(object.y1,0), n(object.y2,0));
      var x2 = Math.max(n(object.x1,0), n(object.x2,0));
      var y2 = Math.max(n(object.y1,0), n(object.y2,0));
      points = [{x:x1,y:y1},{x:x2,y:y1},{x:x2,y:y2},{x:x1,y:y2}];
      closedPath = true;
      fillEnabled = true;
    } else if (object.type === "ellipse") {
      points = ellipsePoints(object, 40);
      closedPath = true;
      fillEnabled = true;
    }
    if (points.length < 2) return null;
    object.demBonesOriginalType = object.demBonesOriginalType || object.type;
    object.type = "freehand";
    object.points = points;
    object.closedPath = closedPath;
    object.fillEnabled = fillEnabled;
    if (object.strokeEnabled === undefined) object.strokeEnabled = true;
    if (object.strokeWidth === undefined) object.strokeWidth = 4;
    return object;
  }

  function rigTargets(rig){
    if (!rig) return [];
    var ids = {};
    (rig.targetIds || []).forEach(function(id){ ids[id] = true; });
    return sceneObjects().filter(function(object){
      return object && ids[object.id] && object.demBones && object.demBones.rigId === rig.id;
    });
  }

  function syncRigReferences(rig){
    if (!rig) return;
    rigTargets(rig).forEach(function(object){
      object.demBones.rig = rig;
    });
  }

  function findRigsInScene(){
    var rigs = {};
    sceneObjects().forEach(function(object){
      if (!object || !object.demBones || !object.demBones.rigId) return;
      var rig = object.demBones.rig;
      if (rig && rig.id) rigs[rig.id] = rigs[rig.id] || rig;
    });
    Object.keys(rigs).forEach(function(rigId){
      var rig = rigs[rigId];
      sceneObjects().forEach(function(object){
        if (object && object.demBones && object.demBones.rigId === rigId) object.demBones.rig = rig;
      });
    });
    return Object.keys(rigs).map(function(id){ return rigs[id]; });
  }

  function createRigFromSelection(){
    var chosen = selectedObjects().filter(supportedObject);
    if (!chosen.length) {
      notify("Select one or more line, rectangle, ellipse or freehand shapes first.", true);
      return;
    }
    API.pushUndo();
    var rig = {
      version: 1,
      id: uid("dem-bones-rig"),
      name: "Dem Bones Rig",
      sceneKey: sceneKey(),
      targetIds: [],
      bones: [],
      createdAt: new Date().toISOString()
    };
    chosen.forEach(function(object){
      convertToWarpPath(object);
      rig.targetIds.push(object.id);
      object.demBones = {
        version: 1,
        rigId: rig.id,
        rig: rig,
        restPoints: (object.points || []).map(function(point){ return {x:n(point.x,0),y:n(point.y,0)}; }),
        weights: [],
        closedPath: !!object.closedPath
      };
    });
    state.activeRig = rig;
    state.currentSceneKey = sceneKey();
    state.selectedBoneId = "";
    state.mode = "select";
    renderPanel();
    API.refreshUI();
    notify("Loaded " + chosen.length + " shape" + (chosen.length === 1 ? "" : "s") + ". Add bones, then Bind.", false);
  }

  function selectExistingRig(){
    var selected = selectedObjects();
    for (var i = 0; i < selected.length; i += 1) {
      if (selected[i].demBones && selected[i].demBones.rig) {
        state.activeRig = selected[i].demBones.rig;
        state.currentSceneKey = sceneKey();
        state.selectedBoneId = state.activeRig.bones.length ? state.activeRig.bones[0].id : "";
        syncRigReferences(state.activeRig);
        renderPanel();
        notify("Existing Dem Bones rig loaded.", false);
        return true;
      }
    }
    var rigs = findRigsInScene();
    if (rigs.length) {
      state.activeRig = rigs[0];
      state.currentSceneKey = sceneKey();
      state.selectedBoneId = state.activeRig.bones.length ? state.activeRig.bones[0].id : "";
      syncRigReferences(state.activeRig);
      renderPanel();
      notify("Existing Dem Bones rig found in this scene.", false);
      return true;
    }
    return false;
  }

  function distancePointToSegment(point, a, b){
    var dx = b.x - a.x;
    var dy = b.y - a.y;
    var lengthSq = dx * dx + dy * dy;
    if (lengthSq <= 0.000001) return Math.hypot(point.x - a.x, point.y - a.y);
    var t = ((point.x - a.x) * dx + (point.y - a.y) * dy) / lengthSq;
    t = clamp(t, 0, 1);
    var px = a.x + dx * t;
    var py = a.y + dy * t;
    return Math.hypot(point.x - px, point.y - py);
  }

  function bindRig(recordUndo){
    if (recordUndo === undefined) recordUndo = true;
    var rig = state.activeRig;
    if (!rig) {
      notify("Use Selection or load an existing rig first.", true);
      return;
    }
    if (!rig.bones.length) {
      notify("Add at least one bone before binding.", true);
      return;
    }
    if (recordUndo) API.pushUndo();
    var targets = rigTargets(rig);
    targets.forEach(function(object){
      var data = object.demBones;
      data.restPoints = (object.points || data.restPoints || []).map(function(point){ return {x:n(point.x,0),y:n(point.y,0)}; });
      data.weights = data.restPoints.map(function(point){
        var distances = rig.bones.map(function(bone){
          return { boneId:bone.id, distance:distancePointToSegment(point, bone.restHead, bone.restTail) };
        }).sort(function(a,b){ return a.distance - b.distance; }).slice(0, Math.min(2, rig.bones.length));
        var total = 0;
        distances.forEach(function(entry){
          entry.weight = 1 / Math.max(1, entry.distance * entry.distance);
          total += entry.weight;
        });
        distances.forEach(function(entry){ entry.weight = total > 0 ? entry.weight / total : 1 / distances.length; });
        return distances;
      });
    });
    syncRigReferences(rig);
    API.refreshUI();
    notify("Bound " + targets.length + " shape" + (targets.length === 1 ? "" : "s") + " to " + rig.bones.length + " bone" + (rig.bones.length === 1 ? "" : "s") + ".", false);
  }

  function angleOf(a, b){
    return Math.atan2(b.y - a.y, b.x - a.x);
  }

  function rotateVector(x, y, angle){
    var c = Math.cos(angle);
    var s = Math.sin(angle);
    return { x:x*c - y*s, y:x*s + y*c };
  }

  function inverseRotateVector(x, y, angle){
    return rotateVector(x, y, -angle);
  }

  function easingValue(value, easing){
    var t = clamp(value, 0, 1);
    if (easing === "ease-in") return t * t;
    if (easing === "ease-out") return 1 - Math.pow(1 - t, 2);
    if (easing === "ease-in-out") return t < 0.5 ? 2*t*t : 1 - Math.pow(-2*t+2, 2)/2;
    if (easing === "hold") return 0;
    return t;
  }

  function interpolatePose(bone, time){
    var base = bone.basePose || {dx:0,dy:0,rotation:0,easing:"linear"};
    var keys = (bone.keys || []).slice().sort(function(a,b){ return a.time - b.time; });
    if (!keys.length) return {dx:n(base.dx,0),dy:n(base.dy,0),rotation:n(base.rotation,0),easing:base.easing || "linear"};
    if (time <= keys[0].time) {
      if (keys[0].time <= 0.0001) return clone(keys[0]);
      var firstRatio = easingValue(time / Math.max(0.0001, keys[0].time), base.easing || "linear");
      return {
        dx:n(base.dx,0) + (n(keys[0].dx,0)-n(base.dx,0))*firstRatio,
        dy:n(base.dy,0) + (n(keys[0].dy,0)-n(base.dy,0))*firstRatio,
        rotation:n(base.rotation,0) + (n(keys[0].rotation,0)-n(base.rotation,0))*firstRatio,
        easing:keys[0].easing || "linear"
      };
    }
    if (time >= keys[keys.length-1].time) return clone(keys[keys.length-1]);
    for (var i = 0; i < keys.length - 1; i += 1) {
      var a = keys[i];
      var b = keys[i+1];
      if (time >= a.time && time <= b.time) {
        var ratio = easingValue((time-a.time)/Math.max(0.0001,b.time-a.time), a.easing || "linear");
        return {
          dx:n(a.dx,0)+(n(b.dx,0)-n(a.dx,0))*ratio,
          dy:n(a.dy,0)+(n(b.dy,0)-n(a.dy,0))*ratio,
          rotation:n(a.rotation,0)+(n(b.rotation,0)-n(a.rotation,0))*ratio,
          easing:a.easing || "linear"
        };
      }
    }
    return clone(base);
  }

  function restTransformMap(rig){
    var map = {};
    function resolve(bone){
      if (!bone) return null;
      if (map[bone.id]) return map[bone.id];
      var angle = angleOf(bone.restHead, bone.restTail);
      var length = Math.max(0.001, Math.hypot(bone.restTail.x-bone.restHead.x,bone.restTail.y-bone.restHead.y));
      var result = {head:{x:bone.restHead.x,y:bone.restHead.y},tail:{x:bone.restTail.x,y:bone.restTail.y},angle:angle,length:length};
      map[bone.id] = result;
      return result;
    }
    rig.bones.forEach(resolve);
    return map;
  }

  function posedTransformMap(rig, time){
    var byId = {};
    rig.bones.forEach(function(bone){ byId[bone.id] = bone; });
    var restMap = restTransformMap(rig);
    var posed = {};
    var resolving = {};
    function resolve(bone){
      if (!bone) return null;
      if (posed[bone.id]) return posed[bone.id];
      if (resolving[bone.id]) return restMap[bone.id];
      resolving[bone.id] = true;
      var rest = restMap[bone.id];
      var pose = interpolatePose(bone, time);
      var head;
      var angle;
      if (bone.parentId && byId[bone.parentId]) {
        var parent = byId[bone.parentId];
        var parentRest = restMap[parent.id];
        var parentPose = resolve(parent);
        var localHead = inverseRotateVector(rest.head.x-parentRest.head.x, rest.head.y-parentRest.head.y, parentRest.angle);
        var rotatedHead = rotateVector(localHead.x, localHead.y, parentPose.angle);
        head = {x:parentPose.head.x+rotatedHead.x+n(pose.dx,0),y:parentPose.head.y+rotatedHead.y+n(pose.dy,0)};
        angle = parentPose.angle + (rest.angle-parentRest.angle) + n(pose.rotation,0)*Math.PI/180;
      } else {
        head = {x:rest.head.x+n(pose.dx,0),y:rest.head.y+n(pose.dy,0)};
        angle = rest.angle + n(pose.rotation,0)*Math.PI/180;
      }
      var tailOffset = rotateVector(rest.length, 0, angle);
      var result = {head:head,tail:{x:head.x+tailOffset.x,y:head.y+tailOffset.y},angle:angle,length:rest.length};
      posed[bone.id] = result;
      delete resolving[bone.id];
      return result;
    }
    rig.bones.forEach(resolve);
    return posed;
  }

  function deformPoint(point, weights, restMap, posedMap){
    if (!weights || !weights.length) return {x:point.x,y:point.y};
    var x = 0;
    var y = 0;
    var total = 0;
    weights.forEach(function(weight){
      var rest = restMap[weight.boneId];
      var posed = posedMap[weight.boneId];
      if (!rest || !posed) return;
      var local = inverseRotateVector(point.x-rest.head.x, point.y-rest.head.y, rest.angle);
      var transformed = rotateVector(local.x, local.y, posed.angle);
      var w = n(weight.weight, 0);
      x += (posed.head.x + transformed.x) * w;
      y += (posed.head.y + transformed.y) * w;
      total += w;
    });
    if (total <= 0.0001) return {x:point.x,y:point.y};
    return {x:x/total,y:y/total};
  }

  function evaluator(object, time, effective){
    if (!object || !object.demBones || !object.demBones.rig) return effective;
    var data = object.demBones;
    var rig = data.rig;
    if (!rig || !Array.isArray(data.restPoints) || !Array.isArray(data.weights) || !data.weights.length) return effective;
    var restMap = restTransformMap(rig);
    var posedMap = posedTransformMap(rig, time);
    var points = data.restPoints.map(function(point,index){
      return deformPoint(point, data.weights[index], restMap, posedMap);
    });
    var result = Object.assign({}, effective);
    result.points = points;
    result.closedPath = !!data.closedPath;
    return result;
  }

  function addBone(head, tail){
    var rig = state.activeRig;
    if (!rig) {
      notify("Use Selection before adding bones.", true);
      return;
    }
    var length = Math.hypot(tail.x-head.x,tail.y-head.y);
    if (length < 4) {
      notify("Drag a longer bone.", true);
      return;
    }
    API.pushUndo();
    var parent = selectedBone();
    var bone = {
      id:uid("bone"),
      name:"Bone " + (rig.bones.length+1),
      parentId:parent ? parent.id : "",
      restHead:{x:round3(head.x),y:round3(head.y)},
      restTail:{x:round3(tail.x),y:round3(tail.y)},
      basePose:{dx:0,dy:0,rotation:0,easing:"linear"},
      keys:[]
    };
    rig.bones.push(bone);
    state.selectedBoneId = bone.id;
    state.mode = "select";
    bindRig(false);
    renderPanel();
    notify(bone.name + " added.", false);
  }

  function removeSelectedBone(){
    var rig = state.activeRig;
    var bone = selectedBone();
    if (!rig || !bone) return;
    API.pushUndo();
    rig.bones = rig.bones.filter(function(entry){ return entry.id !== bone.id; });
    rig.bones.forEach(function(entry){ if (entry.parentId === bone.id) entry.parentId = bone.parentId || ""; });
    state.selectedBoneId = rig.bones.length ? rig.bones[0].id : "";
    if (rig.bones.length) bindRig();
    else rigTargets(rig).forEach(function(object){ object.demBones.weights = []; });
    renderPanel();
    API.refreshUI();
    notify("Bone deleted.", false);
  }

  function renameSelectedBone(){
    var bone = selectedBone();
    if (!bone || !state.panel) return;
    var input = state.panel.querySelector("[data-db-bone-name]");
    var name = input ? input.value.trim() : "";
    if (!name) return;
    bone.name = name;
    renderPanel();
    API.refreshUI();
  }

  function exactKey(bone, time){
    if (!bone) return null;
    var tolerance = 0.5 / Math.max(1, n(API.fps, 30));
    for (var i = 0; i < (bone.keys || []).length; i += 1) {
      if (Math.abs(n(bone.keys[i].time,0)-time) <= tolerance) return bone.keys[i];
    }
    return null;
  }

  function poseForUi(bone){
    if (!bone) return {dx:0,dy:0,rotation:0,easing:"linear"};
    var key = exactKey(bone,currentTime());
    return key || interpolatePose(bone,currentTime());
  }

  function readPoseControls(){
    if (!state.panel) return {dx:0,dy:0,rotation:0,easing:"linear"};
    return {
      dx:n(state.panel.querySelector("[data-db-dx-num]").value,0),
      dy:n(state.panel.querySelector("[data-db-dy-num]").value,0),
      rotation:n(state.panel.querySelector("[data-db-rotation-num]").value,0),
      easing:state.panel.querySelector("[data-db-easing]").value || "linear"
    };
  }

  function writePoseControls(pose){
    if (!state.panel) return;
    pose = pose || {dx:0,dy:0,rotation:0,easing:"linear"};
    [["[data-db-dx]","[data-db-dx-num]",pose.dx],["[data-db-dy]","[data-db-dy-num]",pose.dy],["[data-db-rotation]","[data-db-rotation-num]",pose.rotation]].forEach(function(entry){
      var range = state.panel.querySelector(entry[0]);
      var number = state.panel.querySelector(entry[1]);
      if (range) range.value = String(round3(entry[2]));
      if (number) number.value = String(round3(entry[2]));
    });
    var easing = state.panel.querySelector("[data-db-easing]");
    if (easing) easing.value = pose.easing || "linear";
  }

  function setKeyFromControls(){
    var bone = selectedBone();
    if (!bone) {
      notify("Select a bone first.", true);
      return;
    }
    API.pushUndo();
    var pose = readPoseControls();
    var time = currentTime();
    var key = exactKey(bone,time);
    if (key) Object.assign(key,pose,{time:time});
    else bone.keys.push(Object.assign({time:time},pose));
    bone.keys.sort(function(a,b){return a.time-b.time;});
    API.refreshUI();
    renderPanel();
    notify("Key set for " + bone.name + " on frame " + currentFrame() + ".", false);
  }

  function livePoseChanged(){
    var bone = selectedBone();
    if (!bone) return;
    var pose = readPoseControls();
    var time = currentTime();
    var key = exactKey(bone,time);
    if (key) Object.assign(key,pose);
    else if (time <= 0.0001 && !(bone.keys || []).length) bone.basePose = Object.assign({},pose);
    API.refreshUI();
  }

  function deleteCurrentKey(){
    var bone = selectedBone();
    if (!bone) return;
    var before = bone.keys.length;
    var time = currentTime();
    var tolerance = 0.5 / Math.max(1,n(API.fps,30));
    bone.keys = bone.keys.filter(function(key){ return Math.abs(n(key.time,0)-time) > tolerance; });
    if (bone.keys.length !== before) {
      API.pushUndo();
      renderPanel();
      API.refreshUI();
      notify("Current key deleted.", false);
    } else notify("There is no key on this frame.", true);
  }

  function resetPose(){
    var bone = selectedBone();
    if (!bone) return;
    writePoseControls({dx:0,dy:0,rotation:0,easing:"linear"});
    livePoseChanged();
  }

  function clearRig(){
    var rig = state.activeRig;
    if (!rig) return;
    API.pushUndo();
    rigTargets(rig).forEach(function(object){
      if (object.demBones && Array.isArray(object.demBones.restPoints)) object.points = clone(object.demBones.restPoints);
      delete object.demBones;
    });
    state.activeRig = null;
    state.selectedBoneId = "";
    state.mode = "select";
    renderPanel();
    API.refreshUI();
    notify("Dem Bones rig removed. The original rest geometry was restored.", false);
  }

  function exportRig(){
    var rig = state.activeRig;
    if (!rig) {
      notify("There is no active rig to export.", true);
      return;
    }
    var targets = rigTargets(rig).map(function(object){
      return {
        objectId:object.id,
        objectName:object.name || "",
        restPoints:clone(object.demBones.restPoints || []),
        weights:clone(object.demBones.weights || []),
        closedPath:!!object.demBones.closedPath
      };
    });
    var data = {format:"LukeAnimateDemBones",version:1,rig:clone(rig),targets:targets};
    var blob = new Blob([JSON.stringify(data,null,2)],{type:"application/json"});
    var url = URL.createObjectURL(blob);
    var link = document.createElement("a");
    link.href = url;
    link.download = (rig.name || "DemBonesRig").replace(/[^a-z0-9_-]+/gi,"_") + ".dembones.json";
    document.body.appendChild(link);
    link.click();
    link.remove();
    setTimeout(function(){URL.revokeObjectURL(url);},1000);
    notify("Exported Dem Bones rig JSON.", false);
  }

  function importRigFile(file){
    if (!file) return;
    var reader = new FileReader();
    reader.onload = function(){
      try {
        var data = JSON.parse(String(reader.result || ""));
        if (!data || data.format !== "LukeAnimateDemBones" || !data.rig) throw new Error("Not a Luke Animate Dem Bones file.");
        var rig = data.rig;
        rig.id = rig.id || uid("dem-bones-rig");
        rig.sceneKey = sceneKey();
        var objects = sceneObjects();
        var selected = selectedObjects().filter(supportedObject);
        var targetRecords = Array.isArray(data.targets) ? data.targets : [];
        var resolved = [];
        targetRecords.forEach(function(record,index){
          var object = objects.find(function(entry){ return entry.id === record.objectId; }) || selected[index];
          if (!object) return;
          convertToWarpPath(object);
          object.demBones = {
            version:1,
            rigId:rig.id,
            rig:rig,
            restPoints:clone(record.restPoints || object.points || []),
            weights:clone(record.weights || []),
            closedPath:!!record.closedPath
          };
          object.points = clone(object.demBones.restPoints);
          resolved.push(object);
        });
        if (!resolved.length) throw new Error("No matching shapes were found. Select the target shapes before importing.");
        rig.targetIds = resolved.map(function(object){return object.id;});
        state.activeRig = rig;
        state.currentSceneKey = sceneKey();
        state.selectedBoneId = rig.bones && rig.bones.length ? rig.bones[0].id : "";
        syncRigReferences(rig);
        renderPanel();
        API.refreshUI();
        notify("Imported Dem Bones rig for " + resolved.length + " shape" + (resolved.length===1?"":"s") + ".", false);
      } catch (error) {
        notify("Import failed: " + (error.message || error), true);
      }
    };
    reader.readAsText(file);
  }

  function renderPanel(){
    if (!state.panel) return;
    var frame = state.panel.querySelector("[data-db-frame]");

    var rig = state.activeRig;
    var info = state.panel.querySelector("[data-db-target-info]");

      if (!rig) info.textContent = "Select one or more shapes, then click Use Selection. Rectangles and ellipses are converted to editable vector paths so their points can bend.";
      else info.textContent = rigTargets(rig).length + " target shape" + (rigTargets(rig).length===1?"":"s") + " · " + rig.bones.length + " bone" + (rig.bones.length===1?"":"s") + " · " + (API.getEditingClipId() ? "inside Clip" : "main scene");
    }

  function toScenePoint(event){
    var rect = API.canvas.getBoundingClientRect();
    var stagePoint = {
      x:(event.clientX-rect.left) * API.width / Math.max(1,rect.width),
      y:(event.clientY-rect.top) * API.height / Math.max(1,rect.height)
    };
    return API.stageToScenePoint(stagePoint);
  }

  function stageDistance(eventPoint, scenePoint){
    var stagePoint = API.sceneToStagePoint(scenePoint);
    return Math.hypot(eventPoint.x-stagePoint.x,eventPoint.y-stagePoint.y);
  }

  function eventStagePoint(event){
    var rect = API.canvas.getBoundingClientRect();
    return {
      x:(event.clientX-rect.left) * API.width / Math.max(1,rect.width),
      y:(event.clientY-rect.top) * API.height / Math.max(1,rect.height)
    };
  }

  function closestBoneHit(event){
    var rig = state.activeRig;
    if (!rig) return null;
    var stagePoint = eventStagePoint(event);
    var best = null;
    rig.bones.forEach(function(bone){
      var headDistance = stageDistance(stagePoint,bone.restHead);
      var tailDistance = stageDistance(stagePoint,bone.restTail);
      if (headDistance <= 12 && (!best || headDistance < best.distance)) best={bone:bone,part:"head",distance:headDistance};
      if (tailDistance <= 12 && (!best || tailDistance < best.distance)) best={bone:bone,part:"tail",distance:tailDistance};
      var a = API.sceneToStagePoint(bone.restHead);
      var b = API.sceneToStagePoint(bone.restTail);
      var lineDistance = distancePointToSegment(stagePoint,a,b);
      if (lineDistance <= 8 && (!best || lineDistance < best.distance)) best={bone:bone,part:"line",distance:lineDistance};
    });
    return best;
  }

  function onCanvasPointerDown(event){
    if (!state.open || !state.activeRig) return;
    if (state.currentSceneKey !== sceneKey()) return;
    if (state.mode === "add") {
      event.preventDefault();
      event.stopImmediatePropagation();
      var start = toScenePoint(event);
      state.addPreview = {head:start,tail:start};
      var surface = API.overlay || API.canvas;
      try { if (surface.setPointerCapture) surface.setPointerCapture(event.pointerId); } catch (captureError) {}
      state.drag = {type:"add",pointerId:event.pointerId};
      return;
    }
    var hit = closestBoneHit(event);
    if (!hit) return;
    event.preventDefault();
    event.stopImmediatePropagation();
    state.selectedBoneId = hit.bone.id;
    renderPanel();
    if (hit.part === "head" || hit.part === "tail") {
      API.pushUndo();
      state.drag = {type:"edit",bone:hit.bone,part:hit.part,pointerId:event.pointerId};
      var surface = API.overlay || API.canvas;
      try { if (surface.setPointerCapture) surface.setPointerCapture(event.pointerId); } catch (captureError) {}
    }
  }

  function onCanvasPointerMove(event){
    if (!state.drag) return;
    if (state.drag.pointerId !== undefined && event.pointerId !== state.drag.pointerId) return;
    event.preventDefault();
    event.stopImmediatePropagation();
    var point = toScenePoint(event);
    if (state.drag.type === "add" && state.addPreview) state.addPreview.tail = point;
    if (state.drag.type === "edit") {
      if (state.drag.part === "head") state.drag.bone.restHead = {x:round3(point.x),y:round3(point.y)};
      else state.drag.bone.restTail = {x:round3(point.x),y:round3(point.y)};
      bindRig(false);
    }
  }

  function onCanvasPointerUp(event){
    if (!state.drag) return;
    if (state.drag.pointerId !== undefined && event.pointerId !== state.drag.pointerId) return;
    event.preventDefault();
    event.stopImmediatePropagation();
    if (state.drag.type === "add" && state.addPreview) addBone(state.addPreview.head,state.addPreview.tail);
    state.addPreview = null;
    state.drag = null;
  }

  function drawBone(ctx,bone,transform,selected){
    var head = API.sceneToStagePoint(transform.head);
    var tail = API.sceneToStagePoint(transform.tail);
    ctx.save();
    ctx.lineWidth = selected ? 5 : 3;
    ctx.strokeStyle = selected ? "#0e9c86" : "#626d79";
    ctx.fillStyle = selected ? "#0e9c86" : "#f8f9fa";
    ctx.beginPath();
    ctx.moveTo(head.x,head.y);
    ctx.lineTo(tail.x,tail.y);
    ctx.stroke();
    ctx.beginPath();
    ctx.arc(head.x,head.y,selected?7:6,0,Math.PI*2);
    ctx.fill();
    ctx.strokeStyle = "#26303b";
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.fillRect(tail.x-6,tail.y-6,12,12);
    ctx.strokeRect(tail.x-6,tail.y-6,12,12);
    ctx.restore();
  }

  function overlayRenderer(ctx){
    if (!state.open || !state.activeRig || state.currentSceneKey !== sceneKey()) return;
    var rig = state.activeRig;
    var transforms = posedTransformMap(rig,currentTime());
    rig.bones.forEach(function(bone){ drawBone(ctx,bone,transforms[bone.id],bone.id===state.selectedBoneId); });
    if (state.addPreview) {
      drawBone(ctx,{id:"preview"},{head:state.addPreview.head,tail:state.addPreview.tail},true);
    }
    var frame = currentFrame();
    if (frame !== state.lastFrame) {
      state.lastFrame = frame;
      if (state.panel) {
        var frameNode = state.panel.querySelector("[data-db-frame]");
        if (frameNode) frameNode.textContent = String(frame);
      }
    }
  }

  }
