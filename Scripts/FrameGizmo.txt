    throw new Error("Frame Gizmo requires Luke Animate v104 or later.");

  }

    if(typeof LukeAnimate.setConfiguredToolActive === "function"){
      LukeAnimate.setConfiguredToolActive(TOOL_GROUP_ID, TOOL_ITEM_ID, !!active);
    }

    existingPanel.style.display = "flex";

    if(typeof LukeAnimate.setSelectionInspectorSuppressed === "function") LukeAnimate.setSelectionInspectorSuppressed(PLUGIN_ID, true);
    LukeAnimate.bringFloatingPanelToFront(existingPanel);
    LukeAnimate.clampFloatingPanelToViewport(existingPanel);

    candidateSignature:"",

  function number(value, fallback){
    const parsed = Number(value);
    return Number.isFinite(parsed) ? parsed : fallback;
  }

  function clamp(value, minimum, maximum){
    return Math.max(minimum, Math.min(maximum, value));
  }

  function escapeHtml(value){
    return String(value == null ? "" : value)
      .replace(/&/g,"&amp;")
      .replace(/</g,"&lt;")
      .replace(/>/g,"&gt;")
      .replace(/"/g,"&quot;")
      .replace(/'/g,"&#39;");
  }

  function deepClone(value){
    return JSON.parse(JSON.stringify(value));
  }

  function allSceneObjects(){
    const result=[];
    function visit(object){
      if(!object) return;
      result.push(object);
      if(object.type === "group" && Array.isArray(object.children)) object.children.forEach(visit);
    }
    (LukeAnimate.getDrawObjects() || []).forEach(visit);
    return result;
  }

  function findObjectById(id){
    return allSceneObjects().find(object=>String(object.id || "") === String(id || "")) || null;
  }

  function gizmos(){
    return allSceneObjects().filter(object=>object && object.type === GIZMO_TYPE);
  }

  function clipsOnStageWithKeys(){
    const clips = LukeAnimate.getClips ? LukeAnimate.getClips() : {};
    return allSceneObjects().filter(object=>{
      if(!object || object.type !== "clipinstance" || !clips[object.clipId]) return false;
      const times = LukeAnimate.getClipInternalKeyframeTimes(object.clipId);
      return Array.isArray(times) && times.length > 0;
    }).map(object=>({
      instance:object,
      clip:clips[object.clipId],
      times:LukeAnimate.getClipInternalKeyframeTimes(object.clipId)
    }));
  }

  function objectName(object){
    if(!object) return "Unknown object";
    return String(object.name || object.identifier || object.id || "Object");
  }

  function candidateLabel(candidate){
    const clipName = candidate.clip && candidate.clip.name ? candidate.clip.name : candidate.instance.clipId;
    const instanceName = objectName(candidate.instance);
    return instanceName === clipName ? clipName : instanceName + " — " + clipName;
  }

  function frameFromTime(time){
    return typeof LukeAnimate.timeToFrame === "function"
      ? LukeAnimate.timeToFrame(time)
      : Math.max(1,Math.round(number(time,0) * Math.max(1,number(LukeAnimate.fps,30))) + 1);
  }

  function timeFromFrame(frame){
    return typeof LukeAnimate.frameToTime === "function"
      ? LukeAnimate.frameToTime(frame)
      : (Math.max(1,Math.round(number(frame,1)))-1)/Math.max(1,number(LukeAnimate.fps,30));
  }

  function synchronisedLabels(clipId, times){
    const existing = LukeAnimate.getClipFrameLabels ? LukeAnimate.getClipFrameLabels(clipId) : [];
    const byFrame = new Map();
    (existing || []).forEach(label=>{
      const frame = Math.max(1,Math.round(number(label.frame,frameFromTime(label.time))));
      if(!byFrame.has(frame)) byFrame.set(frame,label);
    });
    return (times || []).map(time=>{
      const frame=frameFromTime(time);
      const old=byFrame.get(frame);
      return {
        frame,
        time:timeFromFrame(frame),
        name:String(old && old.name ? old.name : "Frame " + frame)
      };
    });
  }

  function validateLabels(labels){
    const names=new Set();
    for(const label of labels){
      const name=String(label && label.name || "").trim();
      if(!name) return "Every internal keyframe needs a name.";
      const key=name.toLowerCase();
      if(names.has(key)) return "Stop names must be unique.";
      names.add(key);
    }
    return "";
  }

  function selectedCandidate(){
    const id=String(state.targetInstanceId || "");
    return clipsOnStageWithKeys().find(candidate=>String(candidate.instance.id)===id) || null;
  }

  function activeGizmo(){
    const direct=findObjectById(state.activeGizmoId);
    if(direct && direct.type===GIZMO_TYPE) return direct;
    const selected=(LukeAnimate.getSelectedDrawObjects ? LukeAnimate.getSelectedDrawObjects() : []).find(object=>object && object.type===GIZMO_TYPE);
    if(selected){ state.activeGizmoId=selected.id; return selected; }
    const first=gizmos()[0] || null;
    if(first) state.activeGizmoId=first.id;
    return first;
  }

  function stopCount(gizmo){
    return Math.max(1,Array.isArray(gizmo && gizmo.stops) ? gizmo.stops.length : 0);
  }

  function gizmoHeight(gizmo){
    return GIZMO_TOP + GIZMO_BOTTOM + Math.max(1,stopCount(gizmo)-1) * STOP_SPACING;
  }

  function refreshEditorBounds(gizmo){
    if(!gizmo) return;
    const x=number(gizmo.x,80), y=number(gizmo.y,80);
    gizmo.editorBounds={x1:x,y1:y,x2:x+GIZMO_WIDTH,y2:y+gizmoHeight(gizmo)};
  }

  function heldStopIndex(gizmo,time){
    let index=clamp(Math.round(number(gizmo && gizmo.stopIndex,0)),0,stopCount(gizmo)-1);
    const keys=(Array.isArray(gizmo && gizmo.keyframes)?gizmo.keyframes:[])
      .filter(key=>Number.isFinite(Number(key.time)))
      .slice().sort((a,b)=>number(a.time,0)-number(b.time,0));
    for(const key of keys){
      if(number(key.time,0) <= number(time,0) + 0.000001){
        if(key.stopIndex !== undefined) index=clamp(Math.round(number(key.stopIndex,index)),0,stopCount(gizmo)-1);
      }else break;
    }
    return index;
  }

  function currentStop(gizmo,time){
    const index=heldStopIndex(gizmo,time);
    const stops=Array.isArray(gizmo && gizmo.stops)?gizmo.stops:[];
    return {index,stop:stops[index] || null};
  }

  function findControllingGizmo(clipInstance){
    return gizmos().find(gizmo=>gizmo.enabled !== false && String(gizmo.targetObjectId || "")===String(clipInstance && clipInstance.id || "")) || null;
  }

  LukeAnimate.registerObjectPanelOwner(GIZMO_TYPE,"animate");
  LukeAnimate.registerObjectEvaluator(PLUGIN_ID,function(object,time,effective){
    if(!object) return effective;
    if(object.type===GIZMO_TYPE){
      refreshEditorBounds(object);
      const current=currentStop(object,time);
      return Object.assign({},effective,{stopIndex:current.index,stopName:current.stop ? current.stop.name : ""});
    }
    if(object.type==="clipinstance"){
      const controller=findControllingGizmo(object);
      if(!controller) return effective;
      const current=currentStop(controller,time);
      if(!current.stop) return effective;
      return Object.assign({},effective,{
        clipStopTime:number(current.stop.time,0),
        clipStopFrame:Math.max(1,Math.round(number(current.stop.frame,frameFromTime(current.stop.time)))),
        clipStopName:String(current.stop.name || "")
      });
    }
    return effective;
  });

  function scenePointFromEvent(event){
    const canvas=LukeAnimate.canvas;
    const rect=canvas.getBoundingClientRect();
    const stage={
      x:(event.clientX-rect.left)*canvas.width/Math.max(1,rect.width),
      y:(event.clientY-rect.top)*canvas.height/Math.max(1,rect.height)
    };
    return LukeAnimate.stageToScenePoint ? LukeAnimate.stageToScenePoint(stage) : stage;
  }

  function stopY(gizmo,index){
    return number(gizmo.y,80)+GIZMO_TOP+index*STOP_SPACING;
  }

  function pivotPoint(gizmo){
    return {x:number(gizmo.x,80)+35,y:number(gizmo.y,80)+gizmoHeight(gizmo)/2};
  }

  function leverPoint(gizmo,index){
    return {x:number(gizmo.x,80)+101,y:stopY(gizmo,index)};
  }

  function hitGizmo(point){
    const list=gizmos().slice().reverse();
    for(const gizmo of list){
      refreshEditorBounds(gizmo);
      const b=gizmo.editorBounds;
      if(point.x>=b.x1-8 && point.x<=b.x2+8 && point.y>=b.y1-8 && point.y<=b.y2+8) return gizmo;
    }
    return null;
  }

  function nearestStopIndex(gizmo,sceneY){
    return clamp(Math.round((sceneY-(number(gizmo.y,80)+GIZMO_TOP))/STOP_SPACING),0,stopCount(gizmo)-1);
  }

  function setStatus(message,type){
    if(!state.panel) return;
    const node=state.panel.querySelector("[data-fg-status]");
    if(!node) return;
    node.textContent=String(message || "");
    node.className="fg-status"+(type?" "+type:"");
    if(type==="bad") LukeAnimate.warn(message);
    else if(message) LukeAnimate.log(message);
  }

  function setGizmoStop(gizmo,index,announce){
    if(!gizmo || !Array.isArray(gizmo.stops) || !gizmo.stops.length) return;
    const next=clamp(Math.round(number(index,0)),0,gizmo.stops.length-1);
    if(typeof LukeAnimate.setObjectFieldForCurrentTime === "function"){
      LukeAnimate.setObjectFieldForCurrentTime(gizmo,"stopIndex",next,{easing:"hold"});
    }else{
      LukeAnimate.pushUndo();
      gizmo.stopIndex=next;
      LukeAnimate.refreshUI();
    }
    const stop=gizmo.stops[next];
    if(announce) setStatus("“"+stop.name+"” set at frame "+LukeAnimate.getCurrentFrame()+"."+(LukeAnimate.getAutoKeyframe() ? " A hold key was recorded." : " Auto Keyframe is off."),"good");
    renderPanelLever();
  }

  function refreshCandidates(force){
    if(!state.panel) return;
    const candidates=clipsOnStageWithKeys();
    const signature=candidates.map(candidate=>candidate.instance.id+":"+candidate.instance.clipId+":"+candidate.times.map(frameFromTime).join(",")).join("|");
    const select=state.panel.querySelector("#fg-target");
    if(force || signature!==state.candidateSignature){
      state.candidateSignature=signature;
      const previous=state.targetInstanceId || select.value;
      select.innerHTML="";
      if(!candidates.length){
        const option=document.createElement("option");
        option.value="";option.textContent="No keyed Clips on this stage";select.appendChild(option);
        state.targetInstanceId="";
      }else{
        candidates.forEach(candidate=>{
          const option=document.createElement("option");
          option.value=String(candidate.instance.id);
          option.textContent=candidateLabel(candidate);
          select.appendChild(option);
        });
        state.targetInstanceId=candidates.some(candidate=>String(candidate.instance.id)===String(previous)) ? String(previous) : String(candidates[0].instance.id);
        select.value=state.targetInstanceId;
      }
      loadDraftLabels();
      renderNameRows();
    }
    refreshGizmoSelect(force);
  }

  function loadDraftLabels(){
    const candidate=selectedCandidate();
    state.draftLabels=candidate ? synchronisedLabels(candidate.instance.clipId,candidate.times) : [];
  }

  function renderNameRows(){
    if(!state.panel) return;
    const host=state.panel.querySelector("[data-fg-names]");
    host.innerHTML="";
    if(!state.draftLabels.length){
      host.innerHTML='<div class="fg-empty">The selected Clip must contain at least one internal object, StickRig or Dem Bones keyframe.</div>';
      return;
    }
    state.draftLabels.forEach((label,index)=>{
      const row=document.createElement("div");
      row.className="fg-name-row";
      row.innerHTML='<div class="fg-frame-chip">Frame '+label.frame+'</div><input type="text" value="'+escapeHtml(label.name)+'" aria-label="Name for internal frame '+label.frame+'">';
      row.querySelector("input").addEventListener("input",event=>{state.draftLabels[index].name=event.target.value;});
      host.appendChild(row);
    });
  }

  function saveNames(){
    const candidate=selectedCandidate();
    if(!candidate){setStatus("Choose a keyed Clip instance first.","bad");return false;}
    const problem=validateLabels(state.draftLabels);
    if(problem){setStatus(problem,"bad");return false;}
    LukeAnimate.setClipFrameLabels(candidate.instance.clipId,state.draftLabels);
    candidate.clip.stopClip=true;
    gizmos().filter(gizmo=>gizmo.targetClipId===candidate.instance.clipId).forEach(gizmo=>{
      gizmo.stops=deepClone(state.draftLabels);
      gizmo.stopIndex=clamp(Math.round(number(gizmo.stopIndex,0)),0,Math.max(0,gizmo.stops.length-1));
      refreshEditorBounds(gizmo);
    });
    LukeAnimate.refreshUI();
    renderPanelLever();
    setStatus(state.draftLabels.length+" named stops saved to “"+candidate.clip.name+"”.","good");
    return true;
  }

  function defaultGizmoPosition(candidate){
    const effective=LukeAnimate.getEffectiveObject(candidate.instance,LukeAnimate.time) || candidate.instance;
    const x=number(effective.x,80)+90;
    const y=number(effective.y,80)-Math.max(50,(GIZMO_TOP+GIZMO_BOTTOM+(state.draftLabels.length-1)*STOP_SPACING)/2);
    return {x:clamp(x,20,Math.max(20,LukeAnimate.width-GIZMO_WIDTH-20)),y:clamp(y,20,Math.max(20,LukeAnimate.height-160))};
  }

  function addGizmo(){
    const candidate=selectedCandidate();
    if(!candidate){setStatus("Choose a Clip instance that contains internal keyframes.","bad");return;}
    const problem=validateLabels(state.draftLabels);
    if(problem){setStatus(problem,"bad");return;}
    if(!saveNames()) return;
    const duplicate=gizmos().find(gizmo=>String(gizmo.targetObjectId)===String(candidate.instance.id));
    if(duplicate){
      state.activeGizmoId=duplicate.id;
      LukeAnimate.selectDrawObject(duplicate);
      refreshGizmoSelect(true);
      renderPanelLever();
      setStatus("That Clip instance already has a Frame Gizmo.","bad");
      return;
    }
    const position=defaultGizmoPosition(candidate);
    const gizmo={
      type:GIZMO_TYPE,
      name:"Frame Gizmo — "+candidate.clip.name,
      panelId:"animate",
      editorOnly:true,
      enabled:true,
      x:position.x,
      y:position.y,
      tx:0,ty:0,tscale:1,tscaleX:1,tscaleY:1,trotation:0,opacity:1,
      targetObjectId:candidate.instance.id,
      targetClipId:candidate.instance.clipId,
      stops:deepClone(state.draftLabels),
      stopIndex:0,
      keyframeFields:["stopIndex"],
      forceHoldKeyframes:true,
      keyframes:[]
    };
    refreshEditorBounds(gizmo);
    LukeAnimate.addDrawObject(gizmo,"animate");
    state.activeGizmoId=gizmo.id;
    LukeAnimate.selectDrawObject(gizmo);
    refreshGizmoSelect(true);
    renderPanelLever();
    setStatus("Frame Gizmo added for “"+candidate.clip.name+"”. Turn on Auto Keyframe and move its lever.","good");
  }

  function refreshGizmoSelect(force){
    if(!state.panel) return;
    const list=gizmos();
    const signature=list.map(gizmo=>gizmo.id+":"+gizmo.targetObjectId+":"+stopCount(gizmo)).join("|");
    if(!force && signature===state.gizmoSignature) return;
    state.gizmoSignature=signature;
    const select=state.panel.querySelector("#fg-gizmo");
    const previous=state.activeGizmoId || select.value;
    select.innerHTML="";
    if(!list.length){
      const option=document.createElement("option");option.value="";option.textContent="No Frame Gizmos";select.appendChild(option);state.activeGizmoId="";
    }else{
      list.forEach(gizmo=>{
        const target=findObjectById(gizmo.targetObjectId);
        const option=document.createElement("option");option.value=gizmo.id;option.textContent=gizmo.name+(target?"":" — target missing");select.appendChild(option);
      });
      state.activeGizmoId=list.some(gizmo=>gizmo.id===previous)?previous:list[0].id;
      select.value=state.activeGizmoId;
    }
    renderPanelLever();
  }

  function renderPanelLever(){
    if(!state.panel) return;
    const host=state.panel.querySelector("[data-fg-panel-lever]");
    const buttons=state.panel.querySelector("[data-fg-stop-buttons]");
    const gizmo=activeGizmo();
    host.innerHTML="";buttons.innerHTML="";
    if(!gizmo || !Array.isArray(gizmo.stops) || !gizmo.stops.length){
      host.innerHTML='<div class="fg-empty" style="margin:18px">Add or select a Frame Gizmo.</div>';
      return;
    }
    const current=currentStop(gizmo,LukeAnimate.time);
    const count=gizmo.stops.length;
    const panelHeight=Math.max(150,36+(count-1)*28);
    host.style.height=panelHeight+"px";
    const rail=document.createElement("div");rail.className="fg-lever-rail";host.appendChild(rail);
    const pivot=document.createElement("div");pivot.className="fg-lever-pivot";host.appendChild(pivot);
    const usable=panelHeight-36;
    const yFor=index=>18+(count<=1?usable/2:index*usable/(count-1));
    const selectedY=yFor(current.index);
    const pivotX=41,pivotY=panelHeight/2,endX=102;
    const dx=endX-pivotX,dy=selectedY-pivotY;
    const arm=document.createElement("div");arm.className="fg-lever-arm";arm.style.width=Math.hypot(dx,dy)+"px";arm.style.transform="rotate("+(Math.atan2(dy,dx)*180/Math.PI)+"deg)";host.appendChild(arm);
    gizmo.stops.forEach((stop,index)=>{
      const row=document.createElement("div");row.className="fg-panel-stop"+(index===current.index?" active":"");row.style.top=(yFor(index)-12)+"px";row.textContent=stop.name+" · F"+stop.frame;host.appendChild(row);
      const button=document.createElement("button");button.type="button";button.className=index===current.index?"active":"";button.textContent=stop.name;button.title="Show named Clip frame “"+stop.name+"” (frame "+stop.frame+")";button.addEventListener("click",()=>setGizmoStop(gizmo,index,true));buttons.appendChild(button);
    });
  }

  function removeObjectFromTree(id){
    function removeFrom(list){
      if(!Array.isArray(list)) return false;
      const index=list.findIndex(object=>String(object && object.id || "")===String(id));
      if(index>=0){list.splice(index,1);return true;}
      for(const object of list){if(object && object.type==="group" && removeFrom(object.children)) return true;}
      return false;
    }
    return removeFrom(LukeAnimate.getDrawObjects());
  }

  function deleteActiveGizmo(){
    const gizmo=activeGizmo();
    if(!gizmo){setStatus("There is no Frame Gizmo to delete.","bad");return;}
    LukeAnimate.pushUndo();
    removeObjectFromTree(gizmo.id);
    state.activeGizmoId="";
    LukeAnimate.refreshUI();
    refreshGizmoSelect(true);
    setStatus("Frame Gizmo deleted. The Clip and its named frames were not deleted.","good");
  }

  function reassociateActiveGizmo(){
    const gizmo=activeGizmo();
    const candidate=selectedCandidate();
    if(!gizmo || !candidate){setStatus("Select both an active Frame Gizmo and a keyed Clip instance.","bad");return;}
    const problem=validateLabels(state.draftLabels);
    if(problem){setStatus(problem,"bad");return;}
    saveNames();
    LukeAnimate.pushUndo();
    gizmo.targetObjectId=candidate.instance.id;
    gizmo.targetClipId=candidate.instance.clipId;
    gizmo.name="Frame Gizmo — "+candidate.clip.name;
    gizmo.stops=deepClone(state.draftLabels);
    gizmo.stopIndex=clamp(Math.round(number(gizmo.stopIndex,0)),0,Math.max(0,gizmo.stops.length-1));
    refreshEditorBounds(gizmo);
    LukeAnimate.refreshUI();
    refreshGizmoSelect(true);
    setStatus("The active gizmo now controls “"+candidate.clip.name+"”.","good");
  }

  function canvasPointerDown(event){
    if(event.button!==0 || event.ctrlKey || event.metaKey || event.altKey) return;
    const point=scenePointFromEvent(event);
    const gizmo=hitGizmo(point);
    if(!gizmo) return;
    event.preventDefault();event.stopPropagation();
    state.activeGizmoId=gizmo.id;
    LukeAnimate.selectDrawObject(gizmo);
    refreshGizmoSelect(true);
    const x=number(gizmo.x,80),y=number(gizmo.y,80);
    const current=currentStop(gizmo,LukeAnimate.time);
    const lever=leverPoint(gizmo,current.index);
    const onLever=Math.hypot(point.x-lever.x,point.y-lever.y)<=18 || point.x>=x+86;
    if(onLever){
      state.drag={mode:"lever",gizmoId:gizmo.id};
      setGizmoStop(gizmo,nearestStopIndex(gizmo,point.y),false);
    }else{
      state.drag={mode:"move",gizmoId:gizmo.id,startX:point.x,startY:point.y,originX:x,originY:y,undo:false};
    }
    LukeAnimate.canvas.setPointerCapture && LukeAnimate.canvas.setPointerCapture(event.pointerId);
  }

  function canvasPointerMove(event){
    if(!state.drag) return;
    const gizmo=findObjectById(state.drag.gizmoId);
    if(!gizmo){state.drag=null;return;}
    const point=scenePointFromEvent(event);
    event.preventDefault();event.stopPropagation();
    if(state.drag.mode==="lever"){
      const next=nearestStopIndex(gizmo,point.y);
      const current=currentStop(gizmo,LukeAnimate.time).index;
      if(next!==current) setGizmoStop(gizmo,next,false);
    }else{
      if(!state.drag.undo){LukeAnimate.pushUndo();state.drag.undo=true;}
      gizmo.x=state.drag.originX+(point.x-state.drag.startX);
      gizmo.y=state.drag.originY+(point.y-state.drag.startY);
      refreshEditorBounds(gizmo);
      LukeAnimate.refreshUI();
    }
  }

  function canvasPointerUp(event){
    if(!state.drag) return;
    const gizmo=findObjectById(state.drag.gizmoId);
    const mode=state.drag.mode;
    state.drag=null;
    if(gizmo && mode==="lever"){
      const current=currentStop(gizmo,LukeAnimate.time);
      const stop=current.stop;
      if(stop) setStatus("“"+stop.name+"” set at frame "+LukeAnimate.getCurrentFrame()+"."+(LukeAnimate.getAutoKeyframe() ? " A hold key was recorded." : " Auto Keyframe is off."),"good");
      renderPanelLever();
    }
    try{LukeAnimate.canvas.releasePointerCapture && LukeAnimate.canvas.releasePointerCapture(event.pointerId);}catch(error){}
  }

  LukeAnimate.canvas.addEventListener("pointerdown",canvasPointerDown,true);
  window.addEventListener("pointermove",canvasPointerMove,true);
  window.addEventListener("pointerup",canvasPointerUp,true);

  function poll(){
    const frame=LukeAnimate.getCurrentFrame ? LukeAnimate.getCurrentFrame() : frameFromTime(LukeAnimate.time);
    if(frame!==state.lastFrame){
      state.lastFrame=frame;
      const node=state.panel && state.panel.querySelector("[data-fg-frame]");
      if(node) node.textContent=String(frame);
      renderPanelLever();
    }
    refreshCandidates(false);
    const selected=(LukeAnimate.getSelectedDrawObjects ? LukeAnimate.getSelectedDrawObjects() : []).find(object=>object && object.type===GIZMO_TYPE);
    if(selected && selected.id!==state.activeGizmoId){state.activeGizmoId=selected.id;refreshGizmoSelect(true);}
  }

  makePanel();
  setToolSelected(true);
  if(typeof LukeAnimate.setSelectionInspectorSuppressed === "function") LukeAnimate.setSelectionInspectorSuppressed(PLUGIN_ID,true);
  refreshCandidates(true);
  refreshGizmoSelect(true);
  state.pollTimer=setInterval(poll,180);
  setStatus("Choose a keyed Clip, name its internal pose frames, then add the Frame Gizmo.","");
})();
