/*
  Luke Animate Frame Gizmo GPT v3.1
  Stage-side named-frame controller for selected Clip instances.

  The controller is shown only while exactly one Clip instance is selected.
  It does not create a visible object on the stage. Its hold keys are stored
  inside the selected Clip instance under _lukeAnimateFrameGizmo.
*/
(function(){
  "use strict";

  const PLUGIN_ID = "lukeanimate-frame-gizmo-v3";
  const STYLE_ID = "lukeanimate-frame-gizmo-v3-style";
  const DOCK_ID = "lukeanimate-frame-gizmo-v3-dock";
  const DATA_KEY = "_lukeAnimateFrameGizmo";
  const VERSION = "3.1";
  const TOOL_GROUP_ID = String(group && group.id ? group.id : "animate");
  const TOOL_ITEM_ID = String(item && (item.id || item.label) ? (item.id || item.label) : "FrameGizmo");
  const TOOL_ICON_URL = String(item && item.iconUrl ? item.iconUrl : "https://lukeo25.github.io/LukeAnimate/SVGs/FrameGizmo.svg");

  if(!LukeAnimate || typeof LukeAnimate.getSelectedDrawObjects !== "function" || typeof LukeAnimate.registerObjectEvaluator !== "function"){
    throw new Error("Frame Gizmo requires Luke Animate v105 or later.");
  }
  if(Number(LukeAnimate.version || 0) < 105){
    throw new Error("Frame Gizmo requires Luke Animate v105 or later.");
  }

  function setToolSelected(active){
    if(typeof LukeAnimate.setConfiguredToolActive === "function"){
      LukeAnimate.setConfiguredToolActive(TOOL_GROUP_ID, TOOL_ITEM_ID, !!active);
    }
  }

  const previousV2 = window.__LukeAnimateFrameGizmoV2;
  if(previousV2 && typeof previousV2.disable === "function") previousV2.disable();
  const previous = window.__LukeAnimateFrameGizmoV3;
  if(previous && typeof previous.enable === "function"){
    previous.enable();
    setToolSelected(true);
    return;
  }

  const state = {
    enabled:true,
    dock:null,
    pollTimer:null,
    selectedId:"",
    renderSignature:"",
    lastParentFrame:-1,
    lastCurrentStopFrame:-1,
    lastAutoKeyframe:null,
    x:null,
    y:null,
    dragging:false,
    dragPointerId:null,
    dragOffsetX:0,
    dragOffsetY:0
  };

  function number(value, fallback){
    const parsed = Number(value);
    return Number.isFinite(parsed) ? parsed : fallback;
  }

  function clamp(value, minimum, maximum){
    return Math.max(minimum, Math.min(maximum, value));
  }

  function escapeHtml(value){
    return String(value == null ? "" : value)
      .replace(/&/g,"&amp;")
      .replace(/</g,"&lt;")
      .replace(/>/g,"&gt;")
      .replace(/"/g,"&quot;")
      .replace(/'/g,"&#39;");
  }

  function selectedClipInstance(){
    const selected = LukeAnimate.getSelectedDrawObjects() || [];
    if(selected.length !== 1) return null;
    const object = selected[0];
    if(!object || object.type !== "clipinstance") return null;
    const clips = LukeAnimate.getClips ? LukeAnimate.getClips() : {};
    return clips && clips[object.clipId] ? object : null;
  }

  function selectedClipDefinition(instance){
    const clips = LukeAnimate.getClips ? LukeAnimate.getClips() : {};
    return instance && clips ? clips[instance.clipId] || null : null;
  }

  function frameFromTime(time){
    return typeof LukeAnimate.timeToFrame === "function"
      ? LukeAnimate.timeToFrame(time)
      : Math.max(1, Math.round(number(time,0) * Math.max(1,number(LukeAnimate.fps,30))) + 1);
  }

  function timeFromFrame(frame){
    return typeof LukeAnimate.frameToTime === "function"
      ? LukeAnimate.frameToTime(frame)
      : (Math.max(1,Math.round(number(frame,1)))-1) / Math.max(1,number(LukeAnimate.fps,30));
  }

  function namedStopsForInstance(instance){
    if(!instance) return [];
    const byFrame = new Map();
    const internalTimes = typeof LukeAnimate.getClipInternalKeyframeTimes === "function"
      ? LukeAnimate.getClipInternalKeyframeTimes(instance.clipId) || []
      : [];

    internalTimes.forEach(time=>{
      const frame = frameFromTime(time);
      byFrame.set(frame, {
        frame,
        time:timeFromFrame(frame),
        name:"Frame " + frame,
        named:false
      });
    });

    const labels = typeof LukeAnimate.getClipFrameLabels === "function"
      ? LukeAnimate.getClipFrameLabels(instance.clipId) || []
      : [];
    labels.forEach(label=>{
      const name = String(label && label.name || "").trim();
      if(!name) return;
      const frame = Math.max(1,Math.round(number(label && label.frame,frameFromTime(label && label.time))));
      byFrame.set(frame, {
        frame,
        time:timeFromFrame(frame),
        name,
        named:true
      });
    });

    return Array.from(byFrame.values()).sort((a,b)=>a.frame-b.frame);
  }

  function ensureControllerData(instance){
    if(!instance[DATA_KEY] || typeof instance[DATA_KEY] !== "object"){
      instance[DATA_KEY] = {
        enabled:false,
        baseFrame:1,
        manualFrame:null,
        keys:[]
      };
    }
    const data = instance[DATA_KEY];
    data.enabled = !!data.enabled;
    data.baseFrame = Math.max(1,Math.round(number(data.baseFrame,1)));
    const manualValue = data.manualFrame;
    data.manualFrame = manualValue === null || manualValue === undefined || manualValue === ""
      ? null
      : (Number.isFinite(Number(manualValue)) ? Math.max(1,Math.round(Number(manualValue))) : null);
    if(!Array.isArray(data.keys)) data.keys = [];
    data.keys = data.keys
      .filter(key=>key && Number.isFinite(Number(key.time)) && Number.isFinite(Number(key.frame)))
      .map(key=>({
        time:Math.max(0,number(key.time,0)),
        frame:Math.max(1,Math.round(number(key.frame,1)))
      }))
      .sort((a,b)=>a.time-b.time);
    return data;
  }

  function frameAtTime(instance, time){
    const data = ensureControllerData(instance);
    if(data.manualFrame !== null) return data.manualFrame;
    let frame = data.baseFrame;
    for(const key of data.keys){
      if(key.time <= number(time,0) + 0.000001) frame = key.frame;
      else break;
    }
    return frame;
  }

  function nearestStop(stops, frame){
    if(!stops.length) return null;
    const exact = stops.find(stop=>stop.frame===frame);
    if(exact) return exact;
    return stops.reduce((best,stop)=>{
      if(!best) return stop;
      return Math.abs(stop.frame-frame) < Math.abs(best.frame-frame) ? stop : best;
    },null);
  }

  function currentStop(instance, stops){
    if(!instance || !stops.length) return null;
    return nearestStop(stops, frameAtTime(instance,LukeAnimate.time));
  }

  function upsertHoldKey(data, time, frame){
    const tolerance = 0.5 / Math.max(1,number(LukeAnimate.fps,30));
    const existing = data.keys.find(key=>Math.abs(key.time-time) <= tolerance);
    if(existing) existing.frame = frame;
    else data.keys.push({time,frame});
    data.keys.sort((a,b)=>a.time-b.time);
  }

  function setStop(instance, stop){
    if(!instance || !stop) return;
    LukeAnimate.pushUndo();
    const data = ensureControllerData(instance);
    data.enabled = true;
    const auto = !!LukeAnimate.getAutoKeyframe();
    if(auto){
      data.manualFrame = null;
      const currentTime = Math.max(0,number(LukeAnimate.time,0));
      if(data.keys.length === 0 && currentTime > 0.0001){
        data.keys.push({time:0,frame:data.baseFrame || stop.frame});
      }
      upsertHoldKey(data,currentTime,stop.frame);
      if(currentTime <= 0.0001) data.baseFrame = stop.frame;
    }else{
      data.baseFrame = stop.frame;
      data.manualFrame = stop.frame;
    }
    LukeAnimate.refreshUI();
    renderDock(true);
  }

  function followTimeline(instance){
    if(!instance) return;
    LukeAnimate.pushUndo();
    const data = ensureControllerData(instance);
    data.enabled = false;
    data.manualFrame = null;
    LukeAnimate.refreshUI();
    renderDock(true);
  }

  function clearKeys(instance, stops){
    if(!instance) return;
    LukeAnimate.pushUndo();
    const data = ensureControllerData(instance);
    data.keys = [];
    data.manualFrame = null;
    data.enabled = false;
    data.baseFrame = stops.length ? stops[0].frame : 1;
    LukeAnimate.refreshUI();
    renderDock(true);
  }

  LukeAnimate.registerObjectEvaluator(PLUGIN_ID,function(object,time,effective){
    if(!object || object.type !== "clipinstance") return effective;
    const data = object[DATA_KEY];
    if(!data || data.enabled !== true) return effective;
    const stops = namedStopsForInstance(object);
    if(!stops.length) return effective;
    const stop = nearestStop(stops,frameAtTime(object,time));
    if(!stop) return effective;
    return Object.assign({},effective,{
      clipStopTime:stop.time,
      clipStopFrame:stop.frame,
      clipStopName:stop.name
    });
  });

  const POSITION_KEY = "lukeanimate-frame-gizmo-v3-position";

  function loadSavedPosition(){
    try{
      const saved = JSON.parse(localStorage.getItem(POSITION_KEY) || "null");
      if(saved && Number.isFinite(Number(saved.x)) && Number.isFinite(Number(saved.y))){
        state.x = Number(saved.x);
        state.y = Number(saved.y);
      }
    }catch(error){}
  }

  function savePosition(){
    try{
      if(Number.isFinite(state.x) && Number.isFinite(state.y)){
        localStorage.setItem(POSITION_KEY,JSON.stringify({x:state.x,y:state.y}));
      }
    }catch(error){}
  }

  function ensureStyle(){
    if(document.getElementById(STYLE_ID)) return;
    const style = document.createElement("style");
    style.id = STYLE_ID;
    style.textContent = `
      #${DOCK_ID}{
        position:fixed;
        width:172px;
        height:162px;
        display:none;
        z-index:1800;
        overflow:visible;
        padding:0;
        margin:0;
        background:transparent;
        border:0;
        border-radius:0;
        box-shadow:none;
        color:#26323e;
        opacity:.58;
        cursor:grab;
        user-select:none;
        touch-action:none;
        transition:opacity .12s ease;
        font-family:Inter,-apple-system,BlinkMacSystemFont,"Segoe UI",sans-serif;
      }
      #${DOCK_ID}:hover{opacity:.84}
      #${DOCK_ID}.fg3-dragging{opacity:.94;cursor:grabbing}
      #${DOCK_ID} *{box-sizing:border-box}
      #${DOCK_ID} .fg3-body{width:172px;height:162px;overflow:visible;padding:0;margin:0;background:transparent}
      #${DOCK_ID} .fg3-dial-wrap{width:172px;height:162px;padding:0;margin:0;overflow:visible;background:transparent}
      #${DOCK_ID} .fg3-dial{display:block;width:172px;height:162px;overflow:visible;user-select:none;background:transparent}
      #${DOCK_ID} .fg3-dial-disc{fill:rgba(99,207,154,.22);stroke:rgba(49,189,123,.72);stroke-width:6}
      #${DOCK_ID} .fg3-pointer-outer{stroke:rgba(99,207,154,.74);stroke-width:18;stroke-linecap:round}

      #${DOCK_ID} .fg3-pointer-disabled{opacity:.28}

      #${DOCK_ID} .fg3-stop-label{fill:#111827;font-size:15px;font-family:Inter,-apple-system,BlinkMacSystemFont,"Segoe UI",sans-serif;pointer-events:none}

    `;
    document.head.appendChild(style);
  }

  function beginDrag(event){
    if(event.button !== undefined && event.button !== 0) return;
    if(event.target && event.target.closest && event.target.closest(".fg3-stop-hit")) return;
    const dock = state.dock;
    if(!dock) return;
    const rect = dock.getBoundingClientRect();
    state.dragging = true;
    state.dragPointerId = event.pointerId;
    state.dragOffsetX = event.clientX - rect.left;
    state.dragOffsetY = event.clientY - rect.top;
    state.x = rect.left;
    state.y = rect.top;
    dock.classList.add("fg3-dragging");
    if(dock.setPointerCapture && event.pointerId !== undefined){
      try{ dock.setPointerCapture(event.pointerId); }catch(error){}
    }
    event.preventDefault();
  }

  function moveDrag(event){
    if(!state.dragging || (state.dragPointerId !== null && event.pointerId !== state.dragPointerId)) return;
    const dock = state.dock;
    const wrap = document.getElementById("stage-wrap");
    if(!dock || !wrap) return;
    const wrapRect = wrap.getBoundingClientRect();
    const rect = dock.getBoundingClientRect();
    const width = rect.width || 172;
    const height = rect.height || 162;
    const margin = 4;
    state.x = clamp(event.clientX - state.dragOffsetX,wrapRect.left + margin,wrapRect.right - width - margin);
    state.y = clamp(event.clientY - state.dragOffsetY,wrapRect.top + margin,wrapRect.bottom - height - margin);
    dock.style.left = Math.round(state.x) + "px";
    dock.style.top = Math.round(state.y) + "px";
    event.preventDefault();
  }

  function endDrag(event){
    if(!state.dragging || (state.dragPointerId !== null && event.pointerId !== state.dragPointerId)) return;
    state.dragging = false;
    state.dragPointerId = null;
    if(state.dock) state.dock.classList.remove("fg3-dragging");
    savePosition();
    event.preventDefault();
  }

  function ensureDock(){
    if(state.dock) return state.dock;
    ensureStyle();
    loadSavedPosition();
    const dock = document.createElement("div");
    dock.id = DOCK_ID;
    dock.setAttribute("aria-label","Moveable Frame Gizmo");
    dock.innerHTML = '<div class="fg3-body" data-fg3-body></div>';
    dock.addEventListener("pointerdown",beginDrag);
    dock.addEventListener("pointermove",moveDrag);
    dock.addEventListener("pointerup",endDrag);
    dock.addEventListener("pointercancel",endDrag);
    document.body.appendChild(dock);
    state.dock = dock;
    return dock;
  }

  function renderDock(force){
    const dock = ensureDock();
    if(!state.enabled){
      dock.style.display = "none";
      return;
    }
    const instance = selectedClipInstance();
    if(!instance){
      state.selectedId = "";
      state.renderSignature = "";
      dock.style.display = "none";
      return;
    }

    const stops = namedStopsForInstance(instance);
    if(!stops.length){
      dock.style.display = "none";
      return;
    }

    const data = ensureControllerData(instance);
    const stop = currentStop(instance,stops);
    const signature = [
      instance.id,
      instance.clipId,
      stops.map(item=>item.frame+":"+item.name+":"+(item.named?1:0)).join("|"),
      data.enabled?1:0,
      data.baseFrame,
      data.manualFrame === null ? "" : data.manualFrame,
      data.keys.map(key=>key.time+":"+key.frame).join("|"),
      stop ? stop.frame : ""
    ].join("~");

    dock.style.display = "block";
    state.selectedId = String(instance.id || "");
    if(!force && signature === state.renderSignature){
      positionDock();
      return;
    }
    state.renderSignature = signature;
    const body = dock.querySelector("[data-fg3-body]");
    body.innerHTML = "";

    const dialWrap = document.createElement("div");
    dialWrap.className = "fg3-dial-wrap";
    const svgNS = "http://www.w3.org/2000/svg";
    const svg = document.createElementNS(svgNS,"svg");
    svg.setAttribute("class","fg3-dial");
    svg.setAttribute("viewBox","0 0 320 300");
    svg.setAttribute("role","group");
    svg.setAttribute("aria-label","Frame selection gizmo");

    const centreX = 160;
    const centreY = 177;
    const radius = 112;
    const visibleStops = stops.length > 9 ? stops.slice(0,9) : stops;
    const visibleActiveIndex = stop ? visibleStops.findIndex(item=>item.frame===stop.frame) : -1;
    const angleForIndex = index=>{
      if(visibleStops.length <= 1) return -90;
      return -155 + (130 * index / (visibleStops.length - 1));
    };
    const pointAt = (angle,radiusValue)=>{
      const radians = angle * Math.PI / 180;
      return {x:centreX + Math.cos(radians)*radiusValue,y:centreY + Math.sin(radians)*radiusValue};
    };
    const createSvg = (tag,attrs)=>{
      const node = document.createElementNS(svgNS,tag);
      Object.entries(attrs || {}).forEach(([key,value])=>node.setAttribute(key,String(value)));
      return node;
    };

    svg.appendChild(createSvg("circle",{cx:centreX,cy:centreY,r:radius,class:"fg3-dial-disc"}));

    const activeIndex = visibleActiveIndex >= 0 ? visibleActiveIndex : 0;
    const activeAngle = angleForIndex(activeIndex);
    const activePoint = pointAt(activeAngle,radius - 4);
    const pointerGroup = createSvg("g",{class:data.enabled ? "" : "fg3-pointer-disabled"});
    pointerGroup.appendChild(createSvg("line",{x1:centreX,y1:centreY,x2:activePoint.x,y2:activePoint.y,class:"fg3-pointer-outer"}));
    pointerGroup.appendChild(createSvg("line",{x1:centreX,y1:centreY,x2:activePoint.x,y2:activePoint.y,class:"fg3-pointer-inner"}));
    svg.appendChild(pointerGroup);

    svg.appendChild(createSvg("circle",{cx:centreX,cy:centreY,r:28,class:"fg3-pivot-outer"}));
    svg.appendChild(createSvg("circle",{cx:centreX,cy:centreY,r:20,class:"fg3-pivot-inner"}));

    visibleStops.forEach((item,index)=>{
      const angle = angleForIndex(index);
      const point = pointAt(angle,radius);
      const labelPoint = pointAt(angle,radius + 34);
      const groupNode = createSvg("g",{
        class:"fg3-stop-hit" + (index===activeIndex && data.enabled ? " active" : ""),
        tabindex:"0",
        role:"button",
        "aria-label":"Show Clip frame " + item.frame + (item.named ? (", " + item.name) : "")
      });
      const title = createSvg("title");
      title.textContent = "Show Clip frame " + item.frame + (item.named ? (" — " + item.name) : "");
      groupNode.appendChild(title);
      groupNode.appendChild(createSvg("circle",{cx:point.x,cy:point.y,r:20,class:"fg3-stop-outer"}));
      groupNode.appendChild(createSvg("circle",{cx:point.x,cy:point.y,r:12,class:"fg3-stop-inner"}));
      const activate = ()=>setStop(instance,item);
      groupNode.addEventListener("click",activate);
      groupNode.addEventListener("keydown",event=>{
        if(event.key === "Enter" || event.key === " "){
          event.preventDefault();
          activate();
        }
      });
      svg.appendChild(groupNode);

      const label = createSvg("text",{
        x:labelPoint.x,
        y:labelPoint.y,
        class:"fg3-stop-label",
        "text-anchor":labelPoint.x < centreX - 12 ? "end" : (labelPoint.x > centreX + 12 ? "start" : "middle"),
        "dominant-baseline":"middle"
      });
      const rawLabel = item.named ? item.name : ("Fr" + item.frame);
      label.textContent = rawLabel.length > 13 ? rawLabel.slice(0,12) + "…" : rawLabel;
      svg.appendChild(label);

      if(item.named){
        const frameLabel = createSvg("text",{
          x:labelPoint.x,
          y:labelPoint.y + 13,
          class:"fg3-stop-frame",
          "text-anchor":labelPoint.x < centreX - 12 ? "end" : (labelPoint.x > centreX + 12 ? "start" : "middle"),
          "dominant-baseline":"middle"
        });
        frameLabel.textContent = "Fr" + item.frame;
        svg.appendChild(frameLabel);
      }
    });

    dialWrap.appendChild(svg);
    body.appendChild(dialWrap);
    requestAnimationFrame(positionDock);
  }

  function positionDock(){
    const dock = state.dock;
    if(!dock || dock.style.display === "none" || state.dragging) return;
    const wrap = document.getElementById("stage-wrap");
    const shell = document.getElementById("canvas-shell");
    if(!wrap || !shell) return;
    const wrapRect = wrap.getBoundingClientRect();
    const shellRect = shell.getBoundingClientRect();
    const rect = dock.getBoundingClientRect();
    const width = rect.width || 172;
    const height = rect.height || 162;
    const margin = 8;

    if(!Number.isFinite(state.x) || !Number.isFinite(state.y)){
      state.x = Math.min(wrapRect.right - width - margin,shellRect.right - width - margin);
      state.y = shellRect.top + margin;
    }

    state.x = clamp(state.x,wrapRect.left + 4,wrapRect.right - width - 4);
    state.y = clamp(state.y,wrapRect.top + 4,wrapRect.bottom - height - 4);
    dock.style.left = Math.round(state.x) + "px";
    dock.style.top = Math.round(state.y) + "px";
  }

  function poll(){
    const instance = selectedClipInstance();
    const parentFrame = LukeAnimate.getCurrentFrame ? LukeAnimate.getCurrentFrame() : frameFromTime(LukeAnimate.time);
    const auto = !!LukeAnimate.getAutoKeyframe();
    let currentFrame = -1;
    if(instance){
      const stop = currentStop(instance,namedStopsForInstance(instance));
      currentFrame = stop ? stop.frame : -1;
    }
    const selectionChanged = String(instance && instance.id || "") !== state.selectedId;
    const timeChanged = parentFrame !== state.lastParentFrame || currentFrame !== state.lastCurrentStopFrame;
    const autoChanged = auto !== state.lastAutoKeyframe;
    state.lastParentFrame = parentFrame;
    state.lastCurrentStopFrame = currentFrame;
    state.lastAutoKeyframe = auto;
    renderDock(selectionChanged || timeChanged || autoChanged);
  }

  function enable(){
    state.enabled = true;
    setToolSelected(true);
    renderDock(true);
  }

  function disable(){
    state.enabled = false;
    if(state.dock) state.dock.style.display = "none";
    setToolSelected(false);
  }

  window.__LukeAnimateFrameGizmoV3 = {enable,disable,render:()=>renderDock(true)};
  ensureDock();
  setToolSelected(true);
  window.addEventListener("resize",positionDock);
  const stageWrap = document.getElementById("stage-wrap");
  if(stageWrap) stageWrap.addEventListener("scroll",positionDock,{passive:true});
  state.pollTimer = setInterval(poll,140);
  poll();
})();
