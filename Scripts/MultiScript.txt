  function naturalSortFiles(files) {
    return files.slice().sort(function (a, b) {
      return String(a.name || "").localeCompare(String(b.name || ""), undefined, {
        numeric: true,
        sensitivity: "base"
      });
    });
  }

  function updateFileSummary() {
    if (!state.files.length) {
      fileSummary.textContent = "No files selected.";
      return;
    }
    var first = state.files[0].name;
    var last = state.files[state.files.length - 1].name;
    fileSummary.textContent = state.files.length + " image" + (state.files.length === 1 ? "" : "s") + " selected: " + first + (state.files.length > 1 ? "  →  " + last : "");
    setLog("Selected " + state.files.length + " image" + (state.files.length === 1 ? "." : "s.") + "\nNatural order:\n" + state.files.map(function (file, index) {
      return String(index + 1).padStart(3, "0") + "  " + file.name;
    }).join("\n"));
  }

  function setBusy(busy) {
    state.busy = !!busy;
    importButton.disabled = state.busy;
    fileInput.disabled = state.busy;

    destinationSelect.disabled = state.busy;

    fitInput.disabled = state.busy;
    closeButton.disabled = state.busy;
    progressElement.classList.toggle("is-visible", state.busy);
    progressElement.setAttribute("aria-hidden", state.busy ? "false" : "true");
    if (!state.busy) progressBar.style.width = "0%";
  }

  function setProgress(completed, total) {
    var percentage = total > 0 ? Math.max(0, Math.min(100, completed / total * 100)) : 0;
    progressBar.style.width = percentage.toFixed(2) + "%";
  }

  function setLog(message) {
    logElement.textContent = String(message || "");
    logElement.scrollTop = logElement.scrollHeight;
  }

  function log(message) {
    var text = String(message || "");
    logElement.textContent += (logElement.textContent ? "\n" : "") + text;
    logElement.scrollTop = logElement.scrollHeight;
  }

  function logError(error) {
    var message = error && error.message ? error.message : String(error || "Unknown error");
    log("ERROR: " + message);
    console.error("LukeAnimate Multi Script:", error);
  }

  function baseName(fileName) {
    return String(fileName || "Image").replace(/\.[^.]+$/, "") || "Image";
  }

  function readFileAsDataURL(file) {
    return new Promise(function (resolve, reject) {
      var reader = new FileReader();
      reader.onload = function () { resolve(String(reader.result || "")); };
      reader.onerror = function () { reject(reader.error || new Error("Could not read " + file.name)); };
      reader.readAsDataURL(file);
    });
  }

  function getImageDimensions(dataUrl, fileName) {
    return new Promise(function (resolve, reject) {
      var image = new Image();
      image.onload = function () {
        resolve({
          width: Math.max(1, Number(image.naturalWidth || image.width) || 1),
          height: Math.max(1, Number(image.naturalHeight || image.height) || 1)
        });
      };
      image.onerror = function () {
        reject(new Error("The browser could not decode " + fileName));
      };
      image.src = dataUrl;
    });
  }

  async function loadImageFile(file) {
    var dataUrl = await readFileAsDataURL(file);
    var dimensions = await getImageDimensions(dataUrl, file.name);
    return {
      file: file,
      name: baseName(file.name),
      dataUrl: dataUrl,
      width: dimensions.width,
      height: dimensions.height
    };
  }

  function fittedDimensions(width, height) {
    var naturalWidth = Math.max(1, Number(width) || 1);
    var naturalHeight = Math.max(1, Number(height) || 1);
    if (!fitInput.checked) {
      return { width: naturalWidth, height: naturalHeight };
    }
    return {
      width: Math.max(1, Number(API.width) || 1),
      height: Math.max(1, Number(API.height) || 1)
    };
  }

  function timelineFrameCount() {
    return Math.max(1, Math.round(Number(API.getTimelineDuration()) * Math.max(1, Number(API.fps) || 1)));
  }

  function resolveTargetLayer() {
    if (destinationSelect.value === "new") {
      var requestedName = String(layerNameInput.value || "").trim() || "Image Sequence";
      return API.createDrawingLayer(requestedName, { pushUndo: false, refresh: false });
    }
    var layer = API.getActiveDrawingLayer();
    if (!layer) {
      return API.createDrawingLayer("Image Sequence", { pushUndo: false, refresh: false });
    }
    return layer;
  }

  async function buildSequence() {
    if (state.busy) return;
    state.files = naturalSortFiles(Array.prototype.slice.call(fileInput.files || state.files || []));
    if (!state.files.length) {
      throw new Error("Select at least one image file.");
    }

    var startFrame = Math.max(1, Math.round(Number(startFrameInput.value) || 1));
    startFrameInput.value = String(startFrame);
    var endFrame = startFrame + state.files.length - 1;
    var currentFrameCount = timelineFrameCount();

    if (endFrame > currentFrameCount && !extendTimelineInput.checked) {
      throw new Error("The sequence ends on frame " + endFrame + ", but the timeline currently ends on frame " + currentFrameCount + ". Enable Extend Timeline or choose an earlier start frame.");
    }

    setBusy(true);
    setProgress(0, state.files.length);
    setLog("Preparing " + state.files.length + " image" + (state.files.length === 1 ? "" : "s") + "...\nFrames " + startFrame + " to " + endFrame + ".");

    var loadedImages = [];
    var index;
    for (index = 0; index < state.files.length; index += 1) {
      log("Reading " + (index + 1) + "/" + state.files.length + ": " + state.files[index].name);
      loadedImages.push(await loadImageFile(state.files[index]));
      setProgress(index + 0.35, state.files.length);
    }

    API.pushUndo();

    if (endFrame > currentFrameCount) {
      API.ensureTimelineFrameCount(endFrame, { pushUndo: false, refresh: false });
      log("Timeline extended to frame " + endFrame + ".");
    }

    var targetLayer = resolveTargetLayer();
    if (!targetLayer || !targetLayer.id) {
      throw new Error("LukeAnimate could not resolve a destination layer.");
    }
    log("Destination layer: " + String(targetLayer.name || targetLayer.id));

    if (clearRangeInput.checked) {
      API.clearDrawingLayerRange(targetLayer.id, startFrame, endFrame, { pushUndo: false, refresh: false });
      log("Cleared existing layer content from frames " + startFrame + " to " + endFrame + ".");
    }

    var firstObject = null;
    for (index = 0; index < loadedImages.length; index += 1) {
      var loaded = loadedImages[index];
      var frameNumber = startFrame + index;
      var dimensions = fittedDimensions(loaded.width, loaded.height);
      var object;

      if (wrapInput.checked) {
        var clipName = nameInput.checked ? loaded.name : "Image Clip " + String(index + 1).padStart(3, "0");
        var clip = API.createImageClip({
          src: loaded.dataUrl,
          name: loaded.name,
          width: dimensions.width,
          height: dimensions.height
        }, {
          name: clipName,
          imageName: loaded.name,
          width: dimensions.width,
          height: dimensions.height,
          pushUndo: false,
          refresh: false
        });
        object = API.createClipInstance(clip.id, {
          name: clipName,
          x: Number(API.width) / 2,
          y: Number(API.height) / 2
        });
      } else {
        object = API.createImageObject({
          src: loaded.dataUrl,
          name: loaded.name,
          width: dimensions.width,
          height: dimensions.height
        }, {
          name: loaded.name,
          width: dimensions.width,
          height: dimensions.height,
          x: (Number(API.width) - dimensions.width) / 2,
          y: (Number(API.height) - dimensions.height) / 2
        });
      }

      API.setDrawingLayerFrameObject(targetLayer.id, frameNumber, object, {
        pushUndo: false,
        refresh: false
      });

      if (!firstObject) firstObject = object;
      log("Frame " + frameNumber + ": " + loaded.file.name + (wrapInput.checked ? "  [Clip]" : "  [Image]") + (fitInput.checked ? "  [Fit X+Y]" : ""));
      setProgress(index + 1, loadedImages.length);
    }

    API.setFrame(startFrame);
    API.refreshUI();
    if (firstObject && typeof API.selectDrawObject === "function") {
      API.selectDrawObject(firstObject);
      API.refreshUI();
    }

    log("Complete. Imported " + loadedImages.length + " image" + (loadedImages.length === 1 ? "" : "s") + " across frames " + startFrame + " to " + endFrame + ".");
    setBusy(false);
  }

  function makePanelDraggable(element, handle) {
    var drag = null;

    handle.addEventListener("pointerdown", function (event) {
      if (event.button !== 0 || event.target.closest("button")) return;
      var rect = element.getBoundingClientRect();
      drag = {
        pointerId: event.pointerId,
        offsetX: event.clientX - rect.left,
        offsetY: event.clientY - rect.top
      };
      element.style.left = rect.left + "px";
      element.style.top = rect.top + "px";
      element.style.right = "auto";
      handle.setPointerCapture(event.pointerId);
      event.preventDefault();
    });

    handle.addEventListener("pointermove", function (event) {
      if (!drag || event.pointerId !== drag.pointerId) return;
      var maxLeft = Math.max(0, window.innerWidth - element.offsetWidth);
      var maxTop = Math.max(0, window.innerHeight - 44);
      var left = Math.max(0, Math.min(maxLeft, event.clientX - drag.offsetX));
      var top = Math.max(0, Math.min(maxTop, event.clientY - drag.offsetY));
      element.style.left = left + "px";
      element.style.top = top + "px";
    });

    function stopDrag(event) {
      if (!drag || event.pointerId !== drag.pointerId) return;
      try { handle.releasePointerCapture(event.pointerId); } catch (ignore) {}
      drag = null;
    }

    handle.addEventListener("pointerup", stopDrag);
    handle.addEventListener("pointercancel", stopDrag);
  }
