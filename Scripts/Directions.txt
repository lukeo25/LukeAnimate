/*
  Luke Animate Directions Type Tool v1.2
  Native, opt-in eight-direction movement for selected Clip instances.
*/
(function(){
  "use strict";

  const PANEL_ID="lukeanimate-directions-panel";
  const STYLE_ID="lukeanimate-directions-style";
  const TOOL_ID="directions";
  const TOOL_GROUP_ID=String(group&&group.id?group.id:"type");
  const TOOL_ITEM_ID=String(item&&(item.id||item.label)?(item.id||item.label):"Directions");
  const TOOL_ICON_URL=String(item&&item.iconUrl?item.iconUrl:"https://lukeo25.github.io/LukeAnimate/SVGs/Directions.svg");
  const DEFAULT_CODE=`const { config, runtime, delta, vector } = context;
if (!vector.x && !vector.y) return;
const distance = config.speed * 60 * delta;
runtime.offsetX += vector.x * distance;
runtime.offsetY += vector.y * distance;`;

  if(!LukeAnimate||Number(LukeAnimate.version||0)<118||typeof LukeAnimate.setTypeTool!=="function"){
    throw new Error("Directions requires Luke Animate v118 or later.");
  }

  function setToolSelected(active){
    if(typeof LukeAnimate.setConfiguredToolActive==="function"){
      LukeAnimate.setConfiguredToolActive(TOOL_GROUP_ID,TOOL_ITEM_ID,!!active);
    }
  }

  const existing=document.getElementById(PANEL_ID);
  if(existing){
    existing.style.display="flex";
    existing.classList.add("open");
    setToolSelected(true);
    LukeAnimate.bringFloatingPanelToFront(existing);
    LukeAnimate.clampFloatingPanelToViewport(existing);
    return;
  }

  if(!document.getElementById(STYLE_ID)){
    const style=document.createElement("style");
    style.id=STYLE_ID;
    style.textContent=`
      #${PANEL_ID}{position:fixed;z-index:10000;left:28px;top:18px;width:374px;max-width:calc(100vw - 16px);height:min(650px,calc(100vh - 36px));display:flex;flex-direction:column;background:#f1f3f6;border:1px solid #bfc6cf;border-radius:14px;box-shadow:0 20px 60px rgba(0,0,0,.3);color:#27313d;font:13px Inter,-apple-system,BlinkMacSystemFont,Segoe UI,sans-serif;overflow:hidden}
      #${PANEL_ID} *{box-sizing:border-box}
      #${PANEL_ID} .dt-head{display:flex;align-items:center;gap:9px;padding:8px 10px;background:#fff;border-bottom:1px solid #d3d8df;cursor:move;user-select:none}
      #${PANEL_ID} .dt-head img{width:27px;height:27px;object-fit:contain;filter:grayscale(1) brightness(.72)}
      #${PANEL_ID} .dt-title{flex:1} #${PANEL_ID} .dt-title strong{display:block;font-size:14px;color:#29434a} #${PANEL_ID} .dt-title span{font-size:9px;letter-spacing:.11em;text-transform:uppercase;color:#7b8491}
      #${PANEL_ID} .dt-close{width:32px;height:32px;border:1px solid #bbc3cd;border-radius:8px;background:#fff;font-size:20px;color:#566270;cursor:pointer}
      #${PANEL_ID} .dt-body{flex:1;min-height:0;padding:8px 7px 46px;overflow:auto}
      #${PANEL_ID} .dt-selected{padding:8px 9px;margin-bottom:7px;border:1px solid #d2d7de;border-radius:10px;background:#fff;white-space:nowrap;overflow:hidden;text-overflow:ellipsis}
      #${PANEL_ID} .dt-selected strong{color:#0e8f7c}
      #${PANEL_ID} .dt-controls{padding:9px;border:1px solid #d2d7de;border-radius:10px;background:#fff}
      #${PANEL_ID} .dt-section{margin-top:9px;padding-top:9px;border-top:1px solid #d9dfe6}
      #${PANEL_ID} .dt-label{display:flex;justify-content:space-between;align-items:center;margin-bottom:6px;font-weight:700}
      #${PANEL_ID} .dt-speed-row{display:grid;grid-template-columns:1fr 62px;gap:8px;align-items:center}
      #${PANEL_ID} input[type=range]{width:100%;accent-color:#0e9c86}
      #${PANEL_ID} input[type=number]{width:100%;padding:6px;border:1px solid #b9c3ce;border-radius:6px;background:#fff;color:#263442}
      #${PANEL_ID} .dt-check{display:flex;align-items:center;gap:7px;margin:8px 0}
      #${PANEL_ID} .dt-check input{accent-color:#0e9c86}
      #${PANEL_ID} .dt-muted{color:#74808d;font-size:11px;line-height:1.35}
      #${PANEL_ID} details{margin-top:7px;border:1px solid #d2d7de;border-radius:10px;background:#fff;overflow:hidden}
      #${PANEL_ID} summary{padding:8px 9px;cursor:pointer;font-size:11px;letter-spacing:.08em;text-transform:uppercase;font-weight:700;color:#66717e;background:#fff}
      #${PANEL_ID} .dt-code-wrap{padding:9px}
      #${PANEL_ID} textarea{width:100%;min-height:150px;resize:vertical;padding:8px;border:1px solid #aeb8c4;border-radius:6px;background:#17212b;color:#e7f5f2;font:12px/1.45 Consolas,monospace;tab-size:2}
      #${PANEL_ID} .dt-code-tools{display:flex;justify-content:flex-end;margin-top:6px}
      #${PANEL_ID} .dt-actions{display:grid;grid-template-columns:1fr 1fr;gap:8px;margin-top:12px}
      #${PANEL_ID} .dt-btn{padding:7px;border:1px solid #b8c0ca;border-radius:8px;background:#eceff2;color:#4b5766;font-size:10px;cursor:pointer}
      #${PANEL_ID} .dt-btn:hover{border-color:#0e9c86} #${PANEL_ID} .dt-btn.primary{border-color:#0e9c86;background:#0e9c86;color:#fff;font-weight:700}
      #${PANEL_ID} .dt-btn:disabled{opacity:.45;cursor:not-allowed}
      #${PANEL_ID} .dt-status{position:absolute;left:7px;right:7px;bottom:7px;min-height:26px;max-height:42px;overflow:auto;border:1px solid #cbd2da;border-radius:8px;background:rgba(255,255,255,.97);padding:8px 10px;font-size:10px;line-height:1.35;color:#596573}
      #${PANEL_ID} .dt-status.good{border-color:#0e9c86;color:#117d6d}
      #${PANEL_ID} .dt-status.bad{border-color:#c66a6a;color:#a33d3d}
      @media(max-width:560px){#${PANEL_ID}{left:8px;top:8px;width:calc(100vw - 16px);height:calc(100vh - 16px)}}
    `;
    document.head.appendChild(style);
  }

  const panel=document.createElement("section");
  panel.id=PANEL_ID;
  panel.className="open";
  panel.innerHTML=`
    <div class="dt-head" data-floating-panel-drag>
      <img src="${TOOL_ICON_URL.replace(/"/g,"&quot;")}" alt="">
      <div class="dt-title"><strong>Directions</strong><span>Eight-way Clip movement</span></div>
      <button class="dt-close" type="button" title="Close">×</button>
    </div>
    <div class="dt-body">
      <div class="dt-selected"><strong>Selected:</strong> <span data-dt-selected>No Clip selected</span></div>
      <div class="dt-controls">
        <div class="dt-label"><span>Movement speed</span><span data-dt-speed-label>5</span></div>
        <div class="dt-speed-row">
          <input data-dt-speed type="range" min="0.5" max="20" step="0.5" value="5">
          <input data-dt-speed-number type="number" min="0.5" max="20" step="0.5" value="5">
        </div>
        <div class="dt-section">
          <label class="dt-check"><input data-dt-wasd type="checkbox" checked> WASD keys</label>
          <label class="dt-check"><input data-dt-arrows type="checkbox" checked> Arrow keys</label>
          <label class="dt-check"><input data-dt-diagonal type="checkbox" checked> Allow diagonal movement</label>
          <label class="dt-check"><input data-dt-normalize type="checkbox" checked> Keep diagonal speed even</label>
          <label class="dt-check dt-muted" title="This option becomes active when the Solid Type Tool is added"><input data-dt-solid type="checkbox" disabled> Collide with Solids (coming with Solid)</label>
        </div>
      </div>
      <details>
        <summary>View / edit code</summary>
        <div class="dt-code-wrap">
          <p class="dt-muted">The code runs during playback. Available as <code>context</code>: config, runtime, delta, vector, keys, object, readValue and writeValue.</p>
          <textarea data-dt-code spellcheck="false"></textarea>
          <div class="dt-code-tools"><button class="dt-btn" data-dt-reset-code type="button">Reset code</button></div>
        </div>
      </details>
      <p class="dt-muted">Apply Directions, press Play, then use the chosen keys. Pause holds the live position; Stop returns the Clip to its authored position.</p>
      <div class="dt-actions">
        <button class="dt-btn" data-dt-remove type="button">Remove</button>
        <button class="dt-btn primary" data-dt-apply type="button">Apply</button>
      </div>
      <div class="dt-status" data-dt-status></div>
    </div>
  `;
  document.body.appendChild(panel);
  LukeAnimate.registerFloatingPanel(panel);
  LukeAnimate.bringFloatingPanelToFront(panel);
  LukeAnimate.clampFloatingPanelToViewport(panel);
  setToolSelected(true);

  const get=selector=>panel.querySelector(selector);
  const speed=get("[data-dt-speed]");
  const speedNumber=get("[data-dt-speed-number]");
  const speedLabel=get("[data-dt-speed-label]");
  const applyButton=get("[data-dt-apply]");
  const removeButton=get("[data-dt-remove]");
  const selectedLabel=get("[data-dt-selected]");
  const status=get("[data-dt-status]");
  let selectedObject=null;
  let selectedKey="";

  function selectedClip(){
    const selected=LukeAnimate.getSelectedDrawObjects()||[];
    if(selected.length!==1||!selected[0]||selected[0].type!=="clipinstance")return null;
    const clips=LukeAnimate.getClips?LukeAnimate.getClips():{};
    return clips&&clips[selected[0].clipId]?selected[0]:null;
  }

  function objectName(object){
    if(!object)return "No Clip selected";
    const clips=LukeAnimate.getClips?LukeAnimate.getClips():{};
    return String(object.name||(clips[object.clipId]&&clips[object.clipId].name)||"Clip");
  }

  function setStatus(message,kind){
    status.textContent=String(message||"");
    status.className="dt-status"+(kind?" "+kind:"");
  }

  function setSpeed(value){
    const number=Math.max(.5,Math.min(20,Number(value)||5));
    speed.value=String(number);
    speedNumber.value=String(number);
    speedLabel.textContent=String(number);
  }

  function loadSettings(settings){
    const config=settings||{};
    setSpeed(config.speed===undefined?5:config.speed);
    get("[data-dt-wasd]").checked=config.useWASD!==false;
    get("[data-dt-arrows]").checked=config.useArrows!==false;
    get("[data-dt-diagonal]").checked=config.diagonal!==false;
    get("[data-dt-normalize]").checked=config.normalize!==false;
    get("[data-dt-solid]").checked=!!config.solidCollision;
    get("[data-dt-code]").value=String(config.code||DEFAULT_CODE);
  }

  function refreshSelection(force){
    const next=selectedClip();
    const nextKey=next?String(next.id||next.clipId):"";
    if(!force&&nextKey===selectedKey){
      if(next){const error=LukeAnimate.getTypeToolError(next,TOOL_ID);if(error)setStatus("Code error: "+error,"bad");}
      return;
    }
    selectedObject=next;
    selectedKey=nextKey;
    selectedLabel.textContent=objectName(next);
    applyButton.disabled=!next;
    const settings=next?LukeAnimate.getTypeTool(next,TOOL_ID):null;
    removeButton.disabled=!settings;
    loadSettings(settings);
    setStatus(next?(settings?"Directions is attached to this Clip.":"Ready to attach Directions."):"Select one Clip instance on the stage.",next?"good":"");
  }

  function currentSettings(){
    return {
      enabled:true,
      speed:Number(speedNumber.value)||5,
      useWASD:get("[data-dt-wasd]").checked,
      useArrows:get("[data-dt-arrows]").checked,
      diagonal:get("[data-dt-diagonal]").checked,
      normalize:get("[data-dt-normalize]").checked,
      solidCollision:get("[data-dt-solid]").checked,
      code:get("[data-dt-code]").value
    };
  }

  speed.addEventListener("input",()=>setSpeed(speed.value));
  speedNumber.addEventListener("input",()=>setSpeed(speedNumber.value));
  get("[data-dt-reset-code]").addEventListener("click",()=>{get("[data-dt-code]").value=DEFAULT_CODE;setStatus("Directions code reset. Click Apply to save it.","good");});
  applyButton.addEventListener("click",()=>{
    refreshSelection(false);
    if(!selectedObject){setStatus("Select one Clip instance first.","bad");return;}
    LukeAnimate.setTypeTool(selectedObject,TOOL_ID,currentSettings());
    removeButton.disabled=false;
    setStatus("Directions applied. Press Play to test it.","good");
  });
  removeButton.addEventListener("click",()=>{
    refreshSelection(false);
    if(!selectedObject||!LukeAnimate.removeTypeTool(selectedObject,TOOL_ID)){
      setStatus("This Clip does not have Directions attached.","bad");
      return;
    }
    removeButton.disabled=true;
    loadSettings(null);
    setStatus("Directions removed from the Clip.","good");
  });

  const pollTimer=setInterval(()=>refreshSelection(false),250);
  get(".dt-close").addEventListener("click",()=>{
    clearInterval(pollTimer);
    panel.remove();
    setToolSelected(false);
  });

  refreshSelection(true);
})();
